```python
def post_comment_on_merge_request(repo_link_id: str, mr_tab_id: str, mr_item_id: str,
                                  comment_field_id: str, submit_button_id: str, comment_text: str):
    """Navigate to a repository's merge request and post a comment.

    Args:
        repo_link_id: ID of the repository link to click to open the repo.
        mr_tab_id: ID of the merge requests tab or link inside the repo.
        mr_item_id: ID of the specific merge request item to open.
        comment_field_id: ID of the comment input field on the merge request page.
        submit_button_id: ID of the button to submit/post the comment.
        comment_text: The text content to post as the merge request comment.

    Returns:
        dict: A record of the element IDs that were interacted with.

    Examples:
        post_comment_on_merge_request('286', '304', '817', '701', '730', 'close because non reproducible')
    """
    click(repo_link_id)
    click(mr_tab_id)
    click(mr_item_id)
    fill(comment_field_id, comment_text)
    click(submit_button_id)
    return {
        "repo_clicked": repo_link_id,
        "mr_tab_clicked": mr_tab_id,
        "mr_opened": mr_item_id,
        "comment_field": comment_field_id,
        "submit_button": submit_button_id,
    }
```
