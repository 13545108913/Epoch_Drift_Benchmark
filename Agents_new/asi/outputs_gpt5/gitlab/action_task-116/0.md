```python
def open_issue_and_notify(repo_url: str, issues_tab_id: str, issue_view_id: str, issue_number: str, issue_url: str, assignee: str, due_date: str):
    """Open a repository, navigate to an existing issue, and notify the user about it.

    This function navigates to a repository URL, opens the issues view, clicks the specific issue entry,
    and informs the user with a composed message that includes the issue number, link, assignee, and due date.

    Args:
        repo_url: The full URL of the repository (e.g., "http://host/user/repo").
        issues_tab_id: The UI id for the "Issues" tab or link to open the issues listing.
        issue_view_id: The UI id for the specific issue entry or "View" link to open that issue.
        issue_number: The numeric or string identifier of the existing issue (without '#').
        issue_url: The full URL to the specific issue page.
        assignee: The display name of the user currently assigned to the issue.
        due_date: The due date string for the issue (human readable).

    Returns:
        None

    Examples:
        open_issue_and_notify(
            "http://example.com/org/repo",
            "283",
            "856",
            "1",
            "http://example.com/org/repo/issues/1",
            "Abishek S",
            "Jul 18, 2033"
        )
    """
    goto(repo_url)
    click(issues_tab_id)
    click(issue_view_id)
    send_msg_to_user(
        f"An issue with that title already exists: #{issue_number} ({issue_url}). "
        f"It is already assigned to {assignee} and has due date {due_date}. "
        "Do you want me to open that issue, create a new one anyway, or make any updates to the existing issue?"
    )
```
