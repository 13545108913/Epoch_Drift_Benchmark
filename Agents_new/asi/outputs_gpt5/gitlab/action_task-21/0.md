```python
def open_and_report_latest_issue(issue_nav_id: str, latest_issue_id: str, keyword: str):
    """Open the issues area and open the latest created issue that matches a keyword, then inform the user.

    Steps:
    1. Click the navigation/link that leads to issues.
    2. Click the specific issue entry for the latest matching issue.
    3. Send a user-facing message that the issue was opened (message decided inside).

    Args:
        issue_nav_id: ID of the UI element that opens the issues list (e.g., a sidebar or menu link).
        latest_issue_id: ID of the UI element representing the latest matching issue to open.
        keyword: Keyword that was used to match the issue title (used in the confirmation message).

    Returns:
        None

    Examples:
        open_and_report_latest_issue('161', '669', 'feature')
    """
    click(issue_nav_id)
    click(latest_issue_id)
    send_msg_to_user(f"I opened your latest issue with '{keyword}' in the title. Please check its state on the page.")
```
