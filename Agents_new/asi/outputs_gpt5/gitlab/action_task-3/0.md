```python
def search_and_open_repository(search_bar_id: str, repository: str, repo_result_id: str):
    """Search for a repository by name and open it from the results.

    Args:
        search_bar_id: The ID of the search input element to click and type into.
        repository: The full repository name to search for (e.g., "owner/repo").
        repo_result_id: The element ID to click to open the repository from search results.

    Returns:
        None

    Examples:
        search_and_open_repository('304', 'a11yproject/a11yproject.com', '877')
    """
    click(search_bar_id)                       # focus the search input
    fill(search_bar_id, repository)            # type repository name into the search input
    press(search_bar_id, 'Enter')              # submit the search
    click(repo_result_id)                      # open the repository from the results


def compose_and_send_issue_list(repo: str, label: str, issues: list):
    """Format a list of issues and send it to the user as a single message.

    The message is built inside the function (no fixed message is passed in).

    Args:
        repo: Repository name (e.g., "owner/repo") to include in the header.
        label: Label name used to filter issues (used in the header).
        issues: A list of issue dictionaries. Each dict should have 'title', 'url', and 'labels' keys.

    Returns:
        None

    Examples:
        compose_and_send_issue_list(
            'a11yproject/a11yproject.com',
            'help wanted',
            [
                {'title': 'Audit redesign spacing sizes and turn them into variables — #166',
                 'url': 'http://.../issues/166',
                 'labels': 'claimed, good first issue, help wanted, redesign, styling'},
                ...
            ]
        )
    """
    header = f'Here are the open issues tagged "{label}" in {repo} ({len(issues)}):\n\n'
    body_lines = []
    for idx, issue in enumerate(issues, start=1):
        title_line = f"{idx}. {issue['title']}\n   {issue['url']}\n   Labels: {issue['labels']}\n"
        body_lines.append(title_line)
    message = header + "\n".join(body_lines)
    send_msg_to_user(message)
```
