```python
def open_group_creation(menu_id: str, groups_tab_id: str, new_group_button_id: str):
    """Navigate the UI to open the New Group creation form.

    Args:
        menu_id: ID of the main menu or "My Groups" link to click first.
        groups_tab_id: ID of the groups tab or submenu to click.
        new_group_button_id: ID of the "Create New Group" or similar button to open the creation form.

    Returns:
        None

    Examples:
        open_group_creation('76', '83', '222')
    """
    click(menu_id)
    click(groups_tab_id)
    click(new_group_button_id)


def create_group_with_members(group_name_field_id: str, group_name: str, members_input_id: str, member_action_button_id: str, members: list):
    """Fill in a group name, open the members area, and add a list of members.

    Args:
        group_name_field_id: ID of the group name input field.
        group_name: The desired name for the new group.
        members_input_id: ID of the member input field where each member name is typed.
        member_action_button_id: ID of the button used to open member input or confirm adding a member.
        members: A list of member names (strings) to add to the group.

    Returns:
        dict: {'group': group_name, 'members_added': members}

    Examples:
        create_group_with_members('235', 'webagent', '363', '385', ['pandey2000', 'sayakpaul'])
    """
    fill(group_name_field_id, group_name)
    click(member_action_button_id)
    for member in members:
        fill(members_input_id, member)
        keyboard_press("Enter")
        click(member_action_button_id)
    return {"group": group_name, "members_added": members}
```
