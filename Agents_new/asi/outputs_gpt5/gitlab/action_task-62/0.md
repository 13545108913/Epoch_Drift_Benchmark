def search_repository(search_bar_id: str | int, repo_name: str):
    """Search for a repository using the page search bar.

    Args:
        search_bar_id: The ID of the search input element (string or int).
        repo_name: The repository name or query to search for.

    Returns:
        None. After running, the search results page will be displayed.

    Examples:
        search_repository('239', 'MetaSeq')
        search_repository(241, 'example-repo')
    """
    click(search_bar_id)
    fill(search_bar_id, repo_name)
    keyboard_press('Enter')


def fork_repository(repo_link_id: str | int, fork_button_id: str | int, confirm_button_id: str | int):
    """Open a repository and perform the fork flow by clicking the fork controls.

    Args:
        repo_link_id: The ID of the repository link in the search results.
        fork_button_id: The ID of the "Fork" button on the repository page.
        confirm_button_id: The ID of the final confirm/selection button in the fork flow.

    Returns:
        None. After running, the fork flow will be initiated/confirmed.

    Examples:
        fork_repository('307', '475', '415')
        fork_repository(307, 475, 415)
    """
    click(repo_link_id)
    click(fork_button_id)
    click(confirm_button_id)
