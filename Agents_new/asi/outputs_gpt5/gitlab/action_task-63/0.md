def open_repository(repo_link_id: str, code_tab_id: str, file_id: str):
    """Navigate to a repository and open a file by clicking a sequence of elements.

    Args:
        repo_link_id: The element id that opens the repository (e.g., repo link).
        code_tab_id: The element id that opens the code/files view (e.g., "Code" tab).
        file_id: The element id that opens the specific file (e.g., "LICENSE" file).

    Returns:
        None

    Examples:
        open_repository('834', '752', '551')
    """
    click(repo_link_id)
    click(code_tab_id)
    click(file_id)


def replace_file_content(edit_field_id: str, new_content: str, save_button_id: str):
    """Replace the content of an editable file field and save/commit the change.

    Args:
        edit_field_id: The element id of the editable text area for the file.
        new_content: The full text to write into the file.
        save_button_id: The element id of the button to save or commit the change.

    Returns:
        None

    Examples:
        replace_file_content('639', 'MIT License\n\nCopyright ...', '720')
    """
    click(edit_field_id)
    fill(edit_field_id, new_content)
    click(save_button_id)
