```python
def invite_project_guest(search_field_id: str | int, project_query: str,
                         repo_link_id: str | int, repo_settings_id: str | int,
                         members_tab_id: str | int, invite_button_id: str | int,
                         name_field_id: str | int, confirm_button_id: str | int,
                         user_name: str):
    """Search for a project and invite a user to the repository as a guest.

    Steps:
    1. Search for the project and open the repository.
    2. Navigate to repository settings and open the members/invite area.
    3. Fill the invitee name and confirm the invite.

    Args:
        search_field_id: ID of the project search input.
        project_query: Text to search for the project (e.g., 'dotfiles').
        repo_link_id: ID of the repository link in search results.
        repo_settings_id: ID of the repository settings link/button.
        members_tab_id: ID of the members or access management tab/button.
        invite_button_id: ID of the button to start inviting a new member.
        name_field_id: ID of the input field to enter the invitee's name.
        confirm_button_id: ID of the button to finalize the invite.
        user_name: Display name or username to invite.

    Returns:
        A dict indicating the invite action was performed. Example: {'invited': True}

    Examples:
        invite_project_guest('241', 'dotfiles', '288', '395', '407', '490', '490', '512', 'Koushik')
    """
    search_project(search_field_id, project_query)
    click(repo_link_id)
    click(repo_settings_id)
    click(members_tab_id)
    click(invite_button_id)
    fill(name_field_id, user_name)
    noop(1000)
    keyboard_press("Enter")
    click(confirm_button_id)
    return {"invited": True}
```
