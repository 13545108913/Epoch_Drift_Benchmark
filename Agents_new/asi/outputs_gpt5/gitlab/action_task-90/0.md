```python
def open_project_members(account_link_id: str, project_members_url: str):
    """Navigate from the account link to a project's members page.

    Args:
        account_link_id: ID of the "My Account" or account link to click.
        project_members_url: Full URL of the project's members page to open.

    Returns:
        None

    Examples:
        open_project_members('899', 'http://172.26.116.102:8080/byteblaze/dotfiles/-/project_members')
    """
    click(account_link_id)       # click the account link to open account options
    goto(project_members_url)    # navigate directly to the project's members page
    noop(500)                    # small pause to allow navigation to settle


def invite_repo_guest(search_field_id: str, user_name: str, invite_button_id: str):
    """Search for a user in the project members/invite UI and click the invite control.

    Args:
        search_field_id: ID of the search/input field used to find the user.
        user_name: The display name or username to search for and invite.
        invite_button_id: ID of the invite/confirm button to click after searching.

    Returns:
        None

    Examples:
        invite_repo_guest('490', 'Abishek', '512')
    """
    fill(search_field_id, user_name)  # type the name into the member invite/search field
    keyboard_press("Enter")           # submit the search/input
    noop(1000)                        # wait briefly for results to populate
    click(invite_button_id)           # click the invite/confirm control
```
