def open_repository_issues(base_url: str, repo_path: str):
    """Navigate to a repository and its issues page.

    Steps:
    1. Open the repository issues URL.
    2. Open the repository root URL.
    3. Perform a small scroll to ensure the page renders.

    Args:
        base_url: The scheme and host of the instance, e.g. "http://172.26.116.102:8080"
        repo_path: The repository path "owner/repo", e.g. "umano/AndroidSlidingUpPanel"

    Returns:
        None

    Examples:
        open_repository_issues("http://172.26.116.102:8080", "umano/AndroidSlidingUpPanel")
    """
    goto(f"{base_url}/{repo_path}/-/issues")
    goto(f"{base_url}/{repo_path}")
    scroll(0, 0)


def apply_label_filter(label_link_id: str):
    """Apply a label filter by clicking the label element and stabilizing the page.

    Steps:
    1. Click the label link element to filter issues by that label.
    2. Scroll down a bit to load results.
    3. Scroll back to top to stabilize layout.

    Args:
        label_link_id: The element id or selector for the label link on the issues page.

    Returns:
        None

    Examples:
        apply_label_filter("465")
    """
    click(label_link_id)
    scroll(0, 300)
    scroll(0, 0)


def format_issues_summary(repo: str, label: str, open_issues: list[str], closed_count: int | None = None) -> str:
    """Build a human-readable summary message for issues found under a label.

    Steps:
    1. Create a header mentioning the repository and label.
    2. Append an "Open" section listing provided open issue lines.
    3. Optionally append a summary line for closed issues and totals.

    Args:
        repo: Repository identifier "owner/repo".
        label: Label name used for filtering (e.g., "BUG").
        open_issues: List of formatted strings describing open issues, e.g.
                     ["#271: Title ... http://.../issues/271", ...]
        closed_count: Optional count of closed issues with the same label.

    Returns:
        A formatted multi-line string summarizing the issues.

    Examples:
        format_issues_summary(
            "umano/AndroidSlidingUpPanel",
            "BUG",
            ["#271: It's hard to scroll horizontally ... http://.../issues/271"],
            closed_count=27
        )
    """
    header = f"Found issues labeled {label} in {repo}.\n\n"
    open_header = f"Open ({len(open_issues)}):\n"
    parts = [header, open_header]
    for issue_line in open_issues:
        parts.append(f"- {issue_line}\n")
    if closed_count is not None:
        total = len(open_issues) + closed_count
        parts.append(f"\n(There are {closed_count} closed issues with the {label} label, {total} total.)")
    return "".join(parts)
