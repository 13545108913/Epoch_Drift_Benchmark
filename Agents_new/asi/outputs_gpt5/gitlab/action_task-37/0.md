```python
def open_repository_contributors(repo_link_id: str, repo_page_id: str, contributors_tab_id: str):
    """Navigate from a repository listing to its Contributors page by clicking the required UI elements.
    
    Args:
        repo_link_id: The UI element ID for the repository link on the listing page.
        repo_page_id: The UI element ID to open the repository main page or its menu.
        contributors_tab_id: The UI element ID for the Contributors tab or link on the repository page.
    
    Returns:
        None: This function performs navigation clicks to land on the contributors listing.
    
    Examples:
        open_repository_contributors('354', '544', '271')
    """
    click(repo_link_id)         # open the repository from the repo list
    click(repo_page_id)         # open the repository main page or menu
    click(contributors_tab_id)  # open the Contributors tab/page
```
