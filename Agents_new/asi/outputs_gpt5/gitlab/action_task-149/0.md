```python
from typing import List

def open_groups_page(main_menu_id: str, groups_link_id: str, new_group_button_id: str):
    """Navigate to the Groups creation page from the main UI.

    Steps:
        1. Open the main menu or dashboard.
        2. Click the groups link.
        3. Click the button to start creating a new group.

    Args:
        main_menu_id: ID of the main menu or entry point to account/admin sections.
        groups_link_id: ID of the link or tab that leads to Groups.
        new_group_button_id: ID of the button to create a new group.

    Returns:
        None

    Examples:
        open_groups_page('76', '83', '222')
    """
    click(main_menu_id)
    click(groups_link_id)
    click(new_group_button_id)


def create_group_with_members(name_field_id: str,
                              group_name: str,
                              member_section_button_id: str,
                              member_list_button_id: str,
                              member_input_id: str,
                              add_member_button_id: str,
                              save_button_id: str,
                              members: List[str]):
    """Create a group by setting its name, opening the members UI, adding members, and saving.

    Steps:
        1. Fill the group's name.
        2. Open the members section / list.
        3. For each member, enter the member identifier and confirm adding it.
        4. Save the group.

    Args:
        name_field_id: ID of the input field for the group's name.
        group_name: The name to assign to the new group.
        member_section_button_id: ID to open the members section (e.g., a panel or tab).
        member_list_button_id: ID to open the member list or editor within the members section.
        member_input_id: ID of the input field used to type a member's username.
        add_member_button_id: ID of the button used to confirm adding the currently typed member.
        save_button_id: ID of the button used to save the group after members are added.
        members: List of member usernames (strings) to add to the group.

    Returns:
        None

    Examples:
        create_group_with_members('235', 'n-lab', '287', '291', '363', '385', '385', ['patou', 'egpast'])
    """
    # Step 1: set group name
    fill(name_field_id, group_name)

    # Step 2: open members editing UI
    click(member_section_button_id)
    click(member_list_button_id)

    # Step 3: add each member
    for m in members:
        fill(member_input_id, m)
        noop(500)
        keyboard_press("Enter")
        noop(500)
        click(add_member_button_id)
        noop(800)

    # Step 4: save the newly created group
    click(save_button_id)
```
