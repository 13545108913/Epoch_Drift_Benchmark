def invite_repo_guest(members_page_url: str, search_bar_id: str, username: str, result_click_id: str):
    """Navigate to a repository's project members page, search for a user, and select them to invite as a guest.

    Args:
        members_page_url: URL of the repository's project members page.
        search_bar_id: Element ID of the members search/input field.
        username: The username to search for and invite.
        result_click_id: Element ID to click on the search result or invite action.

    Returns:
        None

    Examples:
        invite_repo_guest(
            "http://example.com/myrepo/-/project_members",
            "490",

        )
