def search_repository(search_bar_id: str | int, repo_name: str):
    """Search for a repository using the page search bar.

    Args:
        search_bar_id: The identifier for the search input element (string or int).
        repo_name: The repository name or query to search for.

    Returns:
        None. (Performs UI actions to submit the search.)

    Examples:
        search_repository('142', 'Pytorch GAN')
    """
    click(search_bar_id)
    fill(search_bar_id, repo_name)
    keyboard_press('Enter')


def open_repository_contributors(repo_link_id: str | int, contributors_tab_id: str | int, contributors_link_id: str | int):
    """Open a repository page and navigate to its contributors section.

    Args:
        repo_link_id: The identifier for the repository link in the search results.
        contributors_tab_id: The identifier for the contributors tab or section element on the repo page.
        contributors_link_id: The identifier for the specific contributors link/button to view detailed contributions.

    Returns:
        None. (Performs UI navigation to the contributors view.)

    Examples:
        open_repository_contributors('281', '519', '267')
    """
    click(repo_link_id)
    click(contributors_tab_id)
    click(contributors_link_id)
