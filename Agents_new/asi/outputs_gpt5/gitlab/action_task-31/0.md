```python
def generate_ssh_clone_command(repo: str) -> str:
    """Generate a git clone command using an SSH URL for a given repository identifier.

    This function accepts several common repo identifier formats and returns a
    ready-to-run `git clone` command using the SSH form of the repository URL.

    Args:
        repo: A repository identifier. Common accepted forms:
              - "owner/repo" (e.g., "NVlabs/stylegan3")
              - full HTTPS URL (e.g., "https://github.com/NVlabs/stylegan3.git")
              - full SSH URL (e.g., "git@github.com:NVlabs/stylegan3.git")

    Returns:
        A string containing the git clone command using an SSH URL.

    Examples:
        generate_ssh_clone_command("NVlabs/stylegan3")
        -> "git clone git@github.com:NVlabs/stylegan3.git"

        generate_ssh_clone_command("https://github.com/NVlabs/stylegan3")
        -> "git clone git@github.com:NVlabs/stylegan3.git"
    """
    repo = repo.strip()
    if repo.startswith("git@") or repo.startswith("ssh://"):
        ssh_url = repo if repo.endswith(".git") else repo + ".git"
    elif repo.startswith("http://") or repo.startswith("https://"):
        # Remove scheme and leading slashes, then convert to git@host:owner/repo.git
        without_scheme = repo.split("://", 1)[1].rstrip("/")
        host_and_path = without_scheme
        ssh_url = "git@" + host_and_path.replace("/", ":", 1)
        if not ssh_url.endswith(".git"):
            ssh_url += ".git"
    else:
        # Assume owner/repo shorthand and default to GitHub
        shorthand = repo.rstrip("/")
        ssh_url = f"git@github.com:{shorthand}"
        if not ssh_url.endswith(".git"):
            ssh_url += ".git"
    return f"git clone {ssh_url}"
```
