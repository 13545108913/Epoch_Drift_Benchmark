def open_template_navigation(nav_id: str | int, template_group_id: str | int, template_choice_id: str | int):
    """Navigate the UI to the template selection area by clicking the provided elements.

    Args:
        nav_id: ID of the top-level navigation element to open the repository creation area.
        template_group_id: ID of the template group or category to open.
        template_choice_id: ID of the specific template option to select.

    Returns:
        None

    Examples:
        open_template_navigation('223', '242', '375')
    """
    click(nav_id)
    click(template_group_id)
    click(template_choice_id)


def fill_repo_details_and_create(name_field_id: str | int, slug_field_id: str | int, create_button_id: str | int, project_name: str):
    """Fill repository name and slug, then click the create button to make the repository.

    Args:
        name_field_id: ID of the repository name input field.
        slug_field_id: ID of the repository slug (or repeat name) input field.
        create_button_id: ID of the final 'Create' button.
        project_name: Desired repository name to fill into the fields.

    Returns:
        None

    Examples:
        fill_repo_details_and_create('562', '592', '624', 'web_agent_nodejs')
    """
    fill(name_field_id, project_name)
    fill(slug_field_id, project_name)
    click(create_button_id)
