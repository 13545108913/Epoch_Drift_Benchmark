```python
def open_new_issue_page(repo_link_id: str | int, issues_tab_id: str | int, new_issue_button_id: str | int):
    """Navigate from a repository link to the "New issue" form.

    Args:
        repo_link_id: ID of the repository link or element that opens the repository page.
        issues_tab_id: ID of the "Issues" tab or link inside the repository page.
        new_issue_button_id: ID of the "New issue" or "New issue" button.

    Returns:
        None

    Examples:
        open_new_issue_page('787', '283', '810')
    """
    # Open the repository page.
    click(repo_link_id)
    # Open the Issues tab within the repository.
    click(issues_tab_id)
    # Open the New Issue form.
    click(new_issue_button_id)
```
