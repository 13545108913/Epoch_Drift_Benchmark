```python
def navigate_to_new_merge_request(nav_menu_id: str | int, merge_requests_id: str | int, new_mr_button_id: str | int):
    """Navigate from the repo UI to the "Create new merge request" form.

    Args:
        nav_menu_id: ID of the navigation/menu element to open (e.g., repository menu).
        merge_requests_id: ID of the Merge Requests entry to click.
        new_mr_button_id: ID of the "New merge request" or similar button.

    Returns:
        None

    Examples:
        navigate_to_new_merge_request('846', '307', '488')
    """
    click(nav_menu_id)
    click(merge_requests_id)
    click(new_mr_button_id)


def select_branch(dropdown_id: str | int, branch_entry_id: str | int, confirm_id: str | int):
    """Select a branch from a branch dropdown list and confirm the selection.

    Args:
        dropdown_id: ID of the branch dropdown to open.
        branch_entry_id: ID of the branch entry to click (the desired branch).
        confirm_id: ID of the element to finalize/confirm the branch choice.

    Returns:
        None

    Examples:
        select_branch('498', '613', '541')  # choose source branch
    """
    click(dropdown_id)
    click(branch_entry_id)
    click(confirm_id)


def assign_reviewer(reviewer_dropdown_id: str | int, reviewer_entry_id: str | int, assign_confirm_id: str | int):
    """Assign a reviewer (or assignee) to the merge request.

    Args:
        reviewer_dropdown_id: ID of the reviewer/assignee dropdown to open.
        reviewer_entry_id: ID of the reviewer entry to select.
        assign_confirm_id: ID of the element to confirm the assignment.

    Returns:
        None

    Examples:
        assign_reviewer('591', '744', '744')
    """
    click(reviewer_dropdown_id)
    click(reviewer_entry_id)
    click(assign_confirm_id)
```
