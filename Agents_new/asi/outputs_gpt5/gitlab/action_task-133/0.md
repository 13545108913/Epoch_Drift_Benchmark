```python
def create_private_project(start_button_id: str, name_field_id: str, project_name: str,
                           visibility_option_id: str, template_option_id: str, confirm_button_id: str):
    """Start a new private project and choose a template.

    Args:
        start_button_id: ID of the control that begins project creation (e.g., "New Project").
        name_field_id: ID of the text field where the project name is entered.
        project_name: The name to set for the new project.
        visibility_option_id: ID of the control to set project visibility to private.
        template_option_id: ID of the template choice to apply to the project.
        confirm_button_id: ID of the control that confirms/creates the project.

    Returns:
        None

    Examples:
        create_private_project('223', '259', 'my_project', '327', '375', '362')
    """
    click(start_button_id)               # open project creation UI
    fill(name_field_id, project_name)    # set project name
    click(visibility_option_id)          # select private visibility
    click(template_option_id)            # pick the desired template
    click(confirm_button_id)             # confirm/create the project


def add_project_members(members_input_id: str, members: list[str], confirm_members_button_id: str):
    """Add a list of members to the current project.

    Args:
        members_input_id: ID of the member input/autocomplete field.
        members: List of member names (strings) to add.
        confirm_members_button_id: ID of the button to finalize adding members.

    Returns:
        None

    Examples:
        add_project_members('455', ['Abishek', 'Vinta'], '477')
    """
    click(members_input_id)                       # focus the member entry field
    for member in members:
        fill(members_input_id, member)            # type the member name
        keyboard_press("Enter")                   # select/confirm the typed member
    click(confirm_members_button_id)              # finalize adding members
```
