```python
def open_repo_members(repo_link_id: str, members_tab_id: str, invite_button_id: str):
    """Navigate into a repository and open its members/invite view.

    Args:
        repo_link_id: ID of the repository link to click.
        members_tab_id: ID of the repository's "Members" or "Settings" tab to click.
        invite_button_id: ID of the "Invite" or "Add member" button to open the invite UI.

    Returns:
        A dict summarizing which elements were clicked.

    Examples:
        open_repo_members('561', '395', '407')
    """
    click(repo_link_id)
    click(members_tab_id)
    click(invite_button_id)
    return {
        "repo_clicked": repo_link_id,
        "members_tab_clicked": members_tab_id,
        "invite_button_clicked": invite_button_id,
    }
```

```python
def invite_users_via_search(search_field_id: str, role_select_id: str, add_button_id: str, user_list: list[str], role: str) -> list[str]:
    """Invite multiple users by searching their username, selecting a role, and confirming the invite.

    This function performs the same sequence for each username:
    - fill the search field with the username
    - press Enter to select it
    - choose the desired role from a role dropdown
    - click the add/confirm button

    Args:
        search_field_id: ID of the username search/input field.
        role_select_id: ID of the role select element.
        add_button_id: ID of the button that confirms adding/inviting the user.
        user_list: List of usernames to invite.
        role: Role label to select for each user (e.g., "Developer").

    Returns:
        A list of usernames that were processed.

    Examples:
        invite_users_via_search('493', '501', '515', ['alice', 'bob'], 'Developer')
    """
    processed = []
    for username in user_list:
        fill(search_field_id, username)
        keyboard_press("Enter")
        select_option(role_select_id, role)
        click(add_button_id)
        processed.append(username)
    return processed
```
