def create_project(new_project_button_id: str, project_name_field_id: str, repo_name_field_id: str,
                   project_name: str, repo_name: str, create_button_id: str, visibility_private_id: str | None = None):
    """Open the new project form, fill project details, set visibility (optional) and create the project.

    Args:
        new_project_button_id: ID of the "New project" (or equivalent) button to open the creation form.
        project_name_field_id: ID of the field to fill the project display name.
        repo_name_field_id: ID of the field to fill the repository path/name.
        project_name: Text to enter for the project's display name.
        repo_name: Text to enter for the repository path/name.
        create_button_id: ID of the button that finalizes/creates the project.
        visibility_private_id: Optional ID of the UI element to select private visibility before creation.

    Returns:
        None

    Examples:
        create_project('223', '259', '270', 'llm_bulk_inference', 'llm_bulk_inference', '375', '362')
    """
    click(new_project_button_id)
    fill(project_name_field_id, project_name)
    fill(repo_name_field_id, repo_name)
    if visibility_private_id:
        click(visibility_private_id)
    click(create_button_id)


def add_project_members(members_field_id: str, members: list[str], suggestion_ids: list[str] | None = None):
    """Add one or more members to the project by filling the member input and confirming each entry.

    For each member, the function fills the member input, presses Enter to trigger suggestions, and
    optionally clicks a corresponding suggestion ID if provided.

    Args:
        members_field_id: ID of the member input field where usernames are typed.
        members: List of usernames (strings) to add as members.
        suggestion_ids: Optional list of clickable suggestion IDs corresponding to members.
                        If provided, the function will attempt to click suggestion_ids[i] after adding members[i].

    Returns:
        None

    Examples:
        add_project_members('455', ['primer', 'convexegg', 'abishek'], ['501', '502', '477'])
        add_project_members('455', ['abishek'])
    """
    for i, member in enumerate(members):
        fill(members_field_id, member)
        keyboard_press("Enter")
        if suggestion_ids and i < len(suggestion_ids):
            click(suggestion_ids[i])
