def update_gitlab_profile_homepage(menu_button_id: str | int, edit_profile_id: str | int, homepage_field_id: str | int, homepage_url: str, save_button_id: str | int):
    """Open GitLab profile edit and set the homepage URL.

    Args:
        menu_button_id: ID of the menu or profile button to access profile/settings.
        edit_profile_id: ID of the "Edit profile" or similar link/button.
        homepage_field_id: ID of the homepage/website input field.
        homepage_url: The URL string to put into the homepage field.
        save_button_id: ID of the save/update button to persist changes.

    Returns:
        None

    Examples:
        update_gitlab_profile_homepage('186', '202', '481', 'byteblaze.github.io', '512')
    """
    click(menu_button_id)
    click(edit_profile_id)
    fill(homepage_field_id, homepage_url)
    click(save_button_id)
