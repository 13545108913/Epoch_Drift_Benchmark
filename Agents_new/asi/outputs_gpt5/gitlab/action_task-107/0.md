```python
def open_project_members_page(page_url: str):
    """Navigate to a repository's project members page.

    Args:
        page_url: Full URL of the repository's project members page (e.g. "https://host/org/repo/-/project_members").

    Returns:
        None

    Examples:
        open_project_members_page("http://172.26.116.102:8080/byteblaze/my-repo/-/project_members")
    """
    goto(page_url)
    noop(500)
    scroll(0, 200)
```

```python
def add_users_to_repo(user_input_id: str, role_select_id: str, add_button_id: str, users: list, role: str):
    """Add multiple users to a repository with the specified role.

    This fills the user input, confirms the user entry, selects the role, and clicks the add button
    for each username in the provided list.

    Args:
        user_input_id: The UI element ID for the user search / entry input.
        role_select_id: The UI element ID for the role selection dropdown.
        add_button_id: The UI element ID for the button that confirms/creates the membership.
        users: A list of username strings to add.
        role: The role string to select for each user (e.g., "Reporter", "Developer").

    Returns:
        A list of usernames that were processed (same as the input users list).

    Examples:
        add_users_to_repo("490", "498", "512", ["alice","bob"], "Reporter")
    """
    for user in users:
        fill(user_input_id, user)
        keyboard_press("Enter")
        select_option(role_select_id, role)
        click(add_button_id)
    return users
```
