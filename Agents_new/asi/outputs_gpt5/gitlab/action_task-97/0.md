def navigate_and_open_first_repo(navigation_id: str, explore_id: str, first_repo_id: str):
    """Navigate to the projects/explore area and open the first repository in a new tab.

    This performs a standard navigation sequence that appears across similar flows:
    1) click a navigation element to go to the account/sections area,
    2) click the explore/projects link,
    3) open the first repository in a new tab using Control/Meta-click.

    Args:
        navigation_id: ID of the primary navigation element to enter the account/sections area.
        explore_id: ID of the "Explore" or "Projects" link on the page.
        first_repo_id: ID of the first repository element to open in a new tab.

    Returns:
        None

    Examples:
        navigate_and_open_first_repo('237', '275', '306')
    """
    click(navigation_id)
    click(explore_id)
    click(first_repo_id, modifiers=["ControlOrMeta"])


def star_repositories(repo_button_ids: list):
    """Star a list of repository items by clicking their star/button elements.

    This function scrolls to bring items into view, clicks each repository's star/button,
    and then restores focus to the original tab context.

    Args:
        repo_button_ids: List of element IDs corresponding to the star buttons (or repo items)
                         to be clicked in order.

    Returns:
        None

    Examples:
        star_repositories(['319', '344', '384', '415'])
    """
    scroll(0, 300)
    for repo_id in repo_button_ids:
        click(repo_id)
    tab_focus(0)
