```python
def navigate_to_repository(repo_link_id: str | int, repo_list_id: str | int, access_tab_id: str | int):
    """Navigate from a repository listing to the repository access (members/collaborators) page.
    
    Args:
        repo_link_id: The ID of the repository link on the page.
        repo_list_id: The ID of the repository item or menu link to open the specific repo.
        access_tab_id: The ID of the "Access", "Collaborators" or similar tab/button to view members.
        
    Returns:
        None
    
    Examples:
        navigate_to_repository('755', '392', '404')
    """
    click(repo_link_id)
    click(repo_list_id)
    click(access_tab_id)
```

```python
def send_repo_collaborators(repo_name: str, usernames: list[str]):
    """Format and send a message listing other users who have access to a repository.
    
    The function constructs the message internally and sends it to the user.
    
    Args:
        repo_name: The name of the repository.
        usernames: A list of collaborator usernames (without leading '@').
        
    Returns:
        None
    
    Examples:
        send_repo_collaborators('my-repo', ['alice', 'bob'])
    """
    header = f"Other members with access to {repo_name}:\n"
    body = "\n".join(f"- @{u}" for u in usernames) if usernames else "No other members found."
    send_msg_to_user(header + body)
```
