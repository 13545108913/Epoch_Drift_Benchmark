```python
def go_to_milestones(page_url: str, milestones_button_id: str):
    """Navigate to a project's milestones page and ensure the milestone area is visible.

    Args:
        page_url: URL of the project's milestones listing page.
        milestones_button_id: The UI element id to click to open the milestones panel or new milestone dialog.

    Returns:
        None

    Examples:
        go_to_milestones("http://example.com/project/milestones", "459")
    """
    goto(page_url)
    click(milestones_button_id)
    scroll(0, 200)


def create_milestone(title_field_id: str, title: str, start_field_id: str, start_date: str, end_field_id: str, end_date: str, submit_button_id: str):
    """Fill in milestone details and submit the milestone form.

    Args:
        title_field_id: UI id of the milestone title input.
        title: Title text for the milestone.
        start_field_id: UI id of the start date input.
        start_date: Start date string to enter (format matching the UI).
        end_field_id: UI id of the end date input.
        end_date: End date string to enter (format matching the UI).
        submit_button_id: UI id of the form submit/create button.

    Returns:
        None

    Examples:
        create_milestone("446", "Product Launch", "517", "Jan 16, 2023", "524", "Jan 30, 2023", "528")
    """
    click(title_field_id)
    fill(title_field_id, title)
    fill(start_field_id, start_date)
    fill(end_field_id, end_date)
    click(submit_button_id)
```
