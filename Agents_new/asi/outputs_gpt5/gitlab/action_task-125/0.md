def open_new_issue_flow(entry_button_id: str, new_issue_button_id: str, title_field_id: str, body_field_id: str, submit_button_id: str, title: str, body: str) -> None:
    """Open a repository/issues entry, launch the New Issue form, fill it, and submit.

    Args:
        entry_button_id: ID of an element to navigate to the repo or issues area (click).
        new_issue_button_id: ID of the "New issue" button to open the issue form.
        title_field_id: ID of the issue title input field.
        body_field_id: ID of the issue body/description field.
        submit_button_id: ID of the button to submit the new issue.
        title: Title text to place into the issue title field.
        body: Body text to place into the issue body field.

    Returns:
        None

    Examples:
        open_new_issue_flow('283', '491', '478', '521', '693',
                            'Add optional dark mode for solarized-prism-theme',
                            'Proposal\\n\\nAdd an optional dark mode...')
    """
    click(entry_button_id)
    click(new_issue_button_id)
    fill(title_field_id, title)
    fill(body_field_id, body)
