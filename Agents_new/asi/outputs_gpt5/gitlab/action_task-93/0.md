## Reusable Functions
def search_repository(search_bar_id: str | int, repo_name: str):
    """
    Search for a repository using the page search bar.

    Args:
        search_bar_id: The element ID of the search input.
        repo_name: The repository name or keyword to search for.

    Returns:
        None

    Examples:
        search_repository('241', 'dotfiles')
    """
    click(search_bar_id)
    fill(search_bar_id, repo_name)
    keyboard_press('Enter')


def navigate_to_project_members(project_link_id: str | int, members_tab_id: str | int, invite_members_button_id: str | int):
    """
    Navigate from search result to the project members invite interface.

    Args:
        project_link_id: The element ID for clicking the project from search results.
        members_tab_id: The element ID for the Members (or Project members) tab.
        invite_members_button_id: The element ID for the 'Invite members' (or similar) button.

    Returns:
        None

    Examples:
        navigate_to_project_members('289', '395', '407')
    """
    click(project_link_id)
    click(members_tab_id)
    click(invite_members_button_id)


def invite_guest(user_input_id: str | int, username: str, wait_ms: int, confirm_button_id: str | int):
    """
    Fill the invite input with a username, select the user from suggestions, and confirm the invite.

    Args:
        user_input_id: The element ID of the member invite input field.
        username: The display name or username to invite.
        wait_ms: Milliseconds to wait for suggestions to appear (use noop to pause).
        confirm_button_id: The element ID of the confirm/invite button.

    Returns:
        None

    Examples:
        invite_guest('490', 'Jakub Klinkovský', 1000, '512')
    """
    fill(user_input_id, username)
    noop(wait_ms)
    keyboard_press('ArrowDown')
    keyboard_press('Enter')
    click(confirm_button_id)
