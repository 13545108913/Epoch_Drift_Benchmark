```python
def navigate_to_status_editor(profile_menu_id: str | int, status_menu_button_id: str | int, status_input_id: str | int):
    """Open the profile menu and navigate to the status editor field.

    Args:
        profile_menu_id: ID of the profile/menu element to open.
        status_menu_button_id: ID of the button/link to open the status editor.
        status_input_id: ID of the status text input field.

    Returns:
        None

    Examples:
        navigate_to_status_editor('186', '198', '941')
    """
    click(profile_menu_id)
    click(status_menu_button_id)
    click(status_input_id)
```

```python
def update_and_save_gitlab_status(status_input_id: str | int, status_text: str, save_button_id: str | int, close_menu_id: str | int):
    """Fill the status text, save it, and close the profile menu.

    Args:
        status_input_id: ID of the status text input field.
        status_text: The status string to set (e.g., "Resting due to leg injury.").
        save_button_id: ID of the save/confirm button for the status.
        close_menu_id: ID of the element to click to close or return to the main view.

    Returns:
        None

    Examples:
        update_and_save_gitlab_status('941', 'Resting due to leg injury.', '947', '186')
    """
    fill(status_input_id, status_text)
    click(save_button_id)
    click(close_menu_id)
```
