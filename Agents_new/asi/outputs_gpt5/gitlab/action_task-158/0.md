```python
def search_and_open_repo(search_bar_id: str | int, repo_name: str, repo_result_id: str | int):
    """Search for a repository and open it from the results.

    Args:
        search_bar_id: ID of the search input element.
        repo_name: The repository name to search for.
        repo_result_id: The ID of the repository result to click in the search results.

    Returns:
        None

    Examples:
        search_and_open_repo('142', 'cloud-to-butt', '313')
    """
    click(search_bar_id)
    fill(search_bar_id, repo_name)
    keyboard_press("Enter")
    click(repo_result_id)
```
