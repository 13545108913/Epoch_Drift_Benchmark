## Reusable Functions
```python
def navigate_to_collaborators(repo_link_id: str, repo_menu_id: str, collaborators_tab_id: str):
    """Navigate from a repository link to the repository's Collaborators (or access control) tab.

    Args:
        repo_link_id: The clickable element id for the repository link on the page.
        repo_menu_id: The clickable element id for the repository menu/settings entry.
        collaborators_tab_id: The clickable element id for the Collaborators or Access control tab.

    Returns:
        None

    Examples:
        navigate_to_collaborators('756', '395', '407')
    """
    click(repo_link_id)          # open the repository
    click(repo_menu_id)          # open repository settings / manage menu
    click(collaborators_tab_id)  # open collaborators / access control tab


def invite_collaborator(input_field_id: str, collaborator_name: str, role_select_id: str, role_option: str, invite_button_id: str):
    """Invite a single collaborator by entering their account name, selecting a role, and sending the invite.

    Args:
        input_field_id: The input field id where collaborator username/account is typed.
        collaborator_name: The collaborator account name to invite (e.g., 'Abishek').
        role_select_id: The select element id used to choose the collaborator role.
        role_option: The visible role option to select (e.g., 'Developer', 'Admin').
        invite_button_id: The button id to confirm/send the invite.

    Returns:
        None

    Examples:
        invite_collaborator('494', 'Abishek', '502', 'Developer', '516')
    """
    fill(input_field_id, collaborator_name)  # type collaborator account name into the input
    keyboard_press("Enter")                  # confirm the typed username/account
    select_option(role_select_id, role_option)  # choose the collaborator role
    click(invite_button_id)                  # send the invite
```
