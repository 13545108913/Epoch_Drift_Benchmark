```python
def post_merge_request_comment(navigate_repo_id: str | int, open_mr_id: str | int, comment_box_id: str | int, submit_button_id: str | int, comment_text: str):
    """Navigate to a project's merge request and post a comment.

    Args:
        navigate_repo_id: The element ID to click to navigate to the repository/project.
        open_mr_id: The element ID to click to open the specific merge request.
        comment_box_id: The element ID of the comment input field.
        submit_button_id: The element ID of the button to submit the comment.
        comment_text: The text content to post as the comment.

    Returns:
        None

    Examples:
        post_merge_request_comment('370', '781', '667', '696', 'Thanks, working on reviews')
    """
    click(navigate_repo_id)           # navigate to the repo/project page
    click(open_mr_id)                 # open the target merge request
    fill(comment_box_id, comment_text) # enter the comment into the MR comment box
    click(submit_button_id)           # submit the comment
```
