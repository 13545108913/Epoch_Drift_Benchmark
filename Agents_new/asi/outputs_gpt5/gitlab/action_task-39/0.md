def open_search_and_search(search_button_id: str | int, search_input_id: str | int, repo_name: str):
    """Open the search UI and perform a repository search.

    Args:
        search_button_id: The element id to open or focus the search UI (e.g., a search icon).
        search_input_id: The element id of the search input field.
        repo_name: The repository name or query to search for.

    Returns:
        None. (After calling this function the search results page should be loaded.)

    Examples:
        open_search_and_search('234', '236', 'AndroidSlidingUpPanel')
    """
    click(search_button_id)       # open or focus the search area
    click(search_input_id)        # focus the input field
    fill(search_input_id, repo_name)  # type the repository name
    keyboard_press("Enter")       # submit the search


def open_repository_and_view_tabs(repo_link_id: str | int, contributors_tab_id: str | int, commits_tab_id: str | int):
    """Open a repository from search results and navigate to contributors/commits sections.

    Args:
        repo_link_id: The element id for the repository link in the search results.
        contributors_tab_id: The element id for the 'Contributors' tab or link on the repo page.
        commits_tab_id: The element id for the 'Commits' tab or link on the repo page.

    Returns:
        None. (After calling this function the UI should be showing the commits/contributors area.)

    Examples:
        open_repository_and_view_tabs('302', '519', '267')
    """
    click(repo_link_id)           # open the repository page from results
    click(contributors_tab_id)    # open the contributors view/tab
    click(commits_tab_id)         # open the commits view/tab


def inspect_top_contributor(contributor_entry_id: str | int, commits_count_id: str | int, details_button_id: str | int):
    """Select a contributor entry and open commit details for further inspection.

    Args:
        contributor_entry_id: The element id of the top contributor entry in the contributors list.
        commits_count_id: The element id that shows the contributor's commit count (or link to commits).
        details_button_id: The element id to open a detailed view of the contributor/commits.

    Returns:
        None. (This sequence navigates into the contributor details so the top contributor and counts are visible.)

    Examples:
        inspect_top_contributor('701', '702', '703')
    """
    click(contributor_entry_id)   # select the contributor entry (e.g., first row)
    click(commits_count_id)       # open or highlight commit count / commits list
    click(details_button_id)      # open detailed view of contributor or commit history
