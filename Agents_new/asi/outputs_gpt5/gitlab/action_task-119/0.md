def navigate_to_repository(home_link_id: str, org_link_id: str, repo_link_id: str):
    """Navigate from a landing page to a specific repository page by clicking the provided links.
    
    Args:
        home_link_id: The ID of the homepage or main navigation link to start from.
        org_link_id: The ID of the organization or user link.
        repo_link_id: The ID of the repository link.
    
    Returns:
        None
    
    Examples:
        navigate_to_repository('376', '283', '476')
    """
    click(home_link_id)
    click(org_link_id)
    click(repo_link_id)


def fill_issue_form(title_field_id: str, body_field_id: str, title: str, body: str):
    """Fill the issue title and body fields on the new-issue page.
    
    Args:
        title_field_id: The input field ID for the issue title.
        body_field_id: The input field ID for the issue body/description.
        title: The issue title text to enter.
        body: The issue body text to enter.
    
    Returns:
        None
    
    Examples:
        fill_issue_form('478', '521', 'Bug: example', 'Steps to reproduce...')
    """
    click(title_field_id)
    fill(title_field_id, title)
    click(body_field_id)
    fill(body_field_id, body)


def submit_issue(submit_button_id: str):
    """Submit the issue by clicking the submit button, with a short pause and confirmation click.

    Returns:
        None
    
    Examples:
        submit_issue('693')
