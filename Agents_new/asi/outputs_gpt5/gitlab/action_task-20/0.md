```python
def open_latest_issue_from_list(issues_list_id: str, scroll_x: int, scroll_y: int, issue_id: str):
    """Open an issue from an issues list by ensuring the list is focused, scrolling to reveal items, and clicking the issue.

    This performs a small navigation sequence that is reusable for opening the latest (or any specified) issue from a list view.

    Args:
        issues_list_id: The UI element id for the issues list or issues tab to focus/click.
        scroll_x: Horizontal scroll offset to apply to the issues list page (use 0 if not needed).
        scroll_y: Vertical scroll offset to apply to the issues list page to reveal items.
        issue_id: The UI element id of the issue to click/open.

    Returns:
        None

    Examples:
        open_latest_issue_from_list('161', 0, 100, '788')
    """
    click(issues_list_id)    # focus the issues list or open the issues tab
    scroll(scroll_x, scroll_y)  # scroll the list to reveal the target issue
    click(issue_id)          # click the issue to open its details
```
