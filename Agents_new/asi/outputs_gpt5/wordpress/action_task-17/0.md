```python
def search_and_click(search_bar_id: str | int, search_button_id: str | int, query: str, result_id: str | int):
    """Search for a keyword using the site's search bar and click a specified result.

    Args:
        search_bar_id: ID of the search input element (string or int).
        search_button_id: ID of the search submit button (string or int).
        query: The search keyword to enter into the search bar.
        result_id: The link/button ID of the result to click after searching.

    Returns:
        None

    Examples:
        search_and_click('370', '371', 'Collector', '85')
    """
    click(search_bar_id)         # Focus or open the search input
    fill(search_bar_id, query)   # Enter the search query
    click(search_button_id)      # Submit the search
    click(result_id)             # Click the desired search result
```
