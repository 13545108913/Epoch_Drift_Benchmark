def navigate_click_sequence(click_ids: list):
    """Click a sequence of elements by their IDs to navigate the UI.

    Args:
        click_ids: A list of element IDs (strings or ints) to click in order.

    Returns:
        int: The number of clicks performed.

    Examples:
        navigate_click_sequence(['135', '45', '135'])
    """
    num_clicks = len(click_ids)
    for element_id in click_ids:
        click(element_id)
    return num_clicks


def format_and_send_post_metadata(category: str, publish_date: str):
    """Format post metadata into a message and send it to the user.

    Args:
        category: The category text of the post (e.g., "Technology").
        publish_date: The publish date text of the post (e.g., "December 26, 2025").

    Returns:
        str: The final message that was sent.

    Examples:
        format_and_send_post_metadata("Technology", "December 26, 2025")
    """
    line1 = f"Category: {category}"
    line2 = f"Publish date: {publish_date}"
    message = line1 + "\n" + line2
    send_msg_to_user(message)
    return message
