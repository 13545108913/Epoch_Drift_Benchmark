def find_and_open_post(nav_link_id: str | int, search_bar_id: str | int, post_title: str, search_button_id: str | int, result_post_id: str | int):
    """Navigate to a posts listing, search for a post by title, and open the post.

    Args:
        nav_link_id: ID of the navigation link to go to the posts or blog section.
        search_bar_id: ID of the search input for posts.
        post_title: The title string to search for.
        search_button_id: ID of the button to execute the search.
        result_post_id: ID of the search result item (post) to click to open.

    Returns:
        None

    Examples:
        find_and_open_post('363', '372', 'My Post Title', '373', '94')
    """
    click(nav_link_id)                 # navigate to posts/blog section
    click(search_bar_id)               # focus the search bar
    fill(search_bar_id, post_title)    # enter the post title to search
    click(search_button_id)            # execute the search
    click(result_post_id)              # open the found post


def submit_comment(comment_field_id: str | int, name_field_id: str | int, email_field_id: str | int, submit_button_id: str | int, comment_content: str, commenter_name: str, commenter_email: str):
    """Fill in a comment form with content, name, and email, then submit.

    Args:
        comment_field_id: ID of the comment text area input.
        name_field_id: ID of the name input field.
        email_field_id: ID of the email input field.
        submit_button_id: ID of the button to submit the comment.
        comment_content: The comment text to post.
        commenter_name: The name to use for the comment.
        commenter_email: The email to use for the comment.

    Returns:
        None

    Examples:
        submit_comment('184', '188', '192', '200', 'Great post!', 'Alice', 'alice@example.com')
    """
    fill(comment_field_id, comment_content)   # enter the comment text
    fill(name_field_id, commenter_name)       # enter the commenter name
    fill(email_field_id, commenter_email)     # enter the commenter email
    click(submit_button_id)                   # submit the comment
