def open_post_by_title(trigger_id: str | int, search_field_id: str | int, post_title: str, search_button_id: str | int):
    """Open a post by searching for its title.

    Args:
        trigger_id: ID to click to focus/open the search input or navigation area.
        search_field_id: ID of the search input field where the title is entered.
        post_title: The exact title of the post to search for.
        search_button_id: ID of the element to click to perform the search or open the post.

    Returns:
        None

    Examples:
        open_post_by_title('363', '372', 'Some Post Title', '373')
    """
    click(trigger_id)
    fill(search_field_id, post_title)
    click(search_button_id)


def submit_comment(comment_field_id: str | int, comment_text: str, name_field_id: str | int, name: str, email_field_id: str | int, email: str, submit_button_id: str | int):
    """Fill out and submit a comment form on a post.

    Args:
        comment_field_id: ID of the comment textarea/input.
        comment_text: The comment content to submit.
        name_field_id: ID of the name input field.
        name: The name to include with the comment.
        email_field_id: ID of the email input field.
        email: The email address to include with the comment.
        submit_button_id: ID of the button to click to submit the comment.

    Returns:
        None

    Examples:
        submit_comment('293', 'Nice post!', '297', 'User', '301', 'user@example.com', '309')
    """

    fill(name_field_id, name)
    fill(email_field_id, email)
    click(submit_button_id)
