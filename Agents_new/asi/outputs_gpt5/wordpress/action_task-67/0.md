```python
def open_article_by_title(search_field_id: str | int, search_button_id: str | int, article_link_id: str | int, title: str):
    """Search for an article by title and open the article link.

    Args:
        search_field_id: The ID of the search input field.
        search_button_id: The ID of the button (or link) to trigger the search.
        article_link_id: The ID of the article link to open from the results.
        title: The article title to search for.

    Returns:
        None

    Examples:
        open_article_by_title('370', '371', '85', "Beijing extends Mars mission lead; stranded astronauts back home: China’s year in space")
    """
    click(search_field_id)
    fill(search_field_id, title)
    click(search_button_id)
    click(article_link_id)
```
