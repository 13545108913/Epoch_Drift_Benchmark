def open_article_and_check_image(search_field_id: str, article_title: str, search_button_id: str, article_link_id: str, keyword: str):
    """Search for an article by title, open it, and check whether it contains an image related to a keyword.

    Args:
        search_field_id: The ID of the search input field on the page.
        article_title: The article title to search for.
        search_button_id: The ID of the search/submit button to execute the search.
        article_link_id: The ID of the link to open the target article from search results.
        keyword: The keyword to check image relevance against inside the opened article.

    Returns:
        None. (The routine performs UI actions and triggers an image relevance report action.)

    Examples:
        open_article_and_check_image('370', 'Disrupting yourself in the age of AI', '371', '85', 'AI')
    """
    fill(search_field_id, article_title)                # Enter the article title into the search field
    click(search_button_id)                             # Submit the search
    click(article_link_id)                              # Open the article from search results
    report_image_relevance_from_metadata(keyword, '', '')  # Check if the article contains an image related to keyword
