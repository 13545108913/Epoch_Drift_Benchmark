```python
def search_and_open_article(search_bar_id: str, title: str, search_button_id: str, article_link_id: str, scroll_y: int = 1000):
    """Search for an article by title, open the article from the results, and scroll the article content.

    Args:
        search_bar_id: ID of the search input field.
        title: The article title or query to search for.
        search_button_id: ID of the element to activate the search (e.g., a search button or result trigger).
        article_link_id: ID of the article link in the search results to open.
        scroll_y: Vertical pixels to scroll once the article is opened (default 1000).

    Returns:
        None

    Examples:
        search_and_open_article('370', 'Vibe coding needs a spec, too', '371', '85', 1000)
    """
    fill(search_bar_id, title)         # enter the article title into the search bar
    click(search_button_id)            # trigger the search
    click(article_link_id)             # open the article from the search results
    scroll(0, scroll_y)                # scroll the article content to reveal images/content
```
