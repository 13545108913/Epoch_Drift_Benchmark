def navigate_to_homepage_and_open_post(home_link_id: str | int, post_id: str | int, detail_id: str | int):
    """Navigate to the homepage, scroll to the posts area, open a specific post and open its details.

    Args:
        home_link_id: The element ID for the homepage link/button.
        post_id: The element ID for the post to open (e.g., the bottom post element).
        detail_id: The element ID for the post detail or expand button.

    Returns:
        A tuple (post_id, detail_id) indicating which elements were interacted with.

    Examples:
        navigate_to_homepage_and_open_post('342', '45', '338')
    """
    click(home_link_id)   # go to homepage
    scroll(0, 1000)       # scroll down to the posts area
    click(post_id)        # open the specified post
    click(detail_id)      # open or reveal the post details
    return (str(post_id), str(detail_id))


def visit_post_and_reveal_metadata(post_container_id: str | int, category_element_id: str | int, date_element_id: str | int):
    """From an open post container, interact with elements that reveal category and publish date.

    Args:
        post_container_id: The element ID of the post container (should already be visible/open).
        category_element_id: The element ID that exposes or highlights the post category.
        date_element_id: The element ID that exposes or highlights the publish date.

    Returns:
        A dict with keys 'category_id' and 'date_id' identifying which elements were used.

    Examples:
        visit_post_and_reveal_metadata('45', '200', '201')
    """
    click(post_container_id)    # ensure the post container is focused/open
    click(category_element_id)  # click to reveal or select the category element
    click(date_element_id)      # click to reveal or select the publish date element
    return {'category_id': str(category_element_id), 'date_id': str(date_element_id)}
