```python
def change_cms_page_title(admin_cms_url: str, cms_list_link_id: str, page_edit_link_id: str, title_field_id: str, new_title: str, save_button_id: str) -> str:
    """Navigate to the CMS pages area in the admin and change a page's title.

    Args:
        admin_cms_url: URL of the admin CMS pages listing (e.g., "http://example.local/index.php/admin/cms_page/")
        cms_list_link_id: ID of the UI element that opens the account or CMS section (clicked before navigating)
        page_edit_link_id: ID of the specific page's edit link to open the page edit form
        title_field_id: ID of the title input field on the page edit form
        new_title: The new title string to put into the title field
        save_button_id: ID of the save button to persist the change

    Returns:
        A short confirmation string indicating the requested title change.

    Examples:
        change_cms_page_title(
            "http://dockerized-magento.local/index.php/admin/cms_page/",
            "202", "619", "529",
            "Cookie monster coming to your place",
            "500"
        )
    """
    click(cms_list_link_id)
    goto(admin_cms_url)
    click(page_edit_link_id)
    fill(title_field_id, new_title)
    click(save_button_id)
    return f"Requested title change to '{new_title}' for page link {page_edit_link_id}"
```
