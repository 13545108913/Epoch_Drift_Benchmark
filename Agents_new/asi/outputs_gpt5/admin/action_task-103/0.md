def open_admin_report(menu_id: str, report_url: str, focus_id: str | None = None):
    """Open an admin report page by navigating through a menu and going to the report URL.
    
    Args:
        menu_id: ID of the menu or link to click to open the admin area (e.g., "219")
        report_url: Full URL of the admin report page to open
        focus_id: Optional ID of an element to click after navigation to ensure focus (or None)
        
    Returns:
        None
    
    Examples:
        open_admin_report("219", "http://example.local/admin/report_sales/sales/")
        open_admin_report("219", "http://example.local/admin/report_sales/sales/", "300")
    """
    click(menu_id)        # open admin menu or initial link
    goto(report_url)      # navigate directly to the report page URL
    if focus_id:
        click(focus_id)   # optionally click an element to set focus


def set_report_period(period_select_id: str, period_value: str, from_id: str, from_date: str, to_id: str, to_date: str):
    """Set the reporting period controls (period type and explicit from/to dates).
    
    Args:
        period_select_id: ID of the period dropdown/select element (e.g., "499")
        period_value: Option value or label to choose (e.g., "Month")
        from_id: ID of the "from" date input field (e.g., "508")
        from_date: From date string to fill (e.g., "02/01/2023")
        to_id: ID of the "to" date input field (e.g., "516")
        to_date: To date string to fill (e.g., "02/28/2023")
    
    Returns:
        None
    
    Examples:
        set_report_period("499", "Month", "508", "02/01/2023", "516", "02/28/2023")
    """
    select_option(period_select_id, period_value)  # choose the period granularity
    fill(from_id, from_date)                       # set the start date
    fill(to_id, to_date)                           # set the end date


def generate_report_and_paginate(generate_button_id: str, next_page_id: str, page_count: int):
    """Generate the report and advance through result pages a given number of times.
    
    Args:
        generate_button_id: ID of the button to generate or refresh the report (e.g., "456")
        next_page_id: ID of the button/link used to advance pages (e.g., "571")
        page_count: Number of times to click the next-page control (integer)
    
    Returns:
        None
    
    Examples:
        generate_report_and_paginate("456", "571", 3)
    """
    click(generate_button_id)         # generate or refresh the report
    scroll(0, 300)                    # adjust view if pagination controls are off-screen
    for _ in range(page_count):       # advance through pages as requested
        click(next_page_id)
