```python
def navigate_to_review_pending(menu_id: str, scroll_y: int, review_link_id: str, pending_url: str):
    """Navigate from the admin landing area to the pending reviews page.

    Args:
        menu_id: ID of the admin menu or link to click to start navigation (e.g., reviews/menu).
        scroll_y: Vertical scroll offset to reveal the target link.
        review_link_id: ID of the specific reviews link to click (e.g., pending link).
        pending_url: Fallback or direct URL to the pending reviews page.

    Returns:
        None

    Examples:
        navigate_to_review_pending('147', 600, '147', 'http://dockerized-magento.local/index.php/admin/catalog_product_review/pending/')
    """
    click(menu_id)
    scroll(0, scroll_y)
    click(review_link_id)
    goto(pending_url)
```

```python
def report_review_count(status: str, count: int, refresh_button_id: str, scroll_offset: int):
    """Refresh/ensure visibility and send a formatted message reporting the count for a review status.

    Args:
        status: The review status label (e.g., 'Pending', 'Approved').
        count: The numeric count of reviews for the given status.
        refresh_button_id: ID of an element to click to refresh or reveal the latest count.
        scroll_offset: Vertical scroll offset to ensure elements are visible after clicking.

    Returns:
        None

    Examples:
        # after extracting review_count from the page:
        report_review_count('Pending', review_count, 'refresh_btn', 100)
    """
    click(refresh_button_id)
    scroll(0, scroll_offset)
    send_msg_to_user(f"The total count of {status} reviews is {count}.")
```
