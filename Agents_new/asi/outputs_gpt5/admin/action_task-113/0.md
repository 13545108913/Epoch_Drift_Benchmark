```python
def search_product(search_bar_id: str | int, query: str):
    """Search for a product or term using a page search bar.

    Args:
        search_bar_id: The clickable ID of the search input element.
        query: The search string to enter.

    Returns:
        None

    Examples:
        search_product('117', 'Desiree Fitness Black XS')
    """
    click(search_bar_id)
    fill(search_bar_id, query)
    keyboard_press("Enter")
```

```python
def filter_vendor(vendor_field_id: str | int, vendor_name: str, vendor_result_click_id: str | int):
    """Filter or select a vendor from a vendor search field and open the vendor result.

    Args:
        vendor_field_id: The ID of the vendor search/select input.
        vendor_name: The vendor name to enter.
        vendor_result_click_id: The ID of the vendor result element to click after searching.

    Returns:
        None

    Examples:
        filter_vendor('56', 'Desiree', '1093')
    """
    click(vendor_field_id)
    fill(vendor_field_id, vendor_name)
    keyboard_press("Enter")
    click(vendor_result_click_id)
```

```python
def update_price(price_field_id: str | int, base_price: float, percent_increase: float, save_button_id: str | int, done_button_id: str | int):
    """Compute a new price by applying a percent increase, fill it into a price field, and save.

    Args:
        price_field_id: The ID of the price input field to update.
        base_price: The current/base price as a float.
        percent_increase: Percent to increase the base price (e.g., 37 for 37%).
        save_button_id: The ID of the save/apply button to commit the price change.
        done_button_id: The ID of a confirmation/close button to finalize the update.

    Returns:
        float: The new computed price (rounded to 2 decimal places).

    Examples:
        update_price('981', 24.00, 37, '581', '1120')
    """
    new_price = round(base_price * (1 + percent_increase / 100), 2)
    fill(price_field_id, str(new_price))
    click(save_button_id)
    click(done_button_id)
    return new_price
```
