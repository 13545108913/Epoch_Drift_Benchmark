```python
def open_and_filter_admin_orders(admin_orders_url: str, status_select_id: str, status_value: str, search_button_id: str, scroll_y: int = 800):
    """Open the admin orders page and filter orders by a given status.
    
    This function navigates to the provided admin orders URL, selects the desired
    status from a status dropdown, triggers the search, and scrolls the results
    into view.

    Args:
        admin_orders_url: The URL of the admin orders listing page.
        status_select_id: The DOM id of the status <select> element.
        status_value: The visible option value to select (e.g., "Canceled").
        search_button_id: The DOM id of the button that applies the filter/search.
        scroll_y: Vertical pixels to scroll down after applying the filter (default 800).

    Returns:
        None

    Examples:
        open_and_filter_admin_orders("http://example.local/admin/sales_order/", "638", "Canceled", "483")
    """
    goto(admin_orders_url)  # navigate to the admin orders listing
    select_option(status_select_id, status_value)  # choose the status to filter by
    click(search_button_id)  # apply the filter / run the search
    scroll(0, scroll_y)  # scroll results into view
```
