def navigate_to_report_page(report_url: str, menu_id: str):
    """
    Navigate to a report page and open the related admin menu item.

    Steps:
    1. Go directly to the provided report URL.
    2. Scroll the page a bit to ensure elements are visible.
    3. Click the provided menu/link id to open the report section.

    Args:
        report_url: The full URL of the admin report page.
        menu_id: The clickable id for the menu or link to open the report.

    Returns:
        None

    Examples:
        navigate_to_report_page(
            "http://dockerized-magento.local/index.php/admin/report_product/lowstock/",
            "263"
        )
    """
    goto(report_url)
    scroll(0, 800)
    click(menu_id)


def set_stock_qty_range(from_field_id: str, to_field_id: str, from_value: str, to_value: str, apply_button_id: str):
    """
    Set the Stock Qty filter range on the Low Stock report and apply it.

    Steps:
    1. Fill the "From" quantity field.
    2. Fill the "To" quantity field.
    3. Click the button to apply the filter.

    Args:
        from_field_id: The id of the "From" quantity input field.
        to_field_id: The id of the "To" quantity input field.
        from_value: The value to enter in the "From" field (e.g., "2").
        to_value: The value to enter in the "To" field (e.g., "3").
        apply_button_id: The id of the button that applies the filter.

    Returns:
        None

    Examples:
        set_stock_qty_range("517", "520", "2", "3", "487")
    """
    fill(from_field_id, from_value)
    fill(to_field_id, to_value)
    click(apply_button_id)
