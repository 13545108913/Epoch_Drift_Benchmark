```python
def update_price_by_percentage(product_click_id: str, price_field_id: str, save_button_id: str, current_price: str, percentage: float, action: str = "increase") -> str:
    """Update a product's price in the UI by a percentage and save the change.

    This function navigates to the product entry, focuses the price field, computes
    the new price according to the percentage and action (increase or decrease),
    fills the new price, and clicks the save button.

    Args:
        product_click_id: ID to click to open/select the product.
        price_field_id: ID of the price input field to focus and fill.
        save_button_id: ID of the save button to persist the price change.
        current_price: Current price as a string (may include $ or commas).
        percentage: Percentage to change the price by (e.g., 10 for 10%).
        action: "increase" to raise the price or "decrease" to lower it.

    Returns:
        The formatted new price string (two decimal places), e.g. "64.90".

    Examples:
        # Increase from $59.00 by 10%
        new = update_price_by_percentage("478", "981", "581", "59.00", 10, "increase")
        # Decrease from 100 by 5%
        new = update_price_by_percentage("478", "981", "581", "100", 5, "decrease")
    """
    click(product_click_id)                 # open/select the product entry
    click(price_field_id)                   # focus the price input field
    # normalize and compute new price
    current = float(current_price.replace("$", "").replace(",", ""))
    if action == "increase":
        new_price = current * (1 + percentage / 100.0)
    else:
        new_price = current * (1 - percentage / 100.0)
    formatted = f"{new_price:.2f}"
    fill(price_field_id, formatted)         # set new price value
    click(save_button_id)                   # save the updated product
    return formatted
```
