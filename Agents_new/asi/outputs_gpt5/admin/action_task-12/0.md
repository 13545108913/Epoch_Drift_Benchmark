def navigate_to_report(menu_id: str | int, report_url: str):
    """Navigate from an admin menu to a specific reports page.

    Args:
        menu_id: The ID of the admin menu or link to click to open the reports section.
        report_url: The direct URL of the report page to navigate to.

    Returns:
        None

    Examples:
        navigate_to_report('279', 'http://example.local/admin/report_customer/orders/')
    """
    click(menu_id)
    scroll(0, 400)
    goto(report_url)


def set_report_date_range(start_field_id: str | int, end_field_id: str | int, start_date: str, end_date: str, apply_button_id: str | int):
    """Set the date range on a report and apply the filter.

    Args:
        start_field_id: The ID of the start date input field.
        end_field_id: The ID of the end date input field.
        start_date: The start date string to enter (e.g., '01/01/2000').
        end_date: The end date string to enter (e.g., '01/05/2026').
        apply_button_id: The ID of the button that applies the date filter.

    Returns:
        None

    Examples:
        set_report_date_range('463', '468', '01/01/2000', '01/05/2026', '476')
    """
    fill(start_field_id, start_date)
    fill(end_field_id, end_date)
    click(apply_button_id)
