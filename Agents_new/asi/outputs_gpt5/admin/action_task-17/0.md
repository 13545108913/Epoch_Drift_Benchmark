def open_admin_reviews_page(menu_click_id: str | int, reviews_page_url: str, post_open_click_id: str | int):
    """Open the admin section and navigate to the reviews page.

    Steps:
    1. Click a menu or link to ensure admin navigation is active.
    2. Go to the reviews page URL.
    3. Optionally click an element on the reviews page to ensure it is active/loaded.

    Args:
        menu_click_id: ID of the admin/menu element to click before navigation.
        reviews_page_url: Full URL of the reviews page to navigate to.
        post_open_click_id: ID of an element on the reviews page to click after navigation.

    Returns:
        None

    Examples:
        open_admin_reviews_page('150', 'http://dockerized-magento.local/index.php/admin/catalog_product_review/', '150')
    """
    click(menu_click_id)
    goto(reviews_page_url)
    click(post_open_click_id)


def filter_reviews_by_status(select_element_id: str | int, status_value: str, submit_button_id: str | int, scroll_y: int):
    """Apply a review status filter and scroll to load results.

    Steps:
    1. Select the desired status value from a dropdown.
    2. Click the submit/search button to apply the filter.
    3. Scroll the page to ensure results are visible/loaded.

    Args:
        select_element_id: ID of the status dropdown element.
        status_value: The visible value to select in the dropdown (e.g., "Not Approved").
        submit_button_id: ID of the button that applies the filter (e.g., "Search" or "Filter").
        scroll_y: Vertical scroll offset to reveal results (e.g., 2000).

    Returns:
        None

    Examples:
        filter_reviews_by_status('608', 'Not Approved', '474', 2000)
    """
    select_option(select_element_id, status_value)
    click(submit_button_id)
    scroll(0, scroll_y)
