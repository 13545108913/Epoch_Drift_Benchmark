```python
def change_cms_page_title(admin_menu_id: str | int, cms_page_url: str, page_edit_link_id: str | int, title_field_id: str | int, save_button_id: str | int, new_title: str):
    """Navigate to the CMS pages list, open a specific page for editing, change its page title, and save.

    Args:
        admin_menu_id: ID of the element to open the admin/CMS section (e.g., a "CMS" or "Pages" link).
        cms_page_url: URL of the CMS pages listing (admin pages).
        page_edit_link_id: ID of the specific page's edit link/button to open the page editor.
        title_field_id: ID of the page title input field.
        save_button_id: ID of the button to save the page.
        new_title: New title string to set for the page.

    Returns:
        None

    Examples:
        change_cms_page_title('202', 'http://example.local/admin/cms_page/', '660', '529', '500', 'New Privacy Title')
    """
    click(admin_menu_id)
    goto(cms_page_url)
    click(page_edit_link_id)
    fill(title_field_id, new_title)
    click(save_button_id)
```
