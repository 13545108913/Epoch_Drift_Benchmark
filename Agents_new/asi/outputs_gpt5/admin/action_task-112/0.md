# Reusable Functions

def open_admin_report_menu(menu_id: str, report_url: str):
    """Open the admin reports menu and navigate to a specific report page.

    Args:
        menu_id: The clickable element id for the admin/reports menu.
        report_url: The direct URL to the specific report page.

    Returns:
        None

    Examples:
        open_admin_report_menu('254', 'http://dockerized-magento.local/index.php/admin/report_sales/bestsellers/')
    """
    click(menu_id)
    scroll(0, 800)
    goto(report_url)


def set_report_date_range(start_field_id: str, start_date: str, end_field_id: str, end_date: str, apply_button_id: str):
    """Fill the start and end date fields for a report and apply the filter.

    Args:
        start_field_id: Element id for the start date input.
        start_date: Start date string to enter (e.g., "05/01/2022").
        end_field_id: Element id for the end date input.
        end_date: End date string to enter (e.g., "05/31/2023").
        apply_button_id: Element id for the button that applies/refreshes the report.

    Returns:
        None

    Examples:
        set_report_date_range('499', '05/01/2022', '507', '05/31/2023', '456')
    """
    fill(start_field_id, start_date)
    fill(end_field_id, end_date)
    click(apply_button_id)


def advance_report_pages(next_button_id: str, clicks: int, scroll_after_each: bool = True):
    """Advance the report pagination by clicking the "next" control multiple times.

    Args:
        next_button_id: Element id for the pagination/next button.
        clicks: Number of times to click the next button.
        scroll_after_each: Whether to scroll after each click to ensure the button is visible.

    Returns:
        None

    Examples:
        advance_report_pages('527', 5)
    """
    # initial click and scroll to ensure we interact with the pagination control at least once
    click(next_button_id)
    if scroll_after_each:
        scroll(0, 800)
    for _ in range(max(0, clicks - 1)):
        click(next_button_id)
        if scroll_after_each:
            scroll(0, 800)
