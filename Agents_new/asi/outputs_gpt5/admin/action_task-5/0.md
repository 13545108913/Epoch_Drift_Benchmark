## Reusable Functions
```python
def open_admin_reviews(admin_reviews_url: str, landing_click_id: str, scroll_y: int = 800):
    """Open the admin product reviews page and land on the reviews area.

    Args:
        admin_reviews_url: Full URL to the admin product reviews page.
        landing_click_id: Element ID to click after the page loads to focus the reviews area.
        scroll_y: Vertical scroll offset to ensure the reviews area is visible (default 800).

    Returns:
        None

    Examples:
        open_admin_reviews("http://dockerized-magento.local/index.php/admin/catalog_product_review/", "150")
    """
    goto(admin_reviews_url)
    scroll(0, scroll_y)
    click(landing_click_id)


def search_reviews(search_field_id: str, term: str, search_button_id: str):
    """Fill the reviews search field with a term and trigger the search.

    Args:
        search_field_id: The element ID of the reviews search input.
        term: The search term to use when filtering reviews.
        search_button_id: The element ID of the search/submit button.

    Returns:
        None

    Examples:
        search_reviews("621", "satisfied", "474")
    """
    fill(search_field_id, term)
    click(search_button_id)
    scroll(0, 300)


def report_review_count(term: str, count: int, notification_anchor_id: str, status_field_id: str):
    """Notify the user about how many reviews mention a given term.

    The notification message is constructed inside this function (not passed in),
    and the function also performs a small UI interaction to attach/update a status field.

    Args:
        term: The term that was searched for in the reviews.
        count: The numeric count of reviews that mention the term.
        notification_anchor_id: Element ID to click or focus before updating status.
        status_field_id: Element ID of a local status/text field to fill with a status message.

    Returns:
        None

    Examples:
        report_review_count("satisfied", 1, "150", "621")
    """
    click(notification_anchor_id)
    fill(status_field_id, f"Search complete for '{term}': found {count} match{'es' if count != 1 else ''}.")
    send_msg_to_user(f"There {'is' if count == 1 else 'are'} {count} review{'s' if count != 1 else ''} that mention the term '{term}'.")
```
