def open_admin_sales_orders(menu_id: str, sales_menu_id: str, orders_link_id: str):
    """Navigate through the admin menu to open the sales orders list.

    Args:
        menu_id: The ID of the main admin/menu element to click first.
        sales_menu_id: The ID of the sales submenu element to click.
        orders_link_id: The ID of the orders link or tab to open.

    Returns:
        None

    Examples:
        open_admin_sales_orders('70', '493', '89')
    """
    click(menu_id)
    scroll(0, 600)
    click(sales_menu_id)
    click(orders_link_id)


def open_url_and_wait(url: str, wait_ms: int, scroll_y: int):
    """Open a specific URL and allow the page to stabilize.

    Args:
        url: The URL to navigate to.
        wait_ms: Number of milliseconds to wait using noop to let the page load.
        scroll_y: Vertical scroll offset to adjust the view after load.

    Returns:
        None

    Examples:
        open_url_and_wait('http://dockerized-magento.local/index.php/admin/sales_order/', 1000, 0)
    """
    goto(url)
    noop(wait_ms)
    scroll(0, scroll_y)
