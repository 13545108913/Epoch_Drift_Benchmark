```python
def navigate_to_admin_orders(menu_link_id: str | int, orders_page_url: str):
    """Navigate from the admin landing to the Orders grid page.

    Args:
        menu_link_id: The clickable ID that opens the admin area or menu (e.g., '70').
        orders_page_url: The direct URL for the admin orders page.

    Returns:
        None

    Examples:
        navigate_to_admin_orders('70', 'http://dockerized-magento.local/index.php/admin/sales_order/')
    """
    click(menu_link_id)
    goto(orders_page_url)
    scroll(0, 200)


def apply_order_status_filter(status_select_id: str | int, status_label: str, apply_button_id: str | int):
    """Set the Orders grid status filter and apply it.

    Args:
        status_select_id: The ID of the status dropdown/select element.
        status_label: The visible label or option text to select (e.g., 'Suspected Fraud').
        apply_button_id: The ID of the button that applies the grid filters.

    Returns:
        None

    Examples:
        apply_order_status_filter('638', 'Suspected Fraud', '483')
    """
    select_option(status_select_id, status_label)
    click(apply_button_id)
    scroll(0, 300)


def multi_scroll(offset1: int, offset2: int, offset3: int):
    """Perform a sequence of scrolls to reveal grid rows or UI elements.

    Args:
        offset1: Vertical pixels to scroll in the first step.
        offset2: Vertical pixels to scroll in the second step.
        offset3: Vertical pixels to scroll in the third step.

    Returns:
        None

    Examples:
        multi_scroll(500, 300, 100)
    """
    scroll(0, offset1)
    scroll(0, offset2)
    scroll(0, offset3)
```
