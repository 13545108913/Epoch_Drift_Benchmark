```python
def open_marketing_price_rules(menu_id: str | int, promotions_id: str | int, price_rules_id: str | int):
    """Navigate to the marketing price rules creation page.

    Args:
        menu_id: ID of the main marketing menu or entry point to start navigation.
        promotions_id: ID of the promotions link within the marketing menu.
        price_rules_id: ID of the "Add New Price Rule" or price rules entry point.

    Returns:
        None

    Examples:
        open_marketing_price_rules('179', '172', '447')
    """
    click(menu_id)
    scroll(0, 1000)
    click(promotions_id)
    scroll(0, 500)
    click(price_rules_id)
```

```python
def fill_price_rule_basic_info(title_field_id: str | int, title: str, description_field_id: str | int, description: str):
    """Fill the basic information for a new marketing price rule.

    Args:
        title_field_id: ID of the title input field.
        title: Title text for the price rule.
        description_field_id: ID of the description input field.
        description: Description text for the price rule.

    Returns:
        None

    Examples:
        fill_price_rule_basic_info('516', 'Summer Promo', '521', 'Discount at checkout for all customers.')
    """
    click(title_field_id)
    fill(title_field_id, title)
    fill(description_field_id, description)
```

```python
def configure_price_rule_scope(website_select_id: str | int, website_option: str | list, customer_groups_select_id: str | int, customer_group_options: list, save_button_id: str | int):
    """Select website scope and customer groups for the price rule and save the basic scope.

    Args:
        website_select_id: ID of the website select element.
        website_option: Option (or list) to choose for website scope.
        customer_groups_select_id: ID of the customer groups multi-select element.
        customer_group_options: List of customer group options to select.
        save_button_id: ID of the save or continue button to persist scope selections.

    Returns:
        None

    Examples:
        configure_price_rule_scope('536', 'Main Website', '544', ['General', 'Wholesale'], '462')
    """
    select_option(website_select_id, website_option)
    select_option(customer_groups_select_id, customer_group_options)
    click(save_button_id)
```

```python
def set_fixed_amount_discount(discount_select_id: str | int, discount_option: str, amount_field_id: str | int, amount: str | int, apply_button_id: str | int):
    """Configure a fixed amount discount for the whole cart and apply/save it.

    Args:
        discount_select_id: ID of the discount type select element.
        discount_option: The discount type option to choose (e.g., "Fixed amount discount for whole cart").
        amount_field_id: ID of the amount input field.
        amount: Discount amount to apply (numeric or string).
        apply_button_id: ID of the button to apply/save the discount configuration.

    Returns:
        None

    Examples:
        set_fixed_amount_discount('664', 'Fixed amount discount for whole cart', '674', '40', '487')
    """
    select_option(discount_select_id, discount_option)
    fill(amount_field_id, str(amount))
    click(apply_button_id)
```
