```python
def open_bestsellers_report(menu_link_id: str, report_url: str):
    """Open the admin bestsellers report page by interacting with the UI and navigating to the report URL.

    Args:
        menu_link_id: The ID of the menu or link element that opens the reporting section.
        report_url: The direct URL of the bestsellers report page.

    Returns:
        None

    Examples:
        open_bestsellers_report('254', 'http://dockerized-magento.local/index.php/admin/report_sales/bestsellers/')
    """
    click(menu_link_id)    # open the reporting/menu area
    scroll(0, 600)         # make sure relevant UI is visible
    goto(report_url)       # navigate directly to the bestsellers report page


def apply_date_range_filter(period_select_id: str, period_option: str,
                            start_field_id: str, start_date: str,
                            end_field_id: str, end_date: str,
                            apply_button_id: str):
    """Set the report period and date range then apply the filter.

    Args:
        period_select_id: The ID of the period select control.
        period_option: The option value or label to choose for the period (e.g., "Year").
        start_field_id: The ID of the start date input field.
        start_date: The start date string to fill (e.g., "01/01/2022").
        end_field_id: The ID of the end date input field.
        end_date: The end date string to fill (e.g., "12/31/2022").
        apply_button_id: The ID of the button that applies the filter.

    Returns:
        None

    Examples:
        apply_date_range_filter('487', 'Year', '496', '01/01/2022', '504', '12/31/2022', '453')
    """
    select_option(period_select_id, period_option)  # choose period type
    fill(start_field_id, start_date)                # set start date
    fill(end_field_id, end_date)                    # set end date
    click(apply_button_id)                          # apply the filter
```
