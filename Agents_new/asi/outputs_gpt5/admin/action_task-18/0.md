```python
def navigate_to_sales_report(menu_button_id: str, report_url: str):
    """Navigate to the admin sales report page.

    Args:
        menu_button_id: The UI element id to click to reveal or navigate to reports (e.g., '219').
        report_url: The direct URL of the sales report page.

    Returns:
        None

    Examples:
        navigate_to_sales_report('219', 'http://dockerized-magento.local/index.php/admin/report_sales/sales/')
    """
    click(menu_button_id)   # open the menu or ensure the reports area is focused
    goto(report_url)        # navigate to the sales report page
    scroll(0, 1000)         # ensure the page is scrolled so form elements are visible


def configure_and_run_report(granularity_select_id: str, granularity_value: str,
                             start_field_id: str, start_date: str,
                             end_field_id: str, end_date: str,
                             run_button_id: str):
    """Set report granularity and date range, then run the report.

    Args:
        granularity_select_id: The select element id for report granularity (e.g., '496').
        granularity_value: The option value to choose (e.g., 'Month').
        start_field_id: The id of the start date input field (e.g., '505').
        start_date: The start date string for the report (e.g., '05/01/2022').
        end_field_id: The id of the end date input field (e.g., '513').
        end_date: The end date string for the report (e.g., '12/31/2022').
        run_button_id: The id of the button to generate/run the report (e.g., '453').

    Returns:
        None

    Examples:
        configure_and_run_report('496', 'Month', '505', '05/01/2022', '513', '12/31/2022', '453')
    """
    select_option(granularity_select_id, granularity_value)  # choose Month granularity
    fill(start_field_id, start_date)                         # set start date
    fill(end_field_id, end_date)                             # set end date
    click(run_button_id)                                     # run/generate the report
```
