def open_low_stock_report(menu_id: str, report_url: str):
    """Navigate from the admin menu to the Low Stock report page.

    Args:
        menu_id: The clickable ID for the admin/report menu (e.g., '117').
        report_url: The full URL of the low stock report page.

    Returns:
        None

    Examples:
        open_low_stock_report('117', 'http://dockerized-magento.local/index.php/admin/report_product/lowstock/')
    """
    click(menu_id)
    scroll(0, 800)
    click(menu_id)
    goto(report_url)


def set_low_stock_filter_and_run(from_field_id: str, to_field_id: str, quantity: str, run_button_id: str):
    """Fill the From and To low stock filter fields with the same quantity and run the report.

    Args:
        from_field_id: The field ID for the "From" filter (e.g., '517').
        to_field_id: The field ID for the "To" filter (e.g., '520').
        quantity: The numeric quantity to set for both From and To filters (as a string).
        run_button_id: The clickable ID for the Run/Search button (e.g., '487').

    Returns:
        None

    Examples:
        set_low_stock_filter_and_run('517', '520', '3', '487')
    """
    fill(from_field_id, quantity)
    fill(to_field_id, quantity)
    click(run_button_id)
