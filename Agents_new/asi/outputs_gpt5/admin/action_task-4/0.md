Reusable Functions
```python
def open_reviews_admin(account_link_id: str | int, reviews_page_url: str):
    """Open the admin Reviews page from the account area.

    Args:
        account_link_id: The clickable ID for the admin/account entry point.
        reviews_page_url: The direct URL to the All Reviews admin page.

    Returns:
        None

    Examples:
        open_reviews_admin('150', 'http://dockerized-magento.local/index.php/admin/catalog_product_review/')
    """
    click(account_link_id)            # open the admin/account area
    goto(reviews_page_url)            # navigate directly to the All Reviews admin page
    scroll(0, 800)                    # ensure the reviews area is visible


def search_reviews(search_input_id: str | int, search_term: str, submit_button_id: str | int):
    """Fill the reviews search input with a term and submit the search.

    Args:
        search_input_id: The input field ID where the search term is entered.
        search_term: The textual term to search for within reviews.
        submit_button_id: The ID of the button to submit the search/filter.

    Returns:
        None

    Examples:
        search_reviews('621', 'disappointed', '474')
    """
    fill(search_input_id, search_term)  # type the term into the search field
    click(submit_button_id)             # submit the search/filter
    scroll(0, 300)                      # reveal search results or updated area
```
