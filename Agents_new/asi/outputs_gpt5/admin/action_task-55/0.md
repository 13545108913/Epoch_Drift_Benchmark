```python
def open_admin_reviews(click_id_first: str, click_id_second: str, scroll_y: int, admin_url: str):
    """Navigate the UI to open the admin product reviews page.

    Args:
        click_id_first: ID of the first element to click (e.g., a menu or link).
        click_id_second: ID of the second element to click (could be same as first).
        scroll_y: Vertical scroll offset to reveal elements before the second click.
        admin_url: Direct URL of the admin reviews page to ensure the page is loaded.

    Returns:
        None: Performs UI navigation actions to reach the admin reviews page.

    Examples:
        open_admin_reviews('150', '150', 800, 'http://dockerized-magento.local/index.php/admin/catalog_product_review/')
    """
    click(click_id_first)
    scroll(0, scroll_y)
    click(click_id_second)
    goto(admin_url)
```
