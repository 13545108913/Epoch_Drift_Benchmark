def open_admin_report(report_link_id: str | int, report_url: str):
    """Navigate to an admin report page by clicking the provided link, ensuring the page is scrolled, and then opening the report URL.

    Args:
        report_link_id: ID of the report link or menu element to click.
        report_url: Direct URL to the desired admin report page.

    Returns:
        None

    Examples:
        open_admin_report('219', 'http://dockerized-magento.local/index.php/admin/report_sales/sales/')
    """
    click(report_link_id)
    scroll(0, 800)
    goto(report_url)


def configure_report_date_range(range_select_id: str | int, range_type: str, from_field_id: str | int, start_date: str, to_field_id: str | int, end_date: str):
    """Configure a report's date range by selecting the range type and filling start and end dates.

    Args:
        range_select_id: ID of the select element controlling the range type (e.g., "Month", "Custom").
        range_type: The option value or visible text to select (e.g., "Month").
        from_field_id: ID of the "from" date input field.
        start_date: Start date string to fill (format depends on the UI, e.g., "MM/DD/YYYY").
        to_field_id: ID of the "to" date input field.
        end_date: End date string to fill.

    Returns:
        None

    Examples:
        configure_report_date_range('499', 'Month', '508', '05/01/2021', '516', '03/31/2022')
    """
    select_option(range_select_id, range_type)
    fill(from_field_id, start_date)
    fill(to_field_id, end_date)


def generate_report(generate_button_id: str | int):
    """Trigger report generation by clicking the generate button and adjusting the view.

    Args:
        generate_button_id: ID of the button that runs or refreshes the report.

    Returns:
        None

    Examples:
        generate_report('456')
    """
    click(generate_button_id)
    scroll(0, 800)
    scroll(0, 300)
