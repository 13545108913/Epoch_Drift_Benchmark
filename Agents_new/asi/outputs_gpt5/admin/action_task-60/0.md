```python
def update_product_price(product_button_id: str, price_field_id: str, price_value: str, save_button_id: str, post_action_id: str | None = None):
    """Open a product editor, set the price value, save, and optionally perform a follow-up click.

    Args:
        product_button_id: ID of the element to click to open the product editor.
        price_field_id: ID of the price input field to fill with the new value.
        price_value: The new price value to write into the field (as a string, e.g., "151.50").
        save_button_id: ID of the button to click to save the price change.
        post_action_id: Optional ID of an element to click after saving (e.g., to navigate or re-open).

    Returns:
        None

    Examples:
        update_product_price('478', '986', '151.50', '586', '700')
        update_product_price('478', '981', '82.00', '581')  # no post action
    """
    click(product_button_id)       # open the product editor / select the product
    fill(price_field_id, price_value)  # set the desired price
    click(save_button_id)          # save the change
    if post_action_id:             # optional follow-up action (reopen, navigate, etc.)
        click(post_action_id)
```
