def search_admin_reviews(open_click_id: str | int, admin_reviews_url: str, search_box_id: str | int, search_button_id: str | int, term: str):
    """Navigate to the admin reviews page and run a search for a term.
    
    Steps:
    1. Click an element (e.g., entry point or menu) to ensure focus or open the admin area.
    2. Navigate to the admin reviews URL.
    3. Fill the search box with the term and submit the search by clicking the search button.
    
    Args:
        open_click_id: ID of an element to click before navigation (e.g., account/menu link).
        admin_reviews_url: URL of the admin reviews listing page.
        search_box_id: ID of the reviews search input box.
        search_button_id: ID of the button that triggers the search/filter.
        term: The search term to look for in reviews.
    
    Returns:
        None
    
    Examples:
        search_admin_reviews('150', 'http://dockerized-magento.local/index.php/admin/catalog_product_review/', '621', '474', 'not useful')
    """
    click(open_click_id)
    goto(admin_reviews_url)
    fill(search_box_id, term)
    click(search_button_id)


def report_review_count(term: str, count: int):
    """Format and send a message reporting how many reviews mention a term.
    
    Steps:
    1. Determine correct verb and pluralization based on count.
    2. Build the message string mentioning the term and count.
    3. Send the message to the user and return it.
    
    Args:
        term: The search term that was used to filter reviews.
        count: The number of reviews that mention the term.
    
    Returns:
        The message string that was sent to the user.
    
    Examples:
        report_review_count('not useful', 0)
    """
    verb = "is" if count == 1 else "are"
    plural = "" if count == 1 else "s"
    message = f"There {verb} {count} review{plural} that mention \"{term}\"."
    send_msg_to_user(message)
    return message
