```python
def open_customer_search(navigation_id: str, results_toggle_id: str, phone_field_id: str, phone_number: str, search_submit_id: str) -> bool:
    """Navigate to the customer search area, enter a phone number, and submit the search.
    
    Args:
        navigation_id: The element ID to open the customer/account area (e.g., "163").
        results_toggle_id: The element ID to reveal or focus the results/search form (e.g., "159").
        phone_field_id: The input field ID for entering the phone number (e.g., "627").
        phone_number: The phone number string to search for.
        search_submit_id: The element ID to submit the search (e.g., "481").
    
    Returns:
        True if the sequence of UI actions to perform the search was executed.
    
    Examples:
        open_customer_search('163', '159', '627', '2137418080', '481')
    """
    click(navigation_id)           # open account/customer area
    scroll(0, 800)                 # scroll to reveal search pane or results
    click(results_toggle_id)       # focus or open the search form
    fill(phone_field_id, phone_number)  # enter the phone number to search
    click(search_submit_id)        # submit the search
    return True
```

```python
def select_customer_from_results(result_item_id: str, details_button_id: str) -> str:
    """Select a customer from the search results and open their details view.
    
    Args:
        result_item_id: The element ID of the result row to click (e.g., '163').
        details_button_id: The element ID to open the detailed view for that result (e.g., '481').
    
    Returns:
        The ID of the selected result item.
    
    Examples:
        select_customer_from_results('163', '481')
    """
    click(result_item_id)   # select the result entry
    scroll(0, 200)          # ensure details button is visible
    click(details_button_id)  # open the detailed customer view
    return result_item_id
```
