def navigate_to_admin_orders(admin_link_id: str, orders_url: str, scroll_y: int):
    """Navigate from the admin homepage to the Sales Orders list and adjust viewport.

    Args:
        admin_link_id: ID of the admin/main navigation link to click first (e.g., "70").
        orders_url: The URL of the admin sales order listing page.
        scroll_y: Vertical scroll offset to bring elements into view.

    Returns:
        None

    Examples:
        navigate_to_admin_orders("70", "http://dockerized-magento.local/index.php/admin/sales_order/", 500)
    """
    click(admin_link_id)
    goto(orders_url)
    scroll(0, scroll_y)


def attempt_order_cancel(order_row_id: str, action_select_id: str, action_value: str, submit_button_id: str, confirm_button_id: str):
    """Open a specific order row and attempt to perform a bulk/action operation (e.g., Cancel).

    Args:
        order_row_id: The clickable ID for the order row or its selector (e.g., "1302").
        action_select_id: The ID of the actions dropdown/select element (e.g., "1147").
        action_value: The visible option value to choose from the actions dropdown (e.g., "Cancel").
        submit_button_id: The ID of the button that submits the chosen action (e.g., "1160").
        confirm_button_id: The ID of the follow-up confirmation/button to finalize the action (e.g., "682").

    Returns:
        None

    Examples:
        attempt_order_cancel("1302", "1147", "Cancel", "1160", "682")
    """
    click(order_row_id)
    select_option(action_select_id, action_value)
    click(submit_button_id)
    click(confirm_button_id)
