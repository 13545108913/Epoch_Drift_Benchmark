def open_admin_bestsellers(admin_bestsellers_url: str, nav_button_id: str | int, scroll_y: int):
    """Open the admin bestsellers report page by navigating to the URL, scrolling, and clicking a navigation element.

    Args:
        admin_bestsellers_url: Full URL to the admin bestsellers report page.
        nav_button_id: The clickable element ID to activate the bestsellers view (e.g., a menu or link id).
        scroll_y: Vertical scroll offset to ensure elements are visible before clicking.

    Returns:
        None

    Examples:
        open_admin_bestsellers("http://dockerized-magento.local/index.php/admin/report_sales/bestsellers/", "254", 800)
    """
    goto(admin_bestsellers_url)
    scroll(0, scroll_y)
    click(nav_button_id)


def set_report_date_range(start_input_id: str | int, end_input_id: str | int, start_date: str, end_date: str, apply_button_id: str | int):
    """Fill the report start and end date fields and apply the date range.

    Args:
        start_input_id: Element ID of the start date input.
        end_input_id: Element ID of the end date input.
        start_date: Start date string in the page's expected format (e.g., "MM/DD/YYYY").
        end_date: End date string in the page's expected format (e.g., "MM/DD/YYYY").
        apply_button_id: Element ID of the button to apply the date range.

    Returns:
        None

    Examples:
        set_report_date_range("496", "504", "01/01/2022", "12/31/2022", "453")
    """
    fill(start_input_id, start_date)
    fill(end_input_id, end_date)
    click(apply_button_id)


def apply_report_period_filter(select_id: str | int, option_value: str, apply_button_id: str | int, post_scroll_y: int):
    """Select a period grouping for the report, apply it, and adjust view to show results.

    Args:
        select_id: Element ID of the period/grouping select control.
        option_value: The option value to select (e.g., "Year", "Month").
        apply_button_id: Element ID of the button to apply the filter.
        post_scroll_y: Vertical scroll offset after applying to reveal results.

    Returns:
        None

    Examples:
        apply_report_period_filter("487", "Year", "453", 1200)
    """
    select_option(select_id, option_value)
    click(apply_button_id)
    scroll(0, post_scroll_y)
