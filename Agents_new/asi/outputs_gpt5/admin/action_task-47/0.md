def open_admin_orders(admin_menu_id: str | int, orders_url: str, focus_element_id: str | int):
    """Open the admin orders page by clicking the admin menu, navigating to the orders URL, and focusing an element.
    
    Args:
        admin_menu_id: ID of the admin/menu element to click first.
        orders_url: URL of the orders page to navigate to.
        focus_element_id: ID of an element to click/focus after navigation (e.g., a filter control).
        
    Returns:
        None
    
    Examples:
        open_admin_orders('70', 'http://dockerized-magento.local/index.php/admin/sales_order/', '638')
    """
    click(admin_menu_id)
    goto(orders_url)
    click(focus_element_id)


def filter_orders_by_status_and_refresh(filter_control_id: str | int, status_label: str, apply_button_id: str | int):
