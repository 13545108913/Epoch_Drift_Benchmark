def navigate_to_admin_orders(menu_click_id: str, orders_url: str):
    """Open the admin sales orders page from the admin menu.

    Args:
        menu_click_id: The ID of the admin menu or "My Account" link to click before navigating.
        orders_url: The direct URL of the admin orders page.

    Returns:
        None

    Examples:
        navigate_to_admin_orders('70', 'http://dockerized-magento.local/index.php/admin/sales_order/')
    """
    click(menu_click_id)        # open admin/menu area
    goto(orders_url)            # navigate to the orders listing page
    scroll(0, 300)              # adjust viewport to load initial results


def filter_orders_by_status(status_select_id: str, status_value: str, filter_button_id: str):
    """Apply a status filter on the orders listing.

    Args:
        status_select_id: The select element ID used to pick an order status.
        status_value: The visible value or option to select (e.g., 'Canceled').
        filter_button_id: The ID of the button to apply the filter.

    Returns:
        None

    Examples:
        filter_orders_by_status('1276', 'Canceled', '1122')
    """
    select_option(status_select_id, status_value)  # choose desired order status
    click(filter_button_id)                        # apply the filter
    scroll(0, 500)                                 # scroll to reveal filtered results


def notify_top_cancellations(customer_name: str, cancellations: int, attribute_label: str):
    """Compose and send a message reporting the top customer's cancellations.

    Args:
        customer_name: The customer's display name.
        cancellations: The number of cancellations for that customer.
        attribute_label: The attribute being reported (e.g., 'total number of cancellations').

    Returns:
        None

    Examples:
        notify_top_cancellations('Jane Doe', 2, 'total number of cancellations')
    """
    scroll(0, 100)  # small viewport adjustment before reporting
    scroll(0, 200)  # ensure any dynamic content is stable
    message = f"The customer with the most {attribute_label} has {cancellations} cancellations ({customer_name})."
    send_msg_to_user(message)
