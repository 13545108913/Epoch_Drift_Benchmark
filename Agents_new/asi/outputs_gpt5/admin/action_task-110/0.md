```python
def navigate_to_report_section(menu_id: str, subsection_id: str, scroll_amount: int):
    """Navigate through the admin UI to open a report subsection.

    This performs a menu click, scrolls the page, and clicks the subsection entry.

    Args:
        menu_id: ID of the main menu entry to click (e.g., reports main link).
        subsection_id: ID of the subsection link to click after scrolling.
        scroll_amount: Vertical pixels to scroll down before clicking the subsection.

    Returns:
        None

    Examples:
        navigate_to_report_section('260', '260', 1000)
    """
    click(menu_id)
    scroll(0, scroll_amount)
    click(subsection_id)


def create_date_range_report(report_url: str, start_field_id: str, end_field_id: str, start_date: str, end_date: str, apply_button_id: str, view_button_id: str | None = None):
    """Open a report URL, set a date range, apply it, and optionally view results.

    Steps: navigate directly to the report page, fill the start and end date fields,
    click the apply button, and optionally click a view button to display the report.

    Args:
        report_url: Direct URL of the report page to open.
        start_field_id: ID of the start date input field.
        end_field_id: ID of the end date input field.
        start_date: Start date string to fill (format as expected by the UI).
        end_date: End date string to fill.
        apply_button_id: ID of the button that applies the date filter.
        view_button_id: Optional ID of a button to view/display the filtered report.

    Returns:
        None

    Examples:
        create_date_range_report(
            "http://dockerized-magento.local/index.php/admin/report_product/viewed/",
            "499", "507", "07/05/2021", "05/31/2023", "456", "460"
        )
    """
    goto(report_url)
    fill(start_field_id, start_date)
    fill(end_field_id, end_date)
    click(apply_button_id)
    if view_button_id:
        click(view_button_id)
```
