def open_admin_cms(admin_url: str, initial_click_id: str, scroll_offset: int):
    """Open the admin CMS pages list by performing an initial click, navigating to the admin URL, and scrolling.
    
    Args:
        admin_url: The full URL of the admin CMS listing page.
        initial_click_id: The element ID to click before navigation (e.g., open menu or entry point).
        scroll_offset: Vertical pixels to scroll after navigating.
    
    Returns:
        None
    
    Examples:
        open_admin_cms('http://example.local/admin/cms_page/', '202', 500)
    """
    click(initial_click_id)
    goto(admin_url)
    scroll(0, scroll_offset)


def select_and_update_page_title(page_edit_link_id: str, title_field_id: str, save_button_id: str, new_title: str):
    """Select a CMS page to edit, update its title field, and save the page.
    
    Args:
        page_edit_link_id: The element ID for the page's edit link.
        title_field_id: The input field ID for the page title.
        save_button_id: The element ID for the save button.
        new_title: The new title string to set for the page.
    
    Returns:
        None
    
    Examples:
        select_and_update_page_title('574', '529', '500', 'New Title')
    """
    click(page_edit_link_id)
    fill(title_field_id, new_title)
    click(save_button_id)
