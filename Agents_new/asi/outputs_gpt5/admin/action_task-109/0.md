```python
def navigate_to_admin_menu(menu_id: str, submenu_id: str, scroll_y: int):
    """
    Navigate the admin menu: click a top-level menu, perform a vertical scroll, then click a submenu.

    Args:
        menu_id: ID of the top-level menu item to click (string).
        submenu_id: ID of the submenu item to click after scrolling (string).
        scroll_y: Vertical scroll offset in pixels (positive or negative integer).

    Returns:
        None

    Examples:
        navigate_to_admin_menu('228', '211', 800)
        navigate_to_admin_menu('100', '101', -200)
    """
    click(menu_id)
    scroll(0, scroll_y)
    click(submenu_id)


def create_date_range_report(report_url: str, start_field_id: str, end_field_id: str,
                             start_date: str, end_date: str, generate_button_id: str,
                             confirm_click_ids: list):
    """
    Open a report page, fill a start and end date range, click the generate button, and optionally perform follow-up clicks.

    Args:
        report_url: Full URL of the report page to open.
        start_field_id: ID of the start-date input field.
        end_field_id: ID of the end-date input field.
        start_date: Start date string to enter (e.g., "08/05/2022").
        end_date: End date string to enter (e.g., "03/01/2023").
        generate_button_id: ID of the button that triggers report generation.
        confirm_click_ids: List of element IDs to click after generation (can be empty).

    Returns:
        None

    Examples:
        create_date_range_report(
            "http://example.local/admin/report_sales/shipping/",
            "506", "514", "08/05/2022", "03/01/2023", "456", ["562"]
        )
        create_date_range_report(
            "http://example.local/admin/report_sales/sales/",
            "600", "601", "01/01/2023", "01/31/2023", "700", []
        )
    """
    goto(report_url)
    fill(start_field_id, start_date)
    fill(end_field_id, end_date)
    click(generate_button_id)
    for cid in confirm_click_ids:
        click(cid)
```
