def open_admin_review_page(menu_link_id: str, admin_reviews_url: str, focus_element_id: str):
    """Navigate to the admin reviews management page.

    This performs the typical steps needed to open the admin review listing: open the admin menu,
    navigate to the reviews URL, and focus/click a known element on the page so it's ready for input.

    Args:
        menu_link_id: The element id for the admin/menu link to reveal admin navigation (e.g., '150').
        admin_reviews_url: The full URL of the admin reviews page.
        focus_element_id: The id of an element on the reviews page to click/focus after navigation (e.g., search field id).

    Returns:
        None. After running, the browser is on the admin reviews page with the focus element clicked.

    Examples:
        open_admin_review_page('150', 'http://dockerized-magento.local/index.php/admin/catalog_product_review/', '621')
    """
    click(menu_link_id)
    goto(admin_reviews_url)
    click(focus_element_id)


def search_reviews_on_admin_page(search_field_id: str, search_button_id: str, term: str):
    """Perform a review text search on the admin reviews page.

    This performs the common actions to search reviews by a term: focus the search field, enter the term,
    and trigger the search via the search/button element.

    Args:
        search_field_id: The id of the search input field on the reviews page (e.g., '621').
        search_button_id: The id of the search/submit button to trigger the search (e.g., '474').
        term: The search term to enter into the review search field.

    Returns:
        None. After running, the search will be submitted and the results page/table will be displayed.

    Examples:
        search_reviews_on_admin_page('621', '474', 'best')
    """
    click(search_field_id)
    fill(search_field_id, term)
    click(search_button_id)
