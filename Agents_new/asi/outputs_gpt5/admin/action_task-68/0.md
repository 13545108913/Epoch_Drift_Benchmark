```python
def navigate_to_cms_pages(admin_link_id: str | int, cms_pages_url: str, scroll_y: int):
    """Navigate from the admin link to the CMS pages listing and scroll to the desired position.

    Args:
        admin_link_id: The element ID for the admin or "My Account" link to open admin area.
        cms_pages_url: The direct URL to the CMS pages listing in the admin.
        scroll_y: Vertical pixel amount to scroll on the CMS pages listing.

    Returns:
        None

    Examples:
        navigate_to_cms_pages('202', 'http://dockerized-magento.local/index.php/admin/cms_page/', 800)
    """
    click(admin_link_id)
    goto(cms_pages_url)
    scroll(0, scroll_y)


def edit_cms_page_title(page_row_id: str | int, title_field_id: str | int, save_button_id: str | int, new_title: str):
    """Open a CMS page for editing, change its page title, and save the changes.

    Args:
        page_row_id: The element ID for the specific CMS page row or its "Edit" link.
        title_field_id: The form field ID where the page title is edited.
        save_button_id: The element ID for the save button on the edit page.
        new_title: The new title string to set for the page.

    Returns:
        None

    Examples:
        edit_cms_page_title('645', '529', '500', 'New page title here')
    """
    click(page_row_id)
    fill(title_field_id, new_title)
    click(save_button_id)
```
