## Reusable Functions
```python
def open_admin_order(admin_link_id: str, order_id: str, view_button_id: str):
    """Open the admin area and navigate to a specific order's page.
    
    Args:
        admin_link_id: ID of the "My Account" or admin link to click first.
        order_id: The order ID to open in the admin URL.
        view_button_id: ID of the element to click to view the order after navigation.
        
    Returns:
        None
    
    Examples:
        open_admin_order("70", "125", "715")
    """
    click(admin_link_id)
    goto(f"http://dockerized-magento.local/index.php/admin/sales_order/view/order_id/{order_id}/")
    click(view_button_id)


def enter_edit_address_mode(edit_button_id: str, first_field_id: str, second_field_id: str):
    """Enter the address edit mode for an order by clicking the edit controls and focusing fields.
    
    Args:
        edit_button_id: ID of the button that opens edit mode for addresses.
        first_field_id: ID of the first address field to focus.
        second_field_id: ID of a second address field to focus (ensures the form is active).
        
    Returns:
        None
    
    Examples:
        enter_edit_address_mode("593", "508", "510")
    """
    click(edit_button_id)
    click(first_field_id)
    click(second_field_id)


def fill_address_lines(street_id: str, street2_id: str, city_id: str, street: str, street2: str, city: str):
    """Fill the street, secondary street, and city fields of an address form.
    
    Args:
        street_id: ID of the primary street address input.
        street2_id: ID of the secondary street address input (apt/suite).
        city_id: ID of the city input.
        street: Primary street address string.
        street2: Secondary street address string.
        city: City name string.
        
    Returns:
        None
    
    Examples:
        fill_address_lines("508", "510", "516", "654 Elm Drive", "Apartment 12", "Miami")
    """
    fill(street_id, street)
    fill(street2_id, street2)
    fill(city_id, city)


def select_state_fill_zip_save(state_select_id: str, zip_id: str, save_button_id: str, state: str, zip_code: str):
    """Select the state, fill the postal code, and save the address edits.
    
    Args:
        state_select_id: ID of the state dropdown/select element.
        zip_id: ID of the postal/ZIP code input.
        save_button_id: ID of the button to save the changes.
        state: The visible option value to select for state (e.g., "Florida").
        zip_code: Postal or ZIP code string.
        
    Returns:
        None
    
    Examples:
        select_state_fill_zip_save("776", "849", "451", "Florida", "33101")
    """
    select_option(state_select_id, state)
    fill(zip_id, zip_code)
    click(save_button_id)
```
