def navigate_to_admin_orders(admin_menu_id: str, sales_menu_id: str, orders_page_url: str):
    """Navigate from the admin/dashboard area to the Sales > Orders listing page.

    Steps:
    1. Open the admin/account menu.
    2. Open the Sales section link.
    3. Go directly to the orders listing URL.

    Args:
        admin_menu_id: ID of the admin/account menu link to click (string or numeric ID).
        sales_menu_id: ID of the Sales or Orders entry to click.
        orders_page_url: Direct URL of the admin orders listing page.

    Returns:
        None

    Examples:
        navigate_to_admin_orders('493', '89', 'http://example.local/admin/sales_order/')
    """
    click(admin_menu_id)
    click(sales_menu_id)
    goto(orders_page_url)


def open_and_scroll_order_list(orders_link_id: str, results_link_id: str, scroll_y: int):
    """Open an orders listing entry, scroll the list, and focus a specific results element.

    Steps:
    1. Click an orders listing link to ensure the list or filter is active.
    2. Scroll the page vertically by the given amount to reveal more rows.
    3. Click the results or pagination link to load additional orders.

    Args:
        orders_link_id: ID of the element that opens/refreshes the orders list.
        results_link_id: ID of the element to click after scrolling (could be a page link or a repeated list entry).
        scroll_y: Vertical scroll amount in pixels (e.g., 300).

    Returns:
        None

    Examples:
        open_and_scroll_order_list('89', '89', 300)
    """
    click(orders_link_id)
    scroll(0, scroll_y)
    click(results_link_id)
