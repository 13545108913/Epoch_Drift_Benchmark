```python
def open_admin_orders(home_link_id: str, orders_url: str, search_field_id: str):
    """Navigate from the admin homepage to the Orders listing and focus the search field.

    Args:
        home_link_id: ID of the admin/home link to click first.
        orders_url: URL of the admin orders page to open.
        search_field_id: ID of the orders search/input field to focus.

    Returns:
        None

    Examples:
        open_admin_orders('70', 'http://example.local/admin/sales_order/', '596')
    """
    click(home_link_id)
    goto(orders_url)
    click(search_field_id)


def search_order_in_list(search_field_id: str, order_number: str, search_button_id: str):
    """Search for an order by its increment/number from the Orders list.

    This will focus the field, type the order number, and trigger the search.

    Args:
        search_field_id: ID of the search input field.
        order_number: The order increment id (e.g., '000000301').
        search_button_id: ID of a search or results button (used if needed).

    Returns:
        None

    Examples:
        search_order_in_list('56', '000000301', '486')
    """
    click(search_field_id)
    fill(search_field_id, order_number)
    keyboard_press("Enter")


def bulk_cancel_selected(order_checkbox_id: str, actions_dropdown_id: str, cancel_option_value: str, apply_button_id: str):
    """Select an order from the list, choose the Cancel action from the Actions dropdown, and apply it.

    Args:
        order_checkbox_id: ID of the order row checkbox to select.
        actions_dropdown_id: ID of the Actions dropdown on the list page.
        cancel_option_value: The option value or visible label for cancellation (e.g., 'Cancel').
        apply_button_id: ID of the button that applies the selected action.

    Returns:
        None

    Examples:
        bulk_cancel_selected('665', '509', 'Cancel', '522')
    """
    click(order_checkbox_id)
    click(actions_dropdown_id)
    select_option(actions_dropdown_id, cancel_option_value)
    click(apply_button_id)


def cancel_from_order_page(order_open_id: str, more_actions_id: str, cancel_option_value: str, apply_button_id: str, confirm_button_id: str):
    """Open an order and invoke the Cancel action from the order page's actions, confirming the operation.

    Args:
        order_open_id: ID of the element to open the specific order (e.g., a 'View' link).
        more_actions_id: ID of the more actions/select element on the order details page.
        cancel_option_value: The option value or label for the Cancel action (e.g., 'Cancel').
        apply_button_id: ID of the button that applies the selected action on the order page.
        confirm_button_id: ID of the final confirmation button to complete the cancellation workflow.

    Returns:
        None

    Examples:
        cancel_from_order_page('1302', '1147', 'Cancel', '1160', '1315')
    """
    click(order_open_id)
    click(more_actions_id)
    select_option(more_actions_id, cancel_option_value)
    click(apply_button_id)
    click(confirm_button_id)
```
