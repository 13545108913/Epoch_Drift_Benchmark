def open_admin_sales_orders(menu_id: str, orders_link_id: str, orders_url: str):
    """Navigate to the admin Sales Orders page.

    Args:
        menu_id: The ID of the admin/menu element to click to open the admin section.
        orders_link_id: The ID of the link or button that navigates to Sales > Orders.
        orders_url: The direct URL to the admin sales orders listing page.

    Returns:
        None

    Examples:
        open_admin_sales_orders("493", "89", "http://dockerized-magento.local/index.php/admin/sales_order/")
    """
    click(menu_id)            # open admin or account menu
    click(orders_link_id)     # click the Sales > Orders link
    goto(orders_url)          # ensure the browser is on the orders listing page


def sum_last_n_payments(payment_strings: list[str], n: int) -> float:
    """Parse payment amount strings, take the last n values, and return their sum.

    This function normalizes currency-formatted strings (e.g., "$1,234.56"),
    converts them to floats, selects the last n values, and returns the total.

    Args:
        payment_strings: A list of payment amount strings as they appear on the page.
        n: Number of most recent payments to include in the sum.

    Returns:
        The sum of the last n payment amounts as a float.

    Examples:
        sum_last_n_payments(["$10.00", "$20.50", "$30.00"], 2)  # returns 50.5
    """
    # Normalize strings: remove currency symbols and commas
    normalized = [s.replace("$", "").replace(",", "").strip() for s in payment_strings]
    # Parse to floats
    values = [float(x) for x in normalized]
    # Sum the last n values
    return sum(values[-n:])
