def open_cart_price_rules(admin_link_id: str | int, admin_url: str, add_new_button_id: str | int):
    """Navigate from the admin homepage to the "Add New Shopping Cart Price Rule" page.
    
    Args:
        admin_link_id: ID of the admin/account link to open admin area (click).
        admin_url: Direct URL to the cart price rules admin listing page.
        add_new_button_id: ID of the "Add New" button on the cart price rules page.
    
    Returns:
        None
    
    Examples:
        open_cart_price_rules('179', 'http://dockerized-magento.local/index.php/admin/promo_quote/', '447')
    """
    click(admin_link_id)
    goto(admin_url)
    click(add_new_button_id)


def configure_price_rule_basic(name_field_id: str | int,
                               name: str,
                               desc_field_id: str | int,
                               description: str,
                               website_select_id: str | int,
                               website_option: str | list,
                               customer_groups_select_id: str | int,
                               customer_groups: list,
                               save_button_id: str | int):
    """Fill in the basic information for a shopping cart price rule and save the basic form.
    
    Args:
        name_field_id: Field ID for the rule name.
        name: The name of the price rule.
        desc_field_id: Field ID for the rule description.
        description: The description for the rule.
        website_select_id: Select element ID for choosing website(s).
        website_option: The website option to select (single string or list).
        customer_groups_select_id: Select element ID for choosing customer group(s).
        customer_groups: List of customer groups to apply the rule to.
        save_button_id: ID of the button to save basic settings / continue.
    
    Returns:
        None
    
    Examples:
        configure_price_rule_basic('516', "Mother's Day $15 Off", '521',
                                   "Automatic $15 discount at checkout for all customers for Mother's Day",
                                   '536', 'Main Website', '544',
                                   ["NOT LOGGED IN","General"], '462')
    """
    fill(name_field_id, name)
    fill(desc_field_id, description)
    select_option(website_select_id, website_option)
    select_option(customer_groups_select_id, customer_groups)
    click(save_button_id)


def set_discount_action(discount_type_select_id: str | int,
                        discount_type_option: str,
                        amount_field_id: str | int,
                        amount: str | int,
                        final_save_button_id: str | int):
    """Set the discount action (type and amount) for a shopping cart price rule and save.
    
    Args:
        discount_type_select_id: ID of the select element that chooses discount type.
        discount_type_option: The option label/value for discount type (e.g., "Fixed amount discount for whole cart").
        amount_field_id: Field ID where discount amount is entered.
        amount: The discount amount to apply (string or numeric).
        final_save_button_id: ID of the final save button to persist the rule.
    
    Returns:
        None
    
    Examples:
        set_discount_action('664', 'Fixed amount discount for whole cart', '674', '15', '487')
    """
    select_option(discount_type_select_id, discount_type_option)
    fill(amount_field_id, str(amount))
    click(final_save_button_id)
