```python
def open_order_admin_page_and_open_edit(order_page_url: str, admin_link_id: str, edit_button_id: str):
    """Open the admin order page for a specific order and open the shipping address edit form.

    Args:
        order_page_url: Full URL to the admin order view page (including order_id).
        admin_link_id: ID of the admin/account link to click before navigating.
        edit_button_id: ID of the button/link that opens the shipping address edit form.

    Returns:
        None

    Examples:
        open_order_admin_page_and_open_edit(
            "http://dockerized-magento.local/index.php/admin/sales_order/view/order_id/300/",
            "70",
            "593",
        )
    """
    click(admin_link_id)
    goto(order_page_url)
    click(edit_button_id)
```

```python
def update_shipping_address_and_save(street_field_id: str, street: str,
                                     city_field_id: str, city: str,
                                     state_select_id: str, state_value: str,
                                     zip_field_id: str, zip_code: str,
                                     save_button_id: str):
    """Fill in shipping address fields and save the updated address.

    Args:
        street_field_id: ID of the street address input field.
        street: Street address to enter.
        city_field_id: ID of the city input field.
        city: City name to enter.
        state_select_id: ID of the state select element.
        state_value: Option value or visible text to select for the state.
        zip_field_id: ID of the postal/zip code input field.
        zip_code: Postal/zip code to enter.
        save_button_id: ID of the save/update button to submit the form.

    Returns:
        None

    Examples:
        update_shipping_address_and_save(
            "508", "987 Cedar Court",
            "516", "Los Angeles",
            "776", "California",
            "849", "90012",
            "451",
        )
    """
    fill(street_field_id, street)
    fill(city_field_id, city)
    select_option(state_select_id, state_value)
    fill(zip_field_id, zip_code)
    click(save_button_id)
```
