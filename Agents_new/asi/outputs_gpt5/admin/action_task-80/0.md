## Reusable Functions

def goto_admin_orders(orders_link_id: str, orders_url: str, search_field_id: str):
    """Open the admin Orders page and focus the order search field.

    Args:
        orders_link_id: Element ID for the "My Account" / admin link to reach admin area.
        orders_url: URL of the admin orders listing page.
        search_field_id: Element ID of the primary order search input on the orders page.

    Returns:
        None

    Examples:
        goto_admin_orders("70", "http://dockerized-magento.local/index.php/admin/sales_order/", "56")
    """
    click(orders_link_id)
    goto(orders_url)
    click(search_field_id)


def perform_order_search(search_field_id: str, order_query_field_id: str, order_number: str, search_button_id: str):
    """Search for an order by order number using two-stage search interactions.

    Args:
        search_field_id: ID of the initial search input to type the order number into.
        order_query_field_id: ID of the secondary/order-list search input (if present).
        order_number: Order number string to search for (formatted as needed, e.g., '000000301').
        search_button_id: ID of the search/submit button to execute the second-stage search.

    Returns:
        None

    Examples:
        perform_order_search("56", "596", "000000301", "486")
    """
    fill(search_field_id, order_number)
    keyboard_press("Enter")
    fill(order_query_field_id, order_number)
    click(search_button_id)


def open_order_edit(view_order_id: str, edit_tab_id: str, confirm_key: str):
    """Open a specific order from the results and enter the edit/address section.

    Args:
        view_order_id: Element ID to click the order row or "View" link for the located order.
        edit_tab_id: Element ID to click into the edit or addresses area for that order.
        confirm_key: Keyboard key to press to confirm or finalize navigation (e.g., 'Enter').

    Returns:
        None

    Examples:
        open_order_edit("1321", "593", "Enter")
    """
    click(view_order_id)
    click(edit_tab_id)
    keyboard_press(confirm_key)


def set_address_lines(line1_id: str, line2_id: str, city_id: str, line1: str, line2: str, city: str):
    """Fill the main address text fields: line1, line2 and city.

    Args:
        line1_id: Element ID for the street address (line 1) input.
        line2_id: Element ID for the street address (line 2 / suite) input.
        city_id: Element ID for the city input.
        line1: Street address line 1 value.
        line2: Street address line 2 / suite value.
        city: City value.

    Returns:
        None

    Examples:
        set_address_lines("508", "510", "516", "321 Birch Boulevard", "Suite 200", "Dallas")
    """
    fill(line1_id, line1)
    fill(line2_id, line2)
    fill(city_id, city)


def set_region_and_zip(region_select_id: str, zipcode_id: str, region: str, zipcode: str, save_button_id: str):
    """Set the region/state, postal code, and save the address.

    Args:
        region_select_id: Element ID of the region/state select control.
        zipcode_id: Element ID of the postal code input.
        region: Region/state value to select (display value or option value depending on the page).
        zipcode: Postal code string.
        save_button_id: Element ID of the save/update button to persist the address.

    Returns:
        None

    Examples:
        set_region_and_zip("776", "849", "Texas", "75201", "451")
    """
    select_option(region_select_id, region)
    fill(zipcode_id, zipcode)
    click(save_button_id)
