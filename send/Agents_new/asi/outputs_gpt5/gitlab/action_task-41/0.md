def search_and_open_repo(search_bar_id: str, repo_name: str, repo_result_id: str):
    """Search for a repository using the page search bar and open the repository result.

    Args:
        search_bar_id: The element ID of the search input on the page.
        repo_name: The repository name or query string to search for.
        repo_result_id: The element ID of the repository result to click after searching.

    Returns:
        None

    Examples:
        search_and_open_repo('142', 'csvkit', '281')
    """
    click(search_bar_id)                 # focus the search input
    fill(search_bar_id, repo_name)       # type the repository name into the search bar
    keyboard_press("Enter")              # submit the search
    click(repo_result_id)                # open the repository from the search results


def open_contributors_and_report(contributors_tab_id: str, top_contributor_id: str, top_contributor_name: str, commit_count: int):
    """Open the contributors section, navigate to the top contributor, and send a concise report message.

    Args:
        contributors_tab_id: The element ID of the contributors tab or link inside the repository page.
        top_contributor_id: The element ID of the top contributor entry to click (if needed).
        top_contributor_name: The name of the top contributor (used to compose the message).
        commit_count: Number of commits attributed to the top contributor (used to compose the message).

    Returns:
        None

    Examples:
        open_contributors_and_report('519', '267', 'Jane Doe', 123)
    """
    click(contributors_tab_id)                                   # open the contributors section
    click(top_contributor_id)                                    # open the top contributor's details (if needed)
    scroll(0, 300)                                               # ensure the contributor info is visible
    message = f"{top_contributor_name} — {commit_count} commits (the most contributions to the project)."
    send_msg_to_user(message)                                    # send the composed message to the user
