```python
def open_repo_and_open_new_issue(repo_link_id: str, issues_tab_or_new_id: str, title_field_id: str):
    """Navigate to a repository and open the new issue form (focus title field).

    Args:
        repo_link_id: ID of the repository link/button to open the repository.
        issues_tab_or_new_id: ID of the Issues tab or the New Issue button to open the issue list/new issue.
        title_field_id: ID of the title input field to focus for the new issue.

    Returns:
        None

    Examples:
        open_repo_and_open_new_issue('799', '453', '440')
    """
    click(repo_link_id)            # Open the repository page
    click(issues_tab_or_new_id)    # Open Issues page or New Issue entry point
    click(title_field_id)          # Focus the title field to start creating the issue
```

```python
def fill_issue_and_assign(title_field_id: str, title: str, assignee_field_id: str, assignee: str, due_field_id: str, due_date: str, save_button_id: str):
    """Fill issue title, assign a user, set a due date, and save/apply the changes.

    Args:
        title_field_id: ID of the issue title input.
        title: Title string to fill for the issue.
        assignee_field_id: ID of the assignee input/selector.
        assignee: Username to assign the issue to (text to type/select).
        due_field_id: ID of the due date input.
        due_date: Due date string in YYYY-MM-DD format.
        save_button_id: ID of the button to save/apply assignment and due date.

    Returns:
        None

    Examples:
        fill_issue_and_assign('440', 'Integrating LLMs for better prompts', '533', 'Roshanjossey', '649', '2033-04-01', '655')
    """
    fill(title_field_id, title)      # Enter the issue title
    click(assignee_field_id)         # Open assignee selector / focus assignee input
    fill(assignee_field_id, assignee)  # Type/select assignee
    fill(due_field_id, due_date)     # Set the due date
    click(save_button_id)            # Save/apply the assignment and due date
```
