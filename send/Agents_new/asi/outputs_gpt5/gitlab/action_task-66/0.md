def open_repository_license(repo_link_id: str | int, license_file_id: str | int):
    """Navigate to a repository page and open its LICENSE file.

    Args:
        repo_link_id: The UI element id (string or int) for the repository link on the page.
        license_file_id: The UI element id (string or int) for the LICENSE file link in the repo file list.

    Returns:
        None. Performs UI actions to open the repository and view the LICENSE file.

    Examples:
        open_repository_license('784', '1003')
        open_repository_license(227, 1701)
    """
    click(repo_link_id)        # Open the repository page
    click(license_file_id)     # Click the LICENSE file link
    scroll(0, 200)             # Scroll a bit to reveal the file contents
