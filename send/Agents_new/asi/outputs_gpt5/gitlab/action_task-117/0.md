def navigate_to_new_issue(repo_menu_id: str | int, issues_tab_id: str | int, new_issue_button_id: str | int):
    """Navigate from a repository page to the "New Issue" form.

    Args:
        repo_menu_id: ID of the repository/menu element to open the repo.
        issues_tab_id: ID of the "Issues" tab element.
        new_issue_button_id: ID of the "New issue" button/link.

    Returns:
        True if the sequence of navigation actions was executed.

    Examples:
        navigate_to_new_issue('318', '253', '438')
    """
    click(repo_menu_id)
    click(issues_tab_id)
    click(new_issue_button_id)
    return True


def create_issue(title_field_id: str | int, body_field_id: str | int, title_text: str, body_text: str, submit_button_id: str | int):
    """Fill in the issue title and body, then submit the new issue.

    Args:
        title_field_id: ID of the issue title input field.
        body_field_id: ID of the issue body/description textarea.
        title_text: The issue title to be entered.
        body_text: The full issue description to be entered.
        submit_button_id: ID of the button that submits/creates the issue.

    Returns:
        True if the fill and submit actions were executed.

    Examples:
        create_issue('440', '483', 'Bug: example', 'Steps to reproduce...', '655')
    """
    fill(title_field_id, title_text)
    fill(body_field_id, body_text)
    click(submit_button_id)
    return True
