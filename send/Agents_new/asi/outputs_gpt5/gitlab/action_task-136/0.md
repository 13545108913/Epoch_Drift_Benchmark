def start_new_project(new_button_id: str, pages_option_id: str, template_option_id: str, template_variant_id: str):
    """Navigate the UI to the template selection for creating a new project.
    
    This function performs the sequence of clicks that opens the "new project" flow,
    selects the Pages category, chooses the specified template option, and picks a variant.

    Args:
        new_button_id: ID of the "New project" (or similar) button to open the creation flow.
        pages_option_id: ID of the Pages/Plain HTML (or equivalent) category option to select.
        template_option_id: ID of the template option to select (e.g., "Pages/Plain HTML").
        template_variant_id: ID of the specific template variant to click (if applicable).

    Returns:
        None

    Examples:
        start_new_project('223', '242', '441', '442')
    """
    click(new_button_id)
    click(pages_option_id)
    click(template_option_id)
    click(template_variant_id)


def configure_and_create_project(name_field_id: str, project_name: str, create_button_id: str):
    """Fill the project name and trigger creation of the project.

    This function fills the project name field, confirms the input, and clicks the create button.

    Args:
        name_field_id: ID of the project name input field.
        project_name: The desired name for the new project.
        create_button_id: ID of the button that finalizes creation (e.g., "Create" or "Create project").

    Returns:
        None

    Examples:
        configure_and_create_project('543', 'agi_index', '586')
    """
    fill(name_field_id, project_name)
    keyboard_press("Enter")
    click(create_button_id)


def add_project_member(members_tab_id: str, invite_button_id: str, member_field_id: str, member_name: str, confirm_button_id: str):
    """Add a member to the newly created project.

    This function opens the members/invite UI, types the member's name, confirms the selection,
    and finalizes the invite/add action.

    Args:
        members_tab_id: ID of the "Members" or project settings tab to click to reach member controls.
        invite_button_id: ID of the button to open the "Add member" input.
        member_field_id: ID of the input field where member name is entered.
        member_name: The name or identifier to add as a member.
        confirm_button_id: ID of the button that confirms adding the member (e.g., "Add" or selection).

    Returns:
        None

    Examples:
        add_project_member('395', '407', '491', 'Vinta Chen', '513')
    """
    click(members_tab_id)
    click(invite_button_id)
    fill(member_field_id, member_name)
    keyboard_press("Enter")
    click(confirm_button_id)
