```python
def set_gitlab_homepage(profile_menu_id: str | int, edit_profile_id: str | int, homepage_input_id: str | int, save_button_id: str | int, url: str):
    """Set the homepage (Website URL) on a GitLab profile.

    Args:
        profile_menu_id: ID of the profile/menu button to open account options.
        edit_profile_id: ID of the "Edit profile" or "Preferences" link to open profile settings.
        homepage_input_id: ID of the Website/Homepage URL input field.
        save_button_id: ID of the "Save changes" button.
        url: The homepage URL string to set.

    Returns:
        dict: A summary with keys 'status' and 'url' indicating the action performed.

    Examples:
        set_gitlab_homepage('186', '202', '481', '512', 'www.byteblaze.com')
    """
    click(profile_menu_id)
    click(edit_profile_id)
    fill(homepage_input_id, url)
    click(save_button_id)
    return {"status": "submitted", "url": url}
```
