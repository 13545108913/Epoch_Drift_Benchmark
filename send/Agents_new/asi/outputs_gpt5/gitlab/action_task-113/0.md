def goto_and_fill_title_and_description(form_url: str, title_field_id: str | int, desc_field_id: str | int, title: str, description: str):
    """Open the milestone creation page and fill the milestone title and description fields.

    Args:
        form_url: The URL of the milestone creation page.
        title_field_id: The field ID for the milestone title.
        desc_field_id: The field ID for the milestone description.
        title: The milestone title text to fill.
        description: The milestone description text to fill.

    Returns:
        None

    Examples:
        goto_and_fill_title_and_description(
            "http://example.com/project/-/milestones/new",
            "482", "519",
            "Add new branch for zsh comprehensive support",
            "Create a dedicated branch to add comprehensive zsh support..."
        )
    """
    goto(form_url)
    fill(title_field_id, title)
    fill(desc_field_id, description)


def fill_dates_and_submit(start_field_id: str | int, due_field_id: str | int, start_date: str, due_date: str, submit_button_id: str | int):
    """Fill the milestone start and due date fields and submit the form.

    Args:
        start_field_id: The field ID for the milestone start date.
        due_field_id: The field ID for the milestone due date.
        start_date: The start date string (e.g., "2044-05-01").
        due_date: The due date string (e.g., "2044-05-21").
        submit_button_id: The button ID to click to submit/create the milestone.

    Returns:
        None

    Examples:
        fill_dates_and_submit("553", "560", "2044-05-01", "2044-05-21", "564")
    """
    fill(start_field_id, start_date)
    fill(due_field_id, due_date)
    click(submit_button_id)
