def navigate_to_site_editor(home_button_id: str, site_item_id: str, editor_field_id: str):
    """Navigate from the homepage to the site's HTML editor field.

    Args:
        home_button_id: ID of the element that opens the project/site list.
        site_item_id: ID of the element that opens the specific project/site editor.
        editor_field_id: ID of the editable HTML field or code editor to focus.

    Returns:
        None

    Examples:
        navigate_to_site_editor('785', '548', '603')

def replace_editor_content_and_commit(editor_field_id: str, new_content: str, save_button_id: str, confirm_button_id: str):

    Args:
        editor_field_id: ID of the editable HTML field.
        new_content: The full HTML (or other) content to place into the editor.
        save_button_id: ID of the primary save/commit button.
        confirm_button_id: ID of a follow-up confirm or finalise button (may be same as save_button_id).
