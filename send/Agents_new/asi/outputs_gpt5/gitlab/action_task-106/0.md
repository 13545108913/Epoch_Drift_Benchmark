def navigate_to_manage_access(repo_menu_id: str, settings_id: str, manage_access_id: str):
    """Navigate from a repository page to its Manage Access (or collaborators) section.

    Args:
        repo_menu_id: ID of the repository menu or dropdown to open repository options (e.g., '629').
        settings_id: ID of the Settings link or button to open repository settings (e.g., '362').
        manage_access_id: ID of the Manage Access / Collaborators link or button (e.g., '375').

    Returns:
        None

    Examples:
        navigate_to_manage_access('629', '362', '375')
    """
    click(repo_menu_id)
    click(settings_id)
    click(manage_access_id)


def add_user_as_role(input_id: str, username: str, role_select_id: str, role: str, confirm_button_id: str):
    """Add a single user to the repository with a specified role using the invite input, role selector, and confirm control.

    Args:
        input_id: ID of the member invite input field where the username is entered (e.g., '452').
        username: The username to invite as a string (e.g., 'abisubramanya27').
        role_select_id: ID of the role selection control (e.g., '460').
        role: The role to assign to the user (e.g., 'Maintainer').
        confirm_button_id: ID of the invite/confirm button to finalize adding the user (e.g., '474').

    Returns:
        None

    Examples:
        add_user_as_role('452', 'exampleuser', '460', 'Maintainer', '474')
    """
    click(input_id)
    fill(input_id, username)
    keyboard_press("Enter")
    select_option(role_select_id, role)
    click(confirm_button_id)
