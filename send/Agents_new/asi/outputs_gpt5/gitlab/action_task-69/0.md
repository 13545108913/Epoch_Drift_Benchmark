def open_profile_menu(avatar_id: str, status_menu_id: str, profile_option_id: str):
    """Open the user's profile/status menu by clicking the avatar and navigating to the status option.
    
    Args:
        avatar_id: The element ID for the user's avatar (click to open the user menu).
        status_menu_id: The element ID for the user menu entry that opens profile/status options.
        profile_option_id: The element ID for the profile or status option to click into the status editor.
    
    Returns:
        bool: True if the sequence of clicks was executed (no error handling performed).
    
    Examples:
        open_profile_menu('186', '198', '950')
    """
    click(avatar_id)         # open user menu
    click(status_menu_id)    # open profile/status section
    click(profile_option_id) # enter the status editor
    return True


def update_gitlab_status(status_field_id: str, status_text: str, save_button_id: str, close_menu_id: str):
    """Fill in the GitLab status text, save it, and close the menu.
    
    Args:
        status_field_id: The element ID of the status text input field.
        status_text: The status text to set (e.g., "Playing Badminton.").
        save_button_id: The element ID of the save/apply button for the status.
        close_menu_id: The element ID to click to close the user menu after saving.
    
    Returns:
        str: A short status string indicating the attempted update (e.g., "updated").
    
    Examples:
        update_gitlab_status('944', 'Playing Badminton.', '950', '186')
    """
    fill(status_field_id, status_text)  # input the desired status text
    click(save_button_id)               # save/apply the status
    click(close_menu_id)                # close the user menu
    return "updated"
