```python
def open_collaborator_invite_panel(repo_link_id: str | int, manage_access_id: str | int, invite_button_id: str | int):
    """Open the repository page, navigate to Manage Access, and open the Invite Collaborators panel.

    Args:
        repo_link_id: The ID of the repository link/button to open the repo.
        manage_access_id: The ID of the "Manage access" (or equivalent) tab/button.
        invite_button_id: The ID of the "Invite collaborators" button to open the invite dialog.

    Returns:
        None

    Examples:
        open_collaborator_invite_panel('725', '395', '407')
    """
    click(repo_link_id)
    click(manage_access_id)
    click(invite_button_id)


def invite_collaborator(input_field_id: str | int, collaborator_name: str, role_select_id: str | int, role_option: str, confirm_button_id: str | int):
    """Type a collaborator's name into the invite field, choose a role, and confirm the invite.

    Args:
        input_field_id: The ID of the collaborator name input field.
        collaborator_name: The full name or username of the collaborator to invite.
        role_select_id: The ID of the role dropdown/select element.
        role_option: The visible option value to select for the collaborator's role (e.g., "Developer").
        confirm_button_id: The ID of the final confirm/submit button to send the invite.

    Returns:
        None

    Examples:
        invite_collaborator('493', 'Benoît Blanchon', '501', 'Developer', '515')
    """
    fill(input_field_id, collaborator_name)
    keyboard_press("Enter")
    select_option(role_select_id, role_option)
    click(confirm_button_id)
```
