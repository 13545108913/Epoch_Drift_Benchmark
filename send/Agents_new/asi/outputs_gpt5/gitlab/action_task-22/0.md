def navigate_to_issues(home_id: str, repo_id: str, issues_section_id: str):
    """Navigate from the homepage to the repository's issues section.
    
    Args:
        home_id: ID of the homepage or main navigation element to click first.
        repo_id: ID of the repository element to click to open the repo.
        issues_section_id: ID of the issues section or link inside the repo.
    
    Returns:
        None
    
    Examples:
        navigate_to_issues('161', '443', '250')
    """
    click(home_id)
    click(repo_id)
    click(issues_section_id)


def filter_issues_by_keyword(search_box_id: str, keyword: str, wait_ms: int = 2000):
    """Enter a keyword into the issues search/filter box and submit the query.
    
    Args:
        search_box_id: ID of the search or filter input for issues.
        keyword: Text to search for in issue titles.
        wait_ms: Milliseconds to wait after submitting the search.
    
    Returns:
        None
    
    Examples:
        filter_issues_by_keyword('144', 'homepage content', 2000)
    """
    click(search_box_id)
    fill(search_box_id, keyword)
    keyboard_press("Enter")
    noop(wait_ms)


def open_latest_created_issue(open_button_id: str, details_tab_id: str, status_button_id: str):
    """Open the most recent issue from the results and navigate to its details/status.
    
    Args:
        open_button_id: ID of the element that opens the latest issue (e.g., first result).
        details_tab_id: ID of a details or related tab to view more info.
        status_button_id: ID of the control that reveals the issue's status or state.
    
    Returns:
        None
    
    Examples:
        open_latest_created_issue('256', '535', '272')
    """
    click(open_button_id)
    click(details_tab_id)
    click(status_button_id)
