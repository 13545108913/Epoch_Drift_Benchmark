```python
def open_latest_issue_with_keyword(nav_button_id: str, issues_list_id: str, keyword: str):
    """Navigate to the issues area, open the latest updated issue matching a keyword, and notify the user.

    Args:
        nav_button_id: ID of the navigation element to access the project or repository issues (e.g., "Issues" link).
        issues_list_id: ID of the element that lists or filters issues (e.g., the issues tab or search result).
        keyword: Keyword expected to appear in the issue title.

    Returns:
        None

    Examples:
        open_latest_issue_with_keyword('161', '788', 'better')
    """
    click(nav_button_id)             # Open the project/repo or main issues navigation
    click(issues_list_id)            # Open the issues list or the filtered results
    send_msg_to_user(
        f'Opened the latest updated issue containing the keyword "{keyword}" in its title. Please check whether it is closed.'
    )
```
