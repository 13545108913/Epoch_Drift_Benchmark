def search_and_open_project(search_bar_id: str | int, project_name: str, project_result_id: str | int):
    """Search for a project by name and open the project's page.

    Args:
        search_bar_id: The UI element id of the search input.
        project_name: The project name to search for.
        project_result_id: The UI element id of the project entry to click in the results.

    Returns:
        None

    Examples:
        search_and_open_project('238', 'Super_Awesome_Robot', '273')
    """
    click(search_bar_id)               # focus the search input
    fill(search_bar_id, project_name)  # enter the project name into the search input
    keyboard_press('Enter')            # submit the search
    click(project_result_id)           # open the project from the results


def compose_and_send_ssh_clone(ssh_host: str, namespace: str, repo_name: str, extra_note: str | None = None):
    """Compose a git clone command for SSH and send it to the user.

    Args:
        ssh_host: The SSH host prefix, e.g. 'git@172.26.116.102'
        namespace: The repository namespace or owner, e.g. 'byteblaze'
        repo_name: The repository name, e.g. 'Super_Awesome_Robot'
        extra_note: Optional additional note to include in the message.

    Returns:
        None

    Examples:
        compose_and_send_ssh_clone('git@172.26.116.102', 'byteblaze', 'Super_Awesome_Robot')
    """
    ssh_url = f"{ssh_host}:{namespace}/{repo_name}.git"
    message = f"Run:\n\ngit clone {ssh_url}\n\nMake sure your SSH key is added to your GitLab account and replace the namespace if needed."
    if extra_note:
        message = message + "\n\n" + extra_note
    send_msg_to_user(message)
