```python
def set_profile_homepage(account_menu_id: str, profile_link_id: str, website_field_id: str, save_button_id: str, homepage_url: str):
    """Navigate to a user's profile settings and set the homepage/website URL.

    Args:
        account_menu_id: ID of the control that opens the account menu (e.g., "186").
        profile_link_id: ID of the control/link that navigates to profile/settings (e.g., "202").
        website_field_id: ID of the input field for the website/homepage URL (e.g., "481").
        save_button_id: ID of the button to save profile changes (e.g., "512").
        homepage_url: The URL string to set as the homepage on the profile.

    Returns:
        A dict with the performed actions and the homepage_url that was set.

    Examples:
        set_profile_homepage('186', '202', '481', '512', 'https://example.com')
    """
    click(account_menu_id)          # open account menu
    click(profile_link_id)          # navigate to profile/edit settings
    fill(website_field_id, homepage_url)  # set the homepage/website URL
    click(save_button_id)           # save the profile changes
    return {"action": "set_homepage", "homepage_url": homepage_url}
```
