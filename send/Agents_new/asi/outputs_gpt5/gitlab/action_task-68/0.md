```python
def set_gitlab_status(menu_id: str, edit_status_id: str, status_input_id: str, save_button_id: str, status_text: str):
    """Open the account menu, set the GitLab status text, save it, and notify the user.

    Args:
        menu_id: The UI element id that opens the account menu (e.g., '186').
        edit_status_id: The UI element id that opens the status editor (e.g., '198').
        status_input_id: The UI element id of the status text input (e.g., '916').
        save_button_id: The UI element id of the button that saves the status (e.g., '922').
        status_text: The status string to set.

    Returns:
        dict: A simple result with keys 'status_text' and 'updated' indicating the status set.

    Examples:
        set_gitlab_status('186', '198', '916', '922', 'Enjoying life.')
    """
    click(menu_id)  # Open account menu
    click(edit_status_id)  # Open status editor
    fill(status_input_id, status_text)  # Enter desired status text
    click(save_button_id)  # Save the status
    send_msg_to_user(f'Your GitLab status has been set to "{status_text}". I can see it in the account menu.')
    return {"status_text": status_text, "updated": True}
```
