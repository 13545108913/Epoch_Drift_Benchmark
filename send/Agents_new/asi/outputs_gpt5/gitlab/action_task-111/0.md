```python
def open_new_milestone_page(milestones_url: str, new_button_id: str, title_field_id: str, title: str):
    """Open the milestones page, start creating a new milestone, and fill the milestone title.
    
    Args:
        milestones_url: The URL of the milestones listing page.
        new_button_id: The UI element id for the "New milestone" or "Create milestone" button.
        title_field_id: The input field id for the milestone title.
        title: The title text to put into the milestone title field.
    
    Returns:
        None. After running this function the new milestone form should be open with the title filled.
    
    Examples:
        open_new_milestone_page(
            "http://example.com/project/-/milestones",
            "459",
            "446",
            "Clean sensitive information"
        )
    """
    goto(milestones_url)
    click(new_button_id)
    fill(title_field_id, title)


def complete_and_submit_milestone(description_field_id: str, description: str,
                                  start_date_field_id: str, start_date: str,
                                  due_date_field_id: str, due_date: str,
                                  create_button_id: str):
    """Fill milestone details (description, start date, due date) and submit the milestone form.
    
    Args:
        description_field_id: The input or textarea id for the milestone description.
        description: The description text for the milestone.
        start_date_field_id: The input field id for the milestone start date.
        start_date: The start date string in the expected format (e.g., "YYYY-MM-DD").
        due_date_field_id: The input field id for the milestone due date.
        due_date: The due date string in the expected format (e.g., "YYYY-MM-DD").
        create_button_id: The UI element id for the "Create" or "Save" button.
    
    Returns:
        None. After running this function the milestone form will be submitted.
    
    Examples:
        complete_and_submit_milestone(
            "483",
            "Task: Clean sensitive information from the repository, issues, and attachments. Start on 2023-02-16 and complete within 20 days.",
            "517",
            "2023-02-16",
            "524",
            "2023-03-08",
            "528"
        )
    """
    fill(description_field_id, description)
    fill(start_date_field_id, start_date)
    fill(due_date_field_id, due_date)
    click(create_button_id)
```
