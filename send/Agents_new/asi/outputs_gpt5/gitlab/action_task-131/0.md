def create_new_project(home_button_id: str, name_field_id: str, slug_field_id: str, project_name: str, visibility_toggle_id: str, create_button_id: str):
    """Create a new project by filling the project name/slug and toggling visibility.

    Args:
        home_button_id: ID of the element to open the project creation UI (e.g., "223").
        name_field_id: ID of the project name input field (e.g., "259").
        slug_field_id: ID of the project slug/input field (e.g., "270").
        project_name: The project name to set for both name and slug.
        visibility_toggle_id: ID of the visibility/public toggle (e.g., "296").
        create_button_id: ID of the final create/confirm button (e.g., "310").

    Returns:
        None

    Examples:
        create_new_project('223', '259', '270', 'awesome-llms', '296', '310')
    """
    click(home_button_id)
    fill(name_field_id, project_name)
    fill(slug_field_id, project_name)
    click(visibility_toggle_id)
    click(create_button_id)


def add_project_member(invite_button_id: str, search_field_id: str, member_name: str, select_result_id: str, wait_before_select_ms: int):
    """Add a member to the current project by opening the invite UI, searching, and selecting the result.

    Args:
        invite_button_id: ID of the button that opens the invite/search input (e.g., "375").
        search_field_id: ID of the member search input field (e.g., "455").
        member_name: The member username to search and add (e.g., "abishek").
        select_result_id: ID of the element to click to select the searched user (e.g., "477").
        wait_before_select_ms: Milliseconds to wait after triggering the search before selecting the result.

    Returns:
        None

    Examples:
        add_project_member('375', '455', 'abishek', '477', 500)
    """
    click(invite_button_id)
    fill(search_field_id, member_name)
    keyboard_press("Enter")
    noop(wait_before_select_ms)
    click(select_result_id)
