```python
def create_repository_issue(repo_link_id: str, issues_tab_id: str, title_field_id: str, body_field_id: str, submit_button_id: str, issue_title: str, issue_body: str) -> dict:
    """Create a new issue in a repository by navigating to the repo, opening the Issues tab,
    filling the title and body, and submitting the issue.

    Args:
        repo_link_id: ID of the repository link/button to open the repository.
        issues_tab_id: ID of the "Issues" tab/button inside the repository.
        title_field_id: ID of the issue title input field.
        body_field_id: ID of the issue body/description input field.
        submit_button_id: ID of the button that submits the new issue.
        issue_title: The title string for the new issue.
        issue_body: The body/description string for the new issue.

    Returns:
        A dict containing the submitted issue's title and body as confirmation.

    Examples:
        create_repository_issue("626", "438", "440", "483", "655",
                                "Feature request: X", "Please consider adding X because...")
    """
    click(repo_link_id)
    click(issues_tab_id)
    fill(title_field_id, issue_title)
    fill(body_field_id, issue_body)
    click(submit_button_id)
    return {"title": issue_title, "body": issue_body}
```
