def post_merge_request_comment(repo_link_id: str | int, mr_link_id: str | int, comment_field_id: str | int, submit_button_id: str | int, comment_text: str):
    """Post a text comment to a merge request in a repository/project UI.

    This performs the typical sequence of UI interactions to open the project, open the merge request,
    enter the comment text, and submit the comment.

    Args:
        repo_link_id: ID of the link/button that opens the repository/project.
        mr_link_id: ID of the merge request link/button to open the specific MR.
        comment_field_id: ID of the comment input field where text is entered.
        submit_button_id: ID of the button used to submit the comment.
        comment_text: The comment text to post on the merge request.

    Returns:
        None

    Examples:
        post_merge_request_comment('302', '786', '735', '764', 'Looks good to me')
    """
    click(repo_link_id)
    click(mr_link_id)
    fill(comment_field_id, comment_text)
    click(submit_button_id)
