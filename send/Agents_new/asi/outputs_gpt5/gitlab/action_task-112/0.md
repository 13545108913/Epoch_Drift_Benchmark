```python
def open_repo_milestones(repo_link_id: str | int, milestones_link_id: str | int):
    """Navigate from a repository link to the repository's milestones page.

    Args:
        repo_link_id: ID of the repository link or element to open the repo.
        milestones_link_id: ID of the link or tab that opens the milestones view.

    Returns:
        None - performs UI navigation actions.

    Examples:
        open_repo_milestones('102', '378')
    """
    click(repo_link_id)          # open the repository page
    click(milestones_link_id)    # open the milestones tab or link
    scroll(0, 200)               # ensure milestone area is visible


def create_milestone(new_button_id: str | int,
                     title_field_id: str | int,
                     date_range_field_id: str | int,
                     create_button_id: str | int,
                     title: str,
                     start_date: str,
                     end_date: str):
    """Open the new milestone form, fill title and date range, and create the milestone.

    Args:
        new_button_id: ID of the "New milestone" button.
        title_field_id: ID of the milestone title input field.
        date_range_field_id: ID of the date-range input (accepts combined start/end).
        create_button_id: ID of the button that submits/creates the milestone.
        title: Milestone title string.
        start_date: Start date string (e.g., 'Mar 15, 2044').
        end_date: End/due date string (e.g., 'Mar 30, 2044').

    Returns:
        None - performs UI actions to create the milestone.

    Examples:
        create_milestone(
            'new_btn', 'title_input', 'date_range_input', 'create_btn',
            'Merge all branches to main', 'Mar 15, 2044', 'Mar 30, 2044'
        )
    """
    click(new_button_id)                                    # open the new milestone form
    fill(title_field_id, title)                             # enter milestone title
    fill(date_range_field_id, f"{start_date} — {end_date}") # enter combined date range
    click(create_button_id)                                 # submit/create the milestone
```
