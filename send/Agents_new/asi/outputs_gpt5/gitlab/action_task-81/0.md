```python
def set_gitlab_profile_homepage(account_button_id: str, edit_profile_button_id: str, homepage_field_id: str, url: str, save_button_id: str):
    """Navigate to GitLab profile settings and set the Website (homepage) URL.
    
    Steps:
    1. Open the account/user menu.
    2. Open the Edit profile / Profile settings page.
    3. Focus the Website field, fill in the new URL, and save changes.
    
    Args:
        account_button_id: ID of the account/user menu button to open (string).
        edit_profile_button_id: ID of the "Edit profile" or profile settings link (string).
        homepage_field_id: ID of the Website/homepage input field (string).
        url: The homepage URL to set (string).
        save_button_id: ID of the button to save profile changes (string).
    
    Returns:
        None
    
    Examples:
        set_gitlab_profile_homepage('186', '199', '478', 'example.com', '509')
    """
    click(account_button_id)            # open account/user menu
    click(edit_profile_button_id)       # open Edit profile / Profile settings
    click(homepage_field_id)            # focus the Website/homepage field
    fill(homepage_field_id, url)        # enter the homepage URL
    click(save_button_id)               # save profile changes
```
