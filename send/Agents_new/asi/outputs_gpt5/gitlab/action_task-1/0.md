def open_repository_issues(repo_link_id: str, issues_tab_id: str, open_filter_id: str):
    """Navigate to a repository's Issues tab and open the list of open issues.

    Args:
        repo_link_id: The clickable ID that opens the repository page.
        issues_tab_id: The clickable ID that opens the Issues tab on the repository page.
        open_filter_id: The clickable ID that applies the "Open issues" filter or otherwise ensures open issues are shown.

    Returns:
        None

    Examples:
        open_repository_issues('280', '684', '807')
    """
    click(repo_link_id)      # open the repository page
    click(issues_tab_id)     # open the Issues tab
    click(open_filter_id)    # ensure the list is filtered to open issues


def refresh_and_toggle_issue_list(refresh_id: str, toggle_id_a: str, toggle_id_b: str):
    """Refresh or toggle the issues list layout/state to surface the most recent items.

    Args:
        refresh_id: The clickable ID for refreshing the issues list or reloading the view.
        toggle_id_a: The first clickable ID used to change the list view or sort state.
        toggle_id_b: The second clickable ID used to finalize the view/sort change.

    Returns:
        None

    Examples:
        refresh_and_toggle_issue_list('284', '807', '284')
    """
    click(refresh_id)   # refresh or reload the issues listing
    click(toggle_id_a)  # toggle or sort to bring recent issues to top
    click(toggle_id_b)  # finalize the view change so the most recent open issues are visible
