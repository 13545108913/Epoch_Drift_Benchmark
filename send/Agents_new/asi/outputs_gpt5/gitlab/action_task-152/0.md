## Reusable Functions

def open_groups_section(menu_id: str | int, groups_link_id: str | int, new_group_btn_id: str | int):
    """Navigate to the Groups creation UI by clicking the sequence of links/buttons.

    Args:
        menu_id: The ID of the main menu or "Groups" entry to open (e.g., '76').
        groups_link_id: The ID of the groups subsection link (e.g., '83').
        new_group_btn_id: The ID of the button to start creating a new group (e.g., '222').

    Returns:
        None

    Examples:
        open_groups_section('76', '83', '222')
    """
    click(menu_id)
    click(groups_link_id)
    click(new_group_btn_id)


def create_new_group(name_field_id: str | int, group_name: str, display_name_field_id: str | int, display_name: str, create_button_id: str | int):
    """Fill in group name and display name, then click the create/save button.

    Args:
        name_field_id: The input field ID for the group's internal name.
        group_name: The group's internal name string to fill.
        display_name_field_id: The input field ID for the group's display name.
        display_name: The group's display name string to fill.
        create_button_id: The button ID to finalize creating the group.

    Returns:
        None

    Examples:
        create_new_group('235', 'coding_friends', '245', 'coding_friends', '287')
    """
    fill(name_field_id, group_name)
    fill(display_name_field_id, display_name)
    click(create_button_id)


def add_group_members(member_input_id: str | int, members: list[str], suggestion_click_id: str | int):
    """Add one or more members to the current group by filling the member input, confirming, and selecting suggestion.

    This will iterate through the provided members list and for each member:
      - fill the member input
      - wait briefly (noop)
      - press Enter to confirm
      - click the suggestion/confirm element

    Args:
        member_input_id: The input field ID where member names are typed.
        members: A list of member name strings to add.
        suggestion_click_id: The ID of the UI element to click to finalize selecting the suggested member.

    Returns:
        None

    Examples:
        add_group_members('363', ['Agnes-U'], '385')
    """
    for m in members:
        fill(member_input_id, m)
        noop(500)
        keyboard_press("Enter")
        click(suggestion_click_id)
