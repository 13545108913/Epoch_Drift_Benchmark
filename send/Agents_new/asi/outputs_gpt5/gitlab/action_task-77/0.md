```python
def assign_github_issue(issue_link_id: str, assignee_menu_id: str, assignee_input_id: str, assignee_select_id: str, issue_number: str, assignee_handle: str):
    """Assign a GitHub issue to a user and notify the user via UI messaging.

    Steps:
    1. Open the issue page.
    2. Open the assignee menu, search for the assignee, and select them.
    3. Send a notification message confirming the assignment.

    Args:
        issue_link_id: The UI element ID that opens the target issue (e.g., '427').
        assignee_menu_id: The UI element ID for the assignee dropdown/menu (e.g., '810').
        assignee_input_id: The UI element ID for the assignee search/input field (e.g., '732').
        assignee_select_id: The UI element ID for selecting the assignee from results (e.g., '987').
        issue_number: The issue number as a string (e.g., '611').
        assignee_handle: The assignee's username without '@' (e.g., 'Roshanjossey').

    Returns:
        None. Performs UI actions and sends a confirmation message via send_msg_to_user.

    Examples:
        assign_github_issue('427', '810', '732', '987', '611', 'Roshanjossey')
    """
    click(issue_link_id)                # Open the specific issue
    click(assignee_menu_id)             # Open the assignee dropdown/menu
    fill(assignee_input_id, assignee_handle)  # Type the assignee's username into the search/input
    click(assignee_select_id)           # Select the assignee from the search results
    send_msg_to_user(f"Assigned issue #{issue_number} to @{assignee_handle}.")  # Notify confirmation
```
