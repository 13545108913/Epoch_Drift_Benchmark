## Reusable Functions
```python
def open_repo_issue_creation(repo_link_id: str, issues_tab_id: str, new_issue_button_id: str):
    """Navigate from a repository link to the new issue creation page.

    Args:
        repo_link_id: The clickable ID for the repository (e.g., "527").
        issues_tab_id: The clickable ID for the repository's "Issues" tab (e.g., "283").
        new_issue_button_id: The clickable ID for the "New issue" button (e.g., "491").

    Returns:
        None

    Examples:
        open_repo_issue_creation('527', '283', '491')
    """
    click(repo_link_id)           # Open the repository
    click(issues_tab_id)          # Open the Issues tab
    click(new_issue_button_id)    # Click "New issue" to start issue creation


def assign_and_set_due(assignee_button_id: str, due_date_field_id: str, due_date: str, save_button_id: str):
    """Assign the issue and set its due date, then save changes.

    Args:
        assignee_button_id: The clickable ID that assigns the selected user (e.g., "712").
        due_date_field_id: The input ID for the due date field (e.g., "687").
        due_date: The due date string in YYYY-MM-DD format (e.g., "2030-12-31").
        save_button_id: The clickable ID for the save/confirm button (e.g., "693").

    Returns:
        None

    Examples:
        assign_and_set_due('712', '687', '2030-12-31', '693')
    """
    click(assignee_button_id)          # Assign the issue to the target account
    fill(due_date_field_id, due_date)  # Enter the due date
    click(save_button_id)              # Save/confirm the issue metadata
```
