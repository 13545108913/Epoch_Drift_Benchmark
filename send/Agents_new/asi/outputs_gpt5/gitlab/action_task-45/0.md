```python
def search_repository(search_bar_id: str | int, repository_name: str):
    """Search for a repository using the provided search bar element.
    
    Args:
        search_bar_id: ID of the search input element (string or int).
        repository_name: The repository name or query to search for.
        
    Returns:
        None. Performs UI actions to execute the search.
        
    Examples:
        search_repository('238', 'metaseq')
    """
    click(search_bar_id)
    fill(search_bar_id, repository_name)
    keyboard_press("Enter")


def open_repo_and_contributors(result_item_id: str | int, repo_link_id: str | int, contributors_tab_id: str | int):
    """Open a repository from search results and navigate to its Contributors tab.
    
    Args:
        result_item_id: ID of the search-result item to click.
        repo_link_id: ID of an element on the repo page to click (e.g., main repo link).
        contributors_tab_id: ID of the Contributors tab or link.
        
    Returns:
        None. Performs the UI navigation to the contributors list.
        
    Examples:
        open_repo_and_contributors('281', '519', '267')
    """
    click(result_item_id)
    scroll(0, 1000)
    click(repo_link_id)
    click(contributors_tab_id)


def report_top_contributors(repo_name: str, attribute: str, top_contributors: list[tuple[str, int]]):
    """Format and send a message reporting the top contributors and their attribute values.
    
    Args:
        repo_name: Name of the repository (used in the message).
        attribute: The attribute to report (e.g., "number of commits").
        top_contributors: A list of tuples (contributor_name, attribute_value), ordered by rank.
        
    Returns:
        None. Sends a formatted message to the user describing the top contributors.
        
    Examples:
        report_top_contributors('metaseq', 'number of commits', [('Susan Zhang', 71), ('Stephen Roller', 51), ('Punit Singh Koura', 14)])
    """
    scroll(0, 300)
    lines = [f"Top {len(top_contributors)} contributors to {repo_name} by {attribute}:"]
    for idx, (name, value) in enumerate(top_contributors, start=1):
        lines.append(f"{idx}. {name} — {value} {attribute}")
    message = "\n".join(lines)
    send_msg_to_user(message)
```
