def create_new_project(open_button_id: str, name_field_id: str, slug_field_id: str, project_name: str, visibility_button_id: str, create_button_id: str):
    """Open the new project dialog, fill project metadata, set visibility and create the project.

    Args:
        open_button_id: ID of the button/link that opens the "new project" UI.
        name_field_id: ID of the project name input field.
        slug_field_id: ID of the project slug/input field.
        project_name: The name (and slug) to set for the new project.
        visibility_button_id: ID of the UI control to set project visibility (e.g., public/private).
        create_button_id: ID of the button that finalizes project creation.

    Returns:
        None

    Examples:
        create_new_project('223', '259', '270', 'web_arena', '296', '310')
    """
    click(open_button_id)
    fill(name_field_id, project_name)
    fill(slug_field_id, project_name)
    click(visibility_button_id)
    click(create_button_id)


def open_project_members(menu_id: str, project_settings_id: str, members_tab_id: str, manage_members_id: str):
    """Navigate through the project UI to reach the member management interface.

    Args:
        menu_id: ID of top-level menu or link to access projects.
        project_settings_id: ID of the project/settings link to open project details.
        members_tab_id: ID of the members tab or section.
        manage_members_id: ID of the manage/add members control.

    Returns:
        None

    Examples:
        open_project_members('362', '375', '372', '375')
    """
    click(menu_id)
    click(project_settings_id)
    click(members_tab_id)
    click(manage_members_id)


def add_members(member_input_id: str, member_names: list, finalize_button_id: str):
    """Add a list of members by searching/typing each name into the member input and confirming.

    Args:
        member_input_id: ID of the member search/input field.
        member_names: List of member names (strings) to add.
        finalize_button_id: ID of the button to confirm/save added members.

    Returns:
        None

    Examples:
        add_members('455', ['Abishek', 'Vinta'], '477')
    """
    for name in member_names:
        click(member_input_id)
        fill(member_input_id, name)
        keyboard_press("Enter")
    click(finalize_button_id)
