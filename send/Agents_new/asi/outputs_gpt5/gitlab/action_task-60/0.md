```python
def fork_repository(step_ids: list):
    """Fork a repository by clicking a sequence of page elements.

    Args:
        step_ids: A list of element IDs to click in order. Typical order:
                  [owner_link_id, repo_link_id, fork_button_id, confirm_button_id].
                  Provide at least three IDs; a fourth is optional for confirmation.

    Returns:
        bool: True if the clicks were executed, False if insufficient IDs were provided.

    Examples:
        fork_repository(['234', '486', '472', '412'])
    """
    if len(step_ids) < 3:
        return False
    click(step_ids[0])
    click(step_ids[1])
    click(step_ids[2])
    if len(step_ids) >= 4:
        click(step_ids[3])
    return True
```
