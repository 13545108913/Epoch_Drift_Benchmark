def open_new_project_from_menu(menu_button_id: str | int, new_project_button_id: str | int, create_from_template_id: str | int):
    """Navigate from the main menu to the "Create from template" page.

    Args:
        menu_button_id: ID of the main menu or "Projects" link to open project options.
        new_project_button_id: ID of the "New project" button/link.
        create_from_template_id: ID of the "Create from template" control.

    Returns:
        None

    Examples:
        open_new_project_from_menu('223', '242', '417')
    """
    click(menu_button_id)
    click(new_project_button_id)
    click(create_from_template_id)


def select_template_with_scrolls(scroll_offsets: list[int], template_card_id: str | int, use_template_button_id: str | int):
    """Reveal and select a project template by scrolling and clicking the template card, then apply it.

    This handles cases where the template controls may be obstructed and require multiple scrolls
    and re-selections before the "Use template" control is clickable.

    Args:
        scroll_offsets: A list of vertical offsets (integers) to scroll to in sequence, e.g., [400, 800].
        template_card_id: The ID of the template card to click (select).
        use_template_button_id: The ID of the "Use template" control to apply the template.

    Returns:
        None

    Examples:
        select_template_with_scrolls([400, 800], '429', '430')
    """
    for offset in scroll_offsets:
        scroll(0, offset)
        click(template_card_id)
    click(use_template_button_id)


def create_private_project(project_name_field_id: str | int, project_slug_field_id: str | int, create_button_id: str | int, project_name: str):
    """Fill project name and slug and create a private project.

    Args:
        project_name_field_id: The input field ID for the project name.
        project_slug_field_id: The input field ID for the project path/slug.
        create_button_id: The button ID to submit and create the project.
        project_name: The string name to use for both project name and slug.

    Returns:
        None

    Examples:
        create_private_project('543', '554', '586', '11711_gitlab')
    """
    fill(project_name_field_id, project_name)
    fill(project_slug_field_id, project_name)
    click(create_button_id)
