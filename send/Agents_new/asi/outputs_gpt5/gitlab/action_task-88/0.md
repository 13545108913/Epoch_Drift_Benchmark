```python
def create_empty_repository(new_repo_button_id: str, repo_name_field_id: str, repo_slug_field_id: str, create_button_id: str, project_name: str):
    """Open the new-repository form and create an empty repository with the given name.

    Steps:
    1. Open the new repository dialog.
    2. Fill the repository name and slug/confirm field.
    3. Click the create button.

    Args:
        new_repo_button_id: ID of the UI element that opens the "new repository" form (e.g., "New" or "Create repository").
        repo_name_field_id: ID of the input field for the repository name.
        repo_slug_field_id: ID of a secondary input related to repository creation (slug/confirm/identifier).
        create_button_id: ID of the button to finalize repository creation.
        project_name: The desired repository name to fill into the form.

    Returns:
        dict: A summary mapping action names to the IDs/values used, e.g.
              {"opened": new_repo_button_id, "name_filled": project_name, "created_with": create_button_id}

    Examples:
        create_empty_repository("223", "259", "270", "310", "awesome_webagent")
    """
    click(new_repo_button_id)
    fill(repo_name_field_id, project_name)
    fill(repo_slug_field_id, project_name)
    click(create_button_id)
    return {
        "opened": new_repo_button_id,
        "name_filled": project_name,
        "created_with": create_button_id
    }
```
