```python
def update_gitlab_profile_homepage(profile_menu_id: str, edit_profile_id: str, homepage_field_id: str, save_button_id: str, url: str) -> None:
    """Open the GitLab profile edit form and set the homepage/website URL.

    This performs the typical sequence of opening the profile menu, entering the edit profile view,
    filling the Website/Homepage field, and saving the changes.

    Args:
        profile_menu_id: The element ID to open the user/profile menu (e.g., "186").
        edit_profile_id: The element ID to open the "Edit profile" page (e.g., "202").
        homepage_field_id: The field ID for the Website/Homepage input (e.g., "481").
        save_button_id: The element ID for the Save/Update button (e.g., "512").
        url: The homepage/website URL to set on the profile.

    Returns:
        None

    Examples:
        update_gitlab_profile_homepage("186", "202", "481", "512", "https://helloworld.xyz")
    """
    click(profile_menu_id)
    click(edit_profile_id)
    fill(homepage_field_id, url)
    click(save_button_id)
```
