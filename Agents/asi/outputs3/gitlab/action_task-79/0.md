```python
def update_gitlab_homepage(profile_link_id: str, settings_link_id: str, website_field_id: str, update_button_id: str, new_url: str):
    """Update the homepage URL in GitLab profile settings.
    
    Args:
        profile_link_id: ID of the profile link element
        settings_link_id: ID of the settings link element
        website_field_id: ID of the website URL input field
        update_button_id: ID of the update profile settings button
        new_url: The new homepage URL to set
        
    Returns:
        None
        
    Examples:
        update_gitlab_homepage('186', '202', '481', '512', 'https://example.com')
    """
    click(profile_link_id)  # Click profile link to access profile page
    click(settings_link_id)  # Click settings link to access profile settings
    fill(website_field_id, new_url)  # Fill website URL field with new URL
    click(update_button_id)  # Click update button to save changes
```
