```python
def navigate_to_issues(issues_link_id: str):
    """Navigate to the issues page from the current page.
    
    Args:
        issues_link_id: The ID of the Issues link
        
    Returns:
        None
        
    Examples:
        navigate_to_issues('161')
    """
    click(issues_link_id)
```

```python
def open_issue_by_title(issue_title: str):
    """Open an issue by clicking on its title link.
    
    Args:
        issue_title: The title of the issue to open
        
    Returns:
        None
        
    Examples:
        open_issue_by_title('Better initial load experience')
    """
    click(issue_title)
```

```python
def report_issue_status(issue_title: str, issue_number: str, status: str):
    """Report the status of an issue to the user.
    
    Args:
        issue_title: The title of the issue
        issue_number: The issue number
        status: The current status of the issue
        
    Returns:
        None
        
    Examples:
        report_issue_status('Better initial load experience', '#8', 'open')
    """
    message = f"The issue '{issue_title}' ({issue_number}) is currently {status}."
    send_msg_to_user(message)
```
