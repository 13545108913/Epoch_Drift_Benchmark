def create_issue(issues_link_id: str, new_issue_button_id: str, title_field_id: str, description_field_id: str, submit_button_id: str, title: str, description: str):
    """Create a new issue with the given title and description.
    
    Args:
        issues_link_id: ID of the Issues link
        new_issue_button_id: ID of the New Issue button
        title_field_id: ID of the title input field
        description_field_id: ID of the description textarea
        submit_button_id: ID of the submit button
        title: The issue title
        description: The issue description
        
    Returns:
        None
        
    Examples:
        create_issue('283', '491', '478', '521', '693', 'Implement default plugins for .zsh', 'Let's discuss which default plugins should be included...')
    """

    click(new_issue_button_id)  # Click New Issue button
    fill(title_field_id, title)  # Fill title field
    fill(description_field_id, description)  # Fill description field
    click(submit_button_id)  # Click submit button
