def search_and_select_repository(search_bar_id: str, repo_name: str, repo_link_id: str):
    """Search for a repository and select it from search results.
    
    Args:
        search_bar_id: ID of the search bar element
        repo_name: Name of the repository to search for
        repo_link_id: ID of the repository link in search results
        
    Returns:
        None
        
    Examples:
        search_and_select_repository('241', 'a11yproject', '289')
    """
    click(search_bar_id)
    fill(search_bar_id, repo_name)
    keyboard_press("Enter")
    click(repo_link_id)

def navigate_to_new_issue(issues_link_id: str, new_issue_button_id: str):
    """Navigate from repository page to new issue creation page.
    
    Args:
        issues_link_id: ID of the Issues link
        new_issue_button_id: ID of the New issue button
        
    Returns:
        None
        
    Examples:
        navigate_to_new_issue('283', '491')
    """
    click(issues_link_id)
    click(new_issue_button_id)

def create_issue_with_details(title_field_id: str, title: str, assign_to_me_id: str, due_date_field_id: str, due_date: str, submit_button_id: str):
    """Fill issue details and submit the issue.
    
    Args:
        title_field_id: ID of the title input field
        title: Issue title text
        assign_to_me_id: ID of the "Assign to me" link/button
        due_date_field_id: ID of the due date field
        due_date: Due date in YYYY-MM-DD format
        submit_button_id: ID of the submit button
        
    Returns:
        None
        
    Examples:
        create_issue_with_details('478', '404 for many URLs', '586', '687', '2030-1-3', '693')
    """
    fill(title_field_id, title)
    click(assign_to_me_id)
    fill(due_date_field_id, due_date)
    click(submit_button_id)
