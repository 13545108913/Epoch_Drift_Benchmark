def search_and_select_repository(search_box_id: str, repo_name: str, repo_link_id: str):
    """Search for a repository and select it from search results.
    
    Args:
        search_box_id: The ID of the search box element
        repo_name: The name of the repository to search for
        repo_link_id: The ID of the repository link in search results
        
    Returns:
        None
        
    Examples:
        search_and_select_repository('142', 'primer design guide', '284')
    """
    search_repository(search_box_id, repo_name)
    click(repo_link_id)

def navigate_to_issues_and_select(issues_link_id: str, issue_link_id: str):
    """Navigate to issues page and select a specific issue.
    
    Args:
        issues_link_id: The ID of the Issues link
        issue_link_id: The ID of the specific issue link
        
    Returns:
        None
        
    Examples:
        navigate_to_issues_and_select('283', '1073')
    """
    click(issues_link_id)
    click(issue_link_id)

def assign_issue_to_user(edit_assignee_id: str, user_option_id: str):
    """Assign an issue to a specific user.
    
    Args:
        edit_assignee_id: The ID of the Edit link in Assignee section
        user_option_id: The ID of the user option in assignee dropdown
        
    Returns:
        None
        
    Examples:
        assign_issue_to_user('815', '1088')
    """
    click(edit_assignee_id)
    click(user_option_id)
