def create_project(project_name: str, visibility: str, new_project_id: str, name_field_id: str, visibility_id: str, create_button_id: str):
    """Create a new project with specified name and visibility.
    
    Args:
        project_name: Name of the project to create
        visibility: Project visibility ('public' or 'private')
        new_project_id: ID of the "New project" button/link
        name_field_id: ID of the project name input field
        visibility_id: ID of the visibility radio button
        create_button_id: ID of the create project button
        
    Returns:
        None
        
    Examples:
        create_project('AutoAGI', 'public', '223', '259', '296', '310')
        create_project('TestProject', 'private', '223', '259', '297', '310')
    """
    click(new_project_id)  # Click "New project" to start creation
    fill(name_field_id, project_name)  # Fill project name
    click(visibility_id)  # Select visibility (public/private)
    click(create_button_id)  # Click create button

def add_project_members(settings_id: str, members_id: str, search_field_id: str, username: str, add_button_id: str):
    """Add members to an existing project.
    
    Args:
        settings_id: ID of the Settings link/button
        members_id: ID of the Members link/button
        search_field_id: ID of the member search field
        username: Username to search for and add
        add_button_id: ID of the "Add to project" button
        
    Returns:
        None
        
    Examples:
        add_project_members('362', '375', '452', 'primer', '474')
        add_project_members('362', '375', '452', 'john_doe', '474')
    """
    click(settings_id)  # Click Settings to access project settings
    click(members_id)  # Click Members to access member management
    fill(search_field_id, username)  # Search for the username
    click(f"{username}_result")  # Click the search result (assumes result ID format)
    click(add_button_id)  # Click "Add to project" button
