## Reusable Functions
```python
def navigate_to_repository_issues(repo_link_id: str, issues_link_id: str):
    """Navigate from homepage to repository issues page.
    
    Args:
        repo_link_id: The ID of the repository link
        issues_link_id: The ID of the Issues link
        
    Returns:
        None
        
    Examples:
        navigate_to_repository_issues('289', '283')
    """
    click(repo_link_id)  # Click repository link
    click(issues_link_id)  # Click Issues link

def filter_issues_by_labels(search_box_id: str, label_terms: list):
    """Filter repository issues by label terms.
    
    Args:
        search_box_id: The ID of the search/filter textbox
        label_terms: List of label terms to filter by
        
    Returns:
        None
        
    Examples:
        filter_issues_by_labels('540', ['help wanted', 'good first issue'])
    """
    # Construct search query with OR conditions for multiple labels
    search_query = " OR ".join([f'label:"{term}"' for term in label_terms])
    fill(search_box_id, search_query)  # Fill search box with label filter
```
