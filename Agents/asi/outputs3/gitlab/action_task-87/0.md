def create_new_repository(new_project_id: str, project_name: str, create_button_id: str):

    Args:
        new_project_id: The ID of the "New project" button/link
        project_name: The name of the repository to create
        create_button_id: The ID of the "Create project" button
        
    Returns:
        None
        
    Examples:
        create_new_repository('223', 'webagent', '310')

    """
