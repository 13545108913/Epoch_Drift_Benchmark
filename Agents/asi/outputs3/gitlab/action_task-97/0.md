def navigate_to_most_starred_repos(explore_id: str, sort_id: str):
    """Navigate to the most starred repositories page.
    
    Args:
        explore_id: The ID of the "Explore projects" link
        sort_id: The ID of the "Most stars" sorting option
        
    Returns:
        None
        
    Examples:
        navigate_to_most_starred_repos('237', '275')
    """
    click(explore_id)  # Click Explore projects link
    click(sort_id)  # Click Most stars sorting option

def star_repository(star_id: str):
    """Star a single repository by clicking its star icon.
    
    Args:
        star_id: The ID of the star icon for a repository
        
    Returns:
        None
        
    Examples:
        star_repository('317')
        star_repository('346')
    """
    click(star_id)  # Click the star icon to star the repository
