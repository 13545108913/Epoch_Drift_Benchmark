```python
def navigate_to_merge_request(merge_requests_id: str, merge_request_link_id: str):
    """Navigate to a specific merge request page.
    
    Args:
        merge_requests_id: The ID of the Merge Requests section/link
        merge_request_link_id: The ID of the specific merge request link
        
    Returns:
        None
        
    Examples:
        navigate_to_merge_request('306', '784')
    """
    click(merge_requests_id)  # Click Merge Requests section
    click(merge_request_link_id)  # Click specific merge request link

def resolve_merge_conflicts(conflicts_link_id: str, resolve_option_id: str, commit_button_id: str):
    """Resolve merge conflicts in a merge request.
    
    Args:
        conflicts_link_id: The ID of the "Resolve conflicts" link
        resolve_option_id: The ID of the conflict resolution option
        commit_button_id: The ID of the commit button
        
    Returns:
        None
        
    Examples:
        resolve_merge_conflicts('551', '708', '4052')
    """
    click(conflicts_link_id)  # Click "Resolve conflicts" link
    click(resolve_option_id)  # Click conflict resolution option
    click(commit_button_id)  # Click commit button
```
