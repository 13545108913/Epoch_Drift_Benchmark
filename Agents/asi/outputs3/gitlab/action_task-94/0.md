def search_and_select_repository(search_box_id: str | int, repo_name: str, repo_link_id: str | int):
    """Search for a repository and select it from search results.
    
    Args:
        search_box_id: The ID of the search box element
        repo_name: The name of the repository to search for
        repo_link_id: The ID of the repository link in search results
        
    Returns:
        None
        
    Examples:
        search_and_select_repository('241', 'dotfiles', '289')
        search_and_select_repository('123', 'config', '456')
    """
    search_project(search_box_id, repo_name)
    click(repo_link_id)

    """Navigate from repository page to member management section.
