```python
def analyze_top_contributor(attribute: str):
    """Analyze commit history to find top contributor and return requested attribute.
    
    Args:
        attribute: The contributor attribute to return (e.g., 'email', 'name')
        
    Returns:
        None (sends message to user with result)
        
    Examples:
        analyze_top_contributor('email')
        analyze_top_contributor('name')
    """
    commit_elements = page.query_selector_all('.commit-author')
    commit_counts = {}
    
    for element in commit_elements:
        author_name = element.get_attribute('data-author-name')
        author_email = element.get_attribute('data-author-email')
        
        if author_name not in commit_counts:
            commit_counts[author_name] = {'count': 0, 'email': author_email}
        commit_counts[author_name]['count'] += 1
    
    top_contributor = max(commit_counts.items(), key=lambda x: x[1]['count'])
    
    if attribute == 'email':
        message = f"The contributor with the most commits has email address {top_contributor[1]['email']}"
    elif attribute == 'name':
        message = f"The contributor with the most commits is {top_contributor[0]}"
    else:
        message = f"The top contributor is {top_contributor[0]} with email {top_contributor[1]['email']}"
    
    send_msg_to_user(message)
```
