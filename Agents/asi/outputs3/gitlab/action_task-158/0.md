```python
def navigate_to_new_issue(repo_id: str):
    """Navigate from repository page to new issue creation form.
    
    Args:
        repo_id: The ID of the repository link
        
    Returns:
        None
        
    Examples:
        navigate_to_new_issue('289')
    """
    click(repo_id)  # Click repository link
    click('283')  # Click Issues tab
    click('492')  # Click New issue button

def create_issue_with_details(title_field_id: str, title: str, assignee_id: str, due_date_field_id: str, due_date: str, submit_id: str):
    """Fill issue details and submit the issue.
    
    Args:
        title_field_id: The ID of the title input field
        title: The issue title
        assignee_id: The ID of the assignee element
        due_date_field_id: The ID of the due date field
        due_date: The due date in YYYY-MM-DD format
        submit_id: The ID of the submit button
        
    Returns:
        None
        
    Examples:
        create_issue_with_details('479', 'Fix bug', '587', '688', '2024-12-31', '694')
    """
    fill(title_field_id, title)  # Fill issue title
    click(assignee_id)  # Assign to specified user
    fill(due_date_field_id, due_date)  # Set due date
    click(submit_id)  # Submit the issue
```
