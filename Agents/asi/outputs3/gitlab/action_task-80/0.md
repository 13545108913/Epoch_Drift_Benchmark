def navigate_to_profile_settings(profile_id: str, settings_id: str):
    """Navigate from homepage to profile settings page.
    
    Args:
        profile_id: The ID of the profile link/icon
        settings_id: The ID of the settings link
        
    Returns:
        None
        
    Examples:
        navigate_to_profile_settings('186', '202')
    """
    click(profile_id)
    click(settings_id)

def update_website_url(website_field_id: str, url: str, update_button_id: str):
    """Update the website URL in profile settings.
    
    Args:
        website_field_id: The ID of the website URL input field
        url: The URL to set as homepage
        update_button_id: The ID of the update/save button
        
    Returns:
        None
        
    Examples:
        update_website_url('481', 'https://example.com', '512')
    """
    fill(website_field_id, url)
    click(update_button_id)
