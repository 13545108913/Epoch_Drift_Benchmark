```python
def navigate_to_merge_request(project_id: str, mr_tab_id: str, mr_link_id: str):
    """Navigate from project page to specific merge request.
    
    Args:
        project_id: ID of project link
        mr_tab_id: ID of Merge Requests tab/link
        mr_link_id: ID of specific merge request link
        
    Returns:
        None
        
    Examples:
        navigate_to_merge_request('289', '306', '845')
    """
    click(project_id)  # Click project link
    click(mr_tab_id)  # Click Merge Requests tab
    click(mr_link_id)  # Click specific merge request
```

```python
def post_comment(comment_box_id: str, comment_button_id: str, content: str):
    """Post a comment in a merge request discussion.
    
    Args:
        comment_box_id: ID of comment textbox
        comment_button_id: ID of Comment button
        content: Text content to post as comment
        
    Returns:
        None
        
    Examples:
        post_comment('705', '734', 'lgtm')
    """
    fill(comment_box_id, content)  # Fill comment textbox
    click(comment_button_id)  # Click Comment button
```
