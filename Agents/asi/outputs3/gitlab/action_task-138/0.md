```python
def create_private_repository(new_project_id: str, project_name_id: str, create_button_id: str, project_name: str):
    """Create a private repository with specified name.
    
    Args:
        new_project_id: ID of the "New project" button/link
        project_name_id: ID of the project name input field
        create_button_id: ID of the create button
        project_name: Name of the repository to create
        
    Returns:
        None
        
    Examples:
        create_private_repository('223', '259', '310', 'web_agent')
    """
    click(new_project_id)  # Click New project button
    fill(project_name_id, project_name)  # Fill project name
    click(create_button_id)  # Click create button
```

```python
def notify_repository_creation(project_name: str, template_type: str = "blank"):
    """Send success message after repository creation.
    
    Args:
        project_name: Name of the created repository
        template_type: Type of template used (blank, starter, etc.)
        
    Returns:
        None
        
    Examples:
        notify_repository_creation('web_agent', 'blank')
    """
    message = f"Successfully created a private {template_type} repository called '{project_name}'. The repository is now ready for development."
    send_msg_to_user(message)
```
