def search_and_navigate_to_repo(search_id: str | int, repo_name: str):
    """Search for a repository and navigate to its main page.
    
    Args:
        search_id: The ID of the search input element
        repo_name: The name of the repository to search for
        
    Returns:
        None
        
    Examples:
        search_and_navigate_to_repo('241', 'thoughtbot/administrate')
        search_and_navigate_to_repo('142', 'facebook/react')
    """
    fill(search_id, repo_name)
    press(search_id, 'Enter')

def navigate_to_contributors_page():
    """Navigate from repository main page to contributors statistics page.
    
    Args:
        None
        
    Returns:
        None
        
    Examples:
        navigate_to_contributors_page()
    """
    click('253')  # Click Repository link
    click('271')  # Click Contributors link
