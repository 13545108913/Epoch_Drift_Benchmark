```python
def create_issue(issues_link_id: str, new_issue_id: str, title_input_id: str, description_input_id: str, submit_button_id: str, feature: str):
    """Create a new issue with title and description for feature implementation.
    
    Args:
        issues_link_id: ID of the Issues navigation link
        new_issue_id: ID of the "New issue" button
        title_input_id: ID of the issue title input field
        description_input_id: ID of the issue description input field
        submit_button_id: ID of the submit issue button
        feature: The feature name to implement
    
    Returns:
        None
    
    Examples:
        create_issue('283', '491', '478', '521', '693', 'dark mode')
        create_issue('283', '491', '478', '521', '693', 'user profiles')
    """
    click(issues_link_id)  # Navigate to issues page
    click(new_issue_id)  # Click new issue button
    fill(title_input_id, f"Implement {feature} support")  # Fill issue title
    fill(description_input_id, f"Let's discuss the implementation of {feature} for this theme. This should include:\n\n- Requirements and specifications\n- Technical considerations\n- User experience impact\n- Implementation timeline\n- Testing strategy\n\nPlease share your thoughts and suggestions on how we should approach this feature.")  # Fill issue description
    click(submit_button_id)  # Submit the issue
```
