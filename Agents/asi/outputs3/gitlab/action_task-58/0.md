def navigate_to_merge_requests(project_id: str, merge_requests_id: str):
    """Navigate from homepage to merge requests page.
    
    Args:
        project_id: The ID of the project link
        merge_requests_id: The ID of the "Merge Requests" link
        
    Returns:
        None
        
    Examples:
        navigate_to_merge_requests('289', '306')
    """

    click(merge_requests_id)  # Click Merge Requests link

def post_comment_to_merge_request(merge_request_id: str, comment_box_id: str, comment_button_id: str, comment_text: str):
    """Post a comment to a specific merge request.
    
    Args:
        merge_request_id: The ID of the merge request link
        comment_box_id: The ID of the comment textbox
        comment_button_id: The ID of the comment button
        comment_text: The text content to post as comment
        
    Returns:
        None
        
    Examples:
        post_comment_to_merge_request('789', '739', '768', 'Good idea')
    """
    click(merge_request_id)  # Click merge request link
    click(comment_box_id)  # Click comment textbox
    fill(comment_box_id, comment_text)  # Fill comment text
    click(comment_button_id)  # Click comment button
