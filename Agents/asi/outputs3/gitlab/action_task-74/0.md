```python
def update_project_title(settings_id: str, title_field_id: str, save_button_id: str, new_title: str):
    """Update the project site title by navigating to settings, filling the title field, and saving.
    
    Args:
        settings_id: The ID of the Settings link/button
        title_field_id: The ID of the project title input field
        save_button_id: The ID of the Save changes button
        new_title: The new title to set for the project
    
    Returns:
        None
    
    Examples:
        update_project_title('395', '481', '503', 'My New Project')
        update_project_title('settings_link', 'title_input', 'save_btn', 'Updated Site')
    """
    click(settings_id)  # Click Settings to access project settings
    fill(title_field_id, new_title)  # Fill the title field with the new title
    click(save_button_id)  # Click Save changes to apply the update
```
