```python
def navigate_to_commits(repo_link_id: str, repo_id: str, commits_id: str):
    """Navigate from homepage to repository commits page.
    
    Args:
        repo_link_id: ID of repository link
        repo_id: ID of repository tab/link
        commits_id: ID of commits tab/link
        
    Returns:
        None
        
    Examples:
        navigate_to_commits('289', '257', '269')
    """
    click(repo_link_id)  # Click repository link
    click(repo_id)  # Click repository tab
    click(commits_id)  # Click commits tab

def filter_commits_by_date(date_filter_id: str, target_date: str):
    """Filter commits by specific date.
    
    Args:
        date_filter_id: ID of date filter input field
        target_date: Date to filter by (YYYY-MM-DD format)
        
    Returns:
        None
        
    Examples:
        filter_commits_by_date('499', '2023-01-03')
    """
    click(date_filter_id)  # Click date filter
    fill(date_filter_id, target_date)  # Enter target date
```
