def search_repository(search_box_id: str | int, repo_name: str):
    """Search for a repository using the search box.
    
    Args:
        search_box_id: The ID of the search box element
        repo_name: The name of the repository to search for
        
    Returns:
        None
        
    Examples:
        search_repository('241', 'ffmpeg-python')

def navigate_to_repository_members(repo_link_id: str | int, members_link_id: str | int):
    """Navigate to a repository's member management page.
    
    Args:
        repo_link_id: The ID of the repository link in search results
        members_link_id: The ID of the Members link on the repository page
        
    Returns:
        None
        
    Examples:
        navigate_to_repository_members('284', '354')
    """
