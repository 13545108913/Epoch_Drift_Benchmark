```python
def navigate_to_merge_request(project_id: str, merge_requests_id: str, mr_link_id: str):
    """Navigate to a specific merge request in a project.
    
    Args:
        project_id: The ID of the project link
        merge_requests_id: The ID of the Merge Requests link
        mr_link_id: The ID of the specific merge request link
        
    Returns:
        None
        
    Examples:
        navigate_to_merge_request('357', '307', '784')
    """
    click(project_id)  # Click project link
    click(merge_requests_id)  # Click Merge Requests link
    click(mr_link_id)  # Click specific merge request link

def post_comment(comment_box_id: str, comment_button_id: str, write_button_id: str, content: str):
    """Post a comment in a merge request discussion.
    
    Args:
        comment_box_id: The ID of the comment textbox
        comment_button_id: The ID of the Comment button
        write_button_id: The ID of the Write button
        content: The comment content to post
        
    Returns:
        None
        
    Examples:
        post_comment('671', '700', '640', 'Thanks, working on reviews')
    """
    fill(comment_box_id, content)  # Type comment in textbox
    click(write_button_id)  # Click Write button to switch mode
    click(comment_button_id)  # Click Comment button to submit
```
