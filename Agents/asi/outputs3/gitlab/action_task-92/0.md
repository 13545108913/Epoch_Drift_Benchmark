```python
def search_and_select_repo(search_id: str | int, repo_name: str):
    """Search for and select a repository from the search results.
    
    Args:
        search_id: The ID of the search input field
        repo_name: The name of the repository to search for
        
    Returns:
        None
        
    Examples:
        search_and_select_repo('241', 'dotfiles')
        search_and_select_repo('241', 'config-files')
    """
    search_project(search_id, repo_name)
    click('289')  # Click on the first repository result
```
