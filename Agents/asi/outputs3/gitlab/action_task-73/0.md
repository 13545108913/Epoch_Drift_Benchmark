```python
def edit_html_file(file_id: str, edit_button_id: str, content_area_id: str, new_title: str):
    """Open and edit an HTML file to update the title tag.
    
    Args:
        file_id: The ID of the HTML file to edit
        edit_button_id: The ID of the edit button
        content_area_id: The ID of the content editing area
        new_title: The new title text to set in the title tag
        
    Returns:
        None
        
    Examples:
        edit_html_file('785', '548', '603', 'Welcome to my site')
        edit_html_file('index.html', 'edit-btn', 'content-area', 'My Project Site')
    """
    click(file_id)  # Click on the HTML file to open it
    click(edit_button_id)  # Click the edit button to enter edit mode
    fill(content_area_id, f'<title>{new_title}</title>')  # Update the title tag content
```
