```python
def set_gitlab_status(profile_id: str, edit_button_id: str, status_textbox_id: str, set_button_id: str, status_message: str):
    """Set GitLab user status with a custom message.
    
    Args:
        profile_id: ID of the user profile link/button
        edit_button_id: ID of the "Edit status" button
        status_textbox_id: ID of the status text input field
        set_button_id: ID of the "Set status" button
        status_message: The status message to set
        
    Returns:
        None
        
    Examples:
        set_gitlab_status('186', '198', '820', '826', 'Playing Badminton')
        set_gitlab_status('186', '198', '820', '826', 'Working from home')
    """
    click(profile_id)  # Click user profile to access settings
    click(edit_button_id)  # Click "Edit status" button
    click(status_textbox_id)  # Click status textbox to focus
    fill(status_textbox_id, status_message)  # Fill status message
    click(set_button_id)  # Click "Set status" to save
```
