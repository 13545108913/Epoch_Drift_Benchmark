```python
def navigate_to_issues(repo_link_id: str, issues_link_id: str):
    """Navigate from repository page to issues page.
    
    Args:
        repo_link_id: The ID of the repository link
        issues_link_id: The ID of the issues link
        
    Returns:
        None
        
    Examples:
        navigate_to_issues('385', '283')
    """
    click(repo_link_id)  # Click repository link
    click(issues_link_id)  # Click issues link

def assign_issue(issue_link_id: str, edit_assignee_id: str, assignee_id: str):
    """Assign a specific issue to a user.
    
    Args:
        issue_link_id: The ID of the issue link
        edit_assignee_id: The ID of the edit assignee button
        assignee_id: The ID of the assignee in dropdown
        
    Returns:
        None
        
    Examples:
        assign_issue('810', '809', '1063')
    """
    click(issue_link_id)  # Click on the specific issue
    click(edit_assignee_id)  # Click edit assignee button
    click(assignee_id)  # Click on the assignee in dropdown
```
