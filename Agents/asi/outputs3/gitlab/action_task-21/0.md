## Reusable Functions
```python
def navigate_to_issues(issues_link_id: str):
    """Navigate to the issues page by clicking the issues link.
    
    Args:
        issues_link_id: The ID of the issues link in the sidebar
        
    Returns:
        None
        
    Examples:
        navigate_to_issues('161')
    """
    click(issues_link_id)  # Click the Issues link to go to issues page


def filter_issues(filter_id: str):
    """Apply a filter to the issues list.
    
    Args:
        filter_id: The ID of the filter button/link
        
    Returns:
        None
        
    Examples:
        filter_issues('253')
    """
    click(filter_id)  # Click the filter to apply it


def check_issue_status_and_report(issue_title: str, status: str):
    """Check issue status and send message to user.
    
    Args:
        issue_title: The title of the issue
        status: The current status of the issue
        
    Returns:
        None
        
    Examples:
        check_issue_status_and_report('Replace mailto links with a contact form', 'closed')
    """
    message = f"Your latest created issue with '{keyword}' in its title is '{issue_title}'. I can see from the issues list that this issue is currently {status}."
    send_msg_to_user(message)
```
