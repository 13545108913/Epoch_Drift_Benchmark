## Reusable Functions
```python
def search_project(search_box_id: str | int, project_name: str):
    """Search for a project using the search box.
    
    Args:
        search_box_id: The ID of the search box element
        project_name: The name of the project to search for
        
    Returns:
        None
        
    Examples:
        search_project('241', 'MetaSeq')
        search_project('141', 'MyProject')
    """
    click(search_box_id)
    fill(search_box_id, project_name)
    keyboard_press("Enter")

def navigate_to_project_page(project_link_id: str | int):
    """Navigate to a project's main page by clicking its link.
    
    Args:
        project_link_id: The ID of the project link element
        
    Returns:
        None
        
    Examples:
        navigate_to_project_page('284')
        navigate_to_project_page('123')
    """
    click(project_link_id)

def fork_project(fork_button_id: str | int, namespace_id: str | int):
    """Fork a project by clicking the fork button and selecting namespace.
    
    Args:
        fork_button_id: The ID of the fork button element
        namespace_id: The ID of the namespace to fork into
        
    Returns:
        None
        
    Examples:
        fork_project('475', '415')
        fork_project('500', '600')
    """
    click(fork_button_id)
    click(namespace_id)
```
