```python
def navigate_to_contributors(commits_id: str, contributors_id: str):
    """Navigate from repository page to contributors statistics page.
    
    Args:
        commits_id: The ID of the commits link
        contributors_id: The ID of the contributors link
        
    Returns:
        None
        
    Examples:
        navigate_to_contributors('522', '270')
    """
    click(commits_id)  # Click commits link to access commit history
    click(contributors_id)  # Click contributors link to access contributor stats
```

```python
def get_top_contributor_info(attribute: str):
    """Get the specified attribute of the top contributor.
    
    Args:
        attribute: The attribute to retrieve ('name', 'number of commits', etc.)
        
    Returns:
        str: The requested attribute value
        
    Examples:
        get_top_contributor_info('name')
        get_top_contributor_info('number of commits')
    """
    # Extract top contributor information from the page
    top_contributor_name = extract_text('top_contributor_name_element')
    top_contributor_commits = extract_text('top_contributor_commits_element')
    
    if attribute == 'name':
        return top_contributor_name
    elif attribute == 'number of commits':
        return top_contributor_commits
    else:
        return f"Unknown attribute: {attribute}"
```
