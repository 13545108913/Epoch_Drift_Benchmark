```python
def sort_repos_by_stars(dropdown_id: str, option_id: str):
    """Sort repositories by star count in descending order.
    
    Args:
        dropdown_id: The ID of the sorting dropdown element
        option_id: The ID of the "Most stars" option in the dropdown
        
    Returns:
        None
        
    Examples:
        sort_repos_by_stars('244', '260')
    """
    click(dropdown_id)  # Click the sorting dropdown
    click(option_id)  # Select "Most stars" option
```

```python
def star_repository(repo_star_button_id: str, repo_name: str):
    """Star a repository and send confirmation message.
    
    Args:
        repo_star_button_id: The ID of the star button for the repository
        repo_name: The name of the repository being starred
        
    Returns:
        None
        
    Examples:
        star_repository('302', 'The A11Y Project / a11yproject.com')
    """
    click(repo_star_button_id)  # Click the star button
    send_msg_to_user(f"Successfully starred '{repo_name}' which is one of the top most starred repositories.")
```
