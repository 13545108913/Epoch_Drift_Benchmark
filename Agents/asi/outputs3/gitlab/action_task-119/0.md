## Reusable Functions
```python
def navigate_to_issues_page(project_link_id: str | int, issues_link_id: str | int):
    """Navigate from homepage to issues page of a repository.
    
    Args:
        project_link_id: The ID of the project/repository link
        issues_link_id: The ID of the Issues tab/link
        
    Returns:
        None
        
    Examples:
        navigate_to_issues_page('434', '283')
    """
    click(project_link_id)  # Click on the project/repository link
    click(issues_link_id)  # Click on the Issues tab/link

def create_new_issue(new_issue_button_id: str | int, title_field_id: str | int, description_field_id: str | int, title: str, description: str):
    """Create a new issue with title and description.
    
    Args:
        new_issue_button_id: The ID of the "New issue" button
        title_field_id: The ID of the title input field
        description_field_id: The ID of the description input field
        title: The issue title text
        description: The issue description text
        
    Returns:
        None
        
    Examples:
        create_new_issue('476', '478', '521', 'Bug Report', 'Detailed bug description...')
    """
    click(new_issue_button_id)  # Click the "New issue" button
    fill(title_field_id, title)  # Fill in the issue title
    fill(description_field_id, description)  # Fill in the issue description
    keyboard_press("Enter")  # Submit the issue
```
