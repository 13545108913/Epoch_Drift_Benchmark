def navigate_to_project_members(project_link_id: str, settings_link_id: str, members_link_id: str):
    """Navigate from homepage to project members management page.
    
    Args:
        project_link_id: The ID of the project link
        settings_link_id: The ID of the Settings link
        members_link_id: The ID of the Members link
        
    Returns:
        None
        
    Examples:
        navigate_to_project_members('824', '395', '407')
    """
    click(project_link_id)  # Click project link
    click(settings_link_id)  # Click Settings link
    click(members_link_id)  # Click Members link

def invite_collaborator(search_box_id: str, username: str, dropdown_id: str, add_button_id: str):
    """Search for and invite a collaborator to the project.
    
    Args:
        search_box_id: The ID of the member search box
        username: The username to invite
        dropdown_id: The ID of the dropdown option for the user
        add_button_id: The ID of the "Add to project" button
        
    Returns:
        None

        invite_collaborator('490', 'yjlou', '646', '512')
    """

    click(dropdown_id)  # Click dropdown option
    click(add_button_id)  # Click add button
