```python
def navigate_to_profile_settings(profile_id: str, settings_id: str):
    """Navigate from homepage to profile settings page.
    
    Args:
        profile_id: The ID of the profile link
        settings_id: The ID of the settings link
        
    Returns:
        None
        
    Examples:
        navigate_to_profile_settings('185', '202')
    """
    click(profile_id)  # Click profile link
    click(settings_id)  # Click settings link

def update_website_url(website_field_id: str, url: str, update_button_id: str):
    """Update website URL in profile settings.
    
    Args:
        website_field_id: The ID of the website URL input field
        url: The URL to set as homepage
        update_button_id: The ID of the update button
        
    Returns:
        None
        
    Examples:
        update_website_url('481', 'a11yproject.contributor.me', '512')
    """
    fill(website_field_id, url)  # Fill website URL field
    click(update_button_id)  # Click update button
```
