def update_project_title(settings_id: str, title_input_id: str, save_button_id: str, new_title: str):
    """Update the project title by navigating to settings, changing the title, and saving.
    
    Args:
        settings_id: The ID of the Settings link/button
        title_input_id: The ID of the project title input field
        save_button_id: The ID of the Save changes button
        new_title: The new title to set for the project
        
    Returns:
        None
        
    Examples:
        update_project_title('395', '481', '503', 'My New Project')
    """

    click(title_input_id)  # Click the title input field
    fill(title_input_id, new_title)  # Fill with new title
    click(save_button_id)  # Save the changes
