```python
def search_and_select_project(search_box_id: str, project_name: str, filter_id: str = None):
    """Search for a project and optionally apply filters.
    
    Args:
        search_box_id: The ID of the search box element
        project_name: The name of the project to search for
        filter_id: Optional ID of filter element to click (e.g., "All" filter)
        
    Returns:
        None
        
    Examples:
        search_and_select_project('242', 'ChatGPT', '276')
        search_and_select_project('242', 'my-project')
    """
    search_project(search_box_id, project_name)  # Search for the project
    if filter_id:
        click(filter_id)  # Apply filter if provided
```

```python
def generate_ssh_clone_command(repo_name: str, base_url: str = "172.26.116.102"):
    """Generate SSH clone command for a repository.
    
    Args:
        repo_name: The name of the repository
        base_url: The GitLab instance base URL
        
    Returns:
        str: The SSH clone command
        
    Examples:
        generate_ssh_clone_command("chatgpt")
        generate_ssh_clone_command("my-project", "gitlab.company.com")
    """
    command = f"git clone git@{base_url}:username/{repo_name.lower()}.git"
    return command
```
