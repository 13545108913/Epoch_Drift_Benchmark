## Reusable Functions
```python
def navigate_to_project(project_id: str):
    """Navigate to a specific project page by clicking its link.
    
    Args:
        project_id: The ID of the project link to click
        
    Returns:
        None
        
    Examples:
        navigate_to_project('526')
        navigate_to_project('594')
    """
    click(project_id)  # Click on the project link to navigate to its page


def navigate_to_dashboard(dashboard_id: str):
    """Navigate back to the main dashboard from any page.
    
    Args:
        dashboard_id: The ID of the dashboard link
        
    Returns:
        None
        
    Examples:
        navigate_to_dashboard('53')
    """
    click(dashboard_id)  # Click on the dashboard link to return to main page


def check_and_report_star_status(repo_names: list, star_counts: list):
    """Check if repositories are starred and report their status.
    
    Args:
        repo_names: List of repository names
        star_counts: List of star counts for each repository
        
    Returns:
        None
        
    Examples:
        check_and_report_star_status(['repo1', 'repo2'], [10, 5])
    """
    message = f"The top {len(repo_names)} most starred repositories are already starred: "
    repo_info = []
    for name, stars in zip(repo_names, star_counts):
        repo_info.append(f"'{name}' ({stars} stars)")
    message += ', '.join(repo_info) + "."
    send_msg_to_user(message)
```
