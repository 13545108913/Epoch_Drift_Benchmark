def search_and_select_repository(search_box_id: str, repo_name: str, result_link_id: str):
    """Search for a repository and select it from search results.
    
    Args:
        search_box_id: The ID of the search box element
        repo_name: The name of the repository to search for
        result_link_id: The ID of the repository link in search results
        
    Returns:
        None
        
    Examples:
        search_and_select_repository('143', 'ChatGPT', '285')
    """
    click(search_box_id)
    fill(search_box_id, repo_name)
    keyboard_press("Enter")
    click(result_link_id)

def fork_repository(fork_button_id: str, namespace_id: str):
    """Fork a repository to a specific namespace.
    
    Args:
        fork_button_id: The ID of the fork button
        namespace_id: The ID of the namespace to fork to
        
    Returns:
        None
        
    Examples:
        fork_repository('476', '416')
    """
    click(fork_button_id)
    click(namespace_id)
