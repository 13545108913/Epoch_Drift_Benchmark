```python
def search_repository(search_box_id: str, repo_name: str):
    """Search for a repository using the search box.
    
    Args:
        search_box_id: The ID of the search box element
        repo_name: The name of the repository to search for
        
    Returns:
        None
        
    Examples:
        search_repository('143', 'metaseq')
        search_repository('search_box', 'django')
    """
    click(search_box_id)
    fill(search_box_id, repo_name)
    keyboard_press("Enter")

def fork_repository(repo_link_id: str, fork_button_id: str, namespace_id: str):
    """Fork a repository to a specific namespace.
    
    Args:
        repo_link_id: The ID of the repository link
        fork_button_id: The ID of the fork button
        namespace_id: The ID of the namespace to fork to
        
    Returns:
        None
        
    Examples:
        fork_repository('285', '476', '416')
        fork_repository('repo_link', 'fork_btn', 'my_namespace')
    """
    click(repo_link_id)
    click(fork_button_id)
    click(namespace_id)
```
