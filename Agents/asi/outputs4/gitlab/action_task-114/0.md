def navigate_to_new_issue(repo_id: str, issues_id: str, new_issue_id: str):
    """Navigate to the new issue creation page.
    
    Args:
        repo_id: The ID of the repository link
        issues_id: The ID of the Issues link
        new_issue_id: The ID of the New issue button
        
    Returns:
        None
        
    Examples:
        navigate_to_new_issue('526', '283', '492')
    """
    click(repo_id)  # Click repository link
    click(issues_id)  # Click Issues link
    click(new_issue_id)  # Click New issue button

def create_issue_with_title(title_field_id: str, title: str):
    """Fill in the issue title field.
    
    Args:
        title_field_id: The ID of the title input field
        title: The issue title text
        
    Returns:
        None
        
    Examples:
        create_issue_with_title('479', '401 bad gateway')
    """
    fill(title_field_id, title)  # Fill issue title

def assign_issue_to_user(assignee_dropdown_id: str, assignee_option_id: str):
    """Assign an issue to a specific user.
    
    Args:
        assignee_dropdown_id: The ID of the assignee dropdown
        assignee_option_id: The ID of the assignee option
        
    Returns:
        None
        
    Examples:
        assign_issue_to_user('572', '713')
    """
    click(assignee_dropdown_id)  # Click assignee dropdown
    click(assignee_option_id)  # Select assignee from dropdown

def set_issue_due_date(due_date_field_id: str, due_date: str):
    """Set the due date for an issue.
    
    Args:
        due_date_field_id: The ID of the due date input field
        due_date: The due date in YYYY-MM-DD format
        
    Returns:
        None
        
    Examples:
        set_issue_due_date('688', '2030-12-31')
    """
    fill(due_date_field_id, due_date)  # Fill due date field
