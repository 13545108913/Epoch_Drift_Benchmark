def search_and_select_repository(search_box_id: str, repo_name: str, repo_result_id: str):
    """Search for and select a repository from the search results.
    
    Args:
        search_box_id: The ID of the search box element
        repo_name: The name of the repository to search for
        repo_result_id: The ID of the repository search result to click
        
    Returns:
        None
        
    Examples:
        search_and_select_repository('143', 'cloud-to-butt', '923')
    """
    click(search_box_id)
    fill(search_box_id, repo_name)
    click(repo_result_id)

def navigate_to_new_issue(issues_link_id: str, new_issue_link_id: str):
    """Navigate from repository page to new issue creation page.
    
    Args:
        issues_link_id: The ID of the issues link
        new_issue_link_id: The ID of the new issue link
        
    Returns:
        None
        
    Examples:
        navigate_to_new_issue('306', '439')
    """
    click(issues_link_id)
    click(new_issue_link_id)

def create_issue_with_details(title_field_id: str, title: str, assign_to_me_id: str, due_date_field_id: str, due_date: str, submit_button_id: str):
    """Fill in issue details and submit the issue.
    
    Args:
        title_field_id: The ID of the title input field
        title: The issue title text
        assign_to_me_id: The ID of the "Assign to me" link
        due_date_field_id: The ID of the due date input field
        due_date: The due date in YYYY-MM-DD format
        submit_button_id: The ID of the submit button
        
    Returns:
        None
        
    Examples:
        create_issue_with_details('441', 'Bug fix', '549', '650', '2024-12-31', '656')
    """
    fill(title_field_id, title)
    click(assign_to_me_id)
    fill(due_date_field_id, due_date)
    click(submit_button_id)
