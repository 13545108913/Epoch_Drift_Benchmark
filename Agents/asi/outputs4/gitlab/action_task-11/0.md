```python
def navigate_to_commit_history(repo_link_id: str):
    """Navigate from repository page to commit history page.
    
    Args:
        repo_link_id: The ID of the repository link
        
    Returns:
        None
        
    Examples:
        navigate_to_commit_history('290')
    """
    click(repo_link_id)  # Click repository link
    click("657")  # Click History link
```

```python
def filter_commits_by_date(date_input_id: str, date: str):
    """Filter commits by specific date.
    
    Args:
        date_input_id: The ID of the date search input field
        date: The date to filter by (format: YYYY-MM-DD)
        
    Returns:
        None
        
    Examples:
        filter_commits_by_date('500', '2023-01-03')
    """
    fill(date_input_id, date)  # Fill date in search box
    keyboard_press("Enter")  # Press Enter to execute search
```

```python
def switch_branch(branch_selector_id: str, branch_name_id: str):
    """Switch to a different branch in the repository.
    
    Args:
        branch_selector_id: The ID of the branch selector button
        branch_name_id: The ID of the specific branch to select
        
    Returns:
        None
        
    Examples:
        switch_branch('477', '1466')
    """
    click(branch_selector_id)  # Click branch selector
    click(branch_name_id)  # Click specific branch
```
