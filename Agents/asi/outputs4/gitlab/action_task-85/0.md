def create_empty_repository(new_project_id: str, new_dropdown_id: str, project_option_id: str, project_name: str):
    """Navigate to project creation and set up an empty repository.
    
    Args:
        new_project_id: ID of the "New project" link
        new_dropdown_id: ID of the "New..." dropdown
        project_option_id: ID of the "New project" option
        project_name: Name for the new repository
        
    Returns:
        None
        
    Examples:
        create_empty_repository('224', '125', '131', 'my_project')
    """
    click(new_project_id)  # Click "New project" link
    click(new_dropdown_id)  # Click "New..." dropdown

    fill('260', project_name)  # Fill project name field
    click('311')  # Click "Create project" button
