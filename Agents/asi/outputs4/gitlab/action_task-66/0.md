def check_and_update_license(current_license: str, target_license: str = 'MIT'):
    """Check if current license matches target license and send appropriate message.
    
    Args:
        current_license: The current license text
        target_license: The desired license type
        
    Returns:
        None
        
    Examples:
        check_and_update_license('MIT License', 'MIT')
    """
    if target_license.lower() in current_license.lower():
        send_msg_to_user(f"The LICENSE file is already set to {target_license} license. No changes are needed.")

        send_msg_to_user(f"The LICENSE file needs to be updated to {target_license} license.")
