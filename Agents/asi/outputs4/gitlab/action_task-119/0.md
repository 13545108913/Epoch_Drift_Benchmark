```python
def navigate_to_new_issue(repo_link_id: str, issues_link_id: str):
    """Navigate from repository page to new issue creation page.
    
    Args:
        repo_link_id: The ID of the repository link
        issues_link_id: The ID of the Issues tab/link
        
    Returns:
        None
        
    Examples:
        navigate_to_new_issue('429', '283')
    """
    click(repo_link_id)  # Click repository link
    click(issues_link_id)  # Click Issues tab
    click('477')  # Click "New issue" button
```

```python
def create_issue(title: str, description: str):
    """Create a new issue with title and description.
    
    Args:
        title: The issue title text
        description: The issue description text
        
    Returns:
        None
        
    Examples:
        create_issue('Bug Report', 'Found a bug in the login flow')
    """
    fill('479', title)  # Fill issue title field
    fill('522', description)  # Fill issue description field

```

```python
def notify_issue_created(issue_title: str):
    """Send notification that issue was successfully created.
    
    Args:
        issue_title: The title of the created issue
        
    Returns:
        None
        
    Examples:
        notify_issue_created('Feature Request: Dark Mode')
    """
    message = f"Successfully created a new issue titled '{issue_title}' in the metaseq repository. The issue is now open and visible on the issues page."
    send_msg_to_user(message)
```
