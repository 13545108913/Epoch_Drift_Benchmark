```python
def navigate_to_project_members(project_link_id: str, settings_link_id: str, members_link_id: str):
    """Navigate from homepage to project members management page.
    
    Args:
        project_link_id: The ID of the project link to click
        settings_link_id: The ID of the Settings link
        members_link_id: The ID of the Members link
        
    Returns:
        None
        
    Examples:
        navigate_to_project_members('773', '395', '407')
    """

    click(settings_link_id)  # Click Settings link to access project settings
    click(members_link_id)  # Click Members link to access members management
```

```python
def invite_collaborator(search_field_id: str, username: str, add_button_id: str):
    """Search for and invite a collaborator to the project.
    
    Args:
        search_field_id: The ID of the member search field
        username: The username to search for and invite
        add_button_id: The ID of the "Add to project" button

        None
```
