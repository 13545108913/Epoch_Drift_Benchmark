    """Navigate from repository page to commit history page.
    
    Args:
        repo_id: The ID of the repository link
        history_id: The ID of the "History" link
        
    Returns:
        None
        
    Examples:
        navigate_to_repo_commits('284', '657')
    """
    click(repo_id)  # Click repository link
    click(history_id)  # Click history link to access commit history


def filter_commits_by_date(date_filter_id: str, target_date: str):
    """Filter commit history by specific date.
    
    Args:
        date_filter_id: The ID of the date filter input field
        target_date: The date to filter by (format: YYYY-MM-DD)
        
    Returns:
        None
        
    Examples:
        filter_commits_by_date('500', '2024-03-02')
    """
    fill(date_filter_id, target_date)  # Fill date filter with target date
