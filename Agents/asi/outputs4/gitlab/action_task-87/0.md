```python
def create_new_project(new_project_id: str, new_dropdown_id: str, project_name_field_id: str, create_button_id: str, project_name: str):
    """Create a new empty project/repository with the given name.
    
    Args:
        new_project_id: ID of the "New project" link
        new_dropdown_id: ID of the "New..." dropdown link
        project_name_field_id: ID of the project name input field
        create_button_id: ID of the "Create project" button
        project_name: Name for the new project/repository
        
    Returns:
        None
        
    Examples:
        create_new_project('224', '125', '260', '311', 'webagent')
    """
    click(new_project_id)  # Click "New project" link
    click(new_dropdown_id)  # Click "New..." dropdown
    click(project_name_field_id)  # Click project name field
    fill(project_name_field_id, project_name)  # Fill project name
    click(create_button_id)  # Click "Create project" button
```
