```python
def navigate_to_profile_settings(profile_id: str, settings_id: str):
    """Navigate from the current page to profile settings page.
    
    Args:
        profile_id: The ID of the user profile link/icon
        settings_id: The ID of the Settings link in the dropdown
        
    Returns:
        None
        
    Examples:
        navigate_to_profile_settings('187', '203')
    """
    click(profile_id)  # Click user profile link/icon
    click(settings_id)  # Click Settings link in dropdown

def update_profile_url(url_field_id: str, update_button_id: str, url: str):
    """Update the homepage URL in profile settings and save changes.
    
    Args:
        url_field_id: The ID of the website URL input field
        update_button_id: The ID of the update/save button
        url: The URL to set as homepage
        
    Returns:
        None
        
    Examples:
        update_profile_url('482', '513', 'https://example.com')
    """
    fill(url_field_id, url)  # Fill the website URL field
    click(update_button_id)  # Click update button to save changes
```
