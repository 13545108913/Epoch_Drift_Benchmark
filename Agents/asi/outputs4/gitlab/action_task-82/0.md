```python
def update_gitlab_profile_homepage(profile_id: str, settings_id: str, url_field_id: str, update_button_id: str, new_url: str):
    """Update the homepage URL in GitLab profile settings.
    
    Args:
        profile_id: ID of the user profile link
        settings_id: ID of the Settings link in profile dropdown
        url_field_id: ID of the Website URL input field
        update_button_id: ID of the Update profile settings button
        new_url: The new homepage URL to set
        
    Returns:
        None
        
    Examples:
        update_gitlab_profile_homepage('187', '203', '482', '513', 'www.example.com')
    """
    click(profile_id)  # Click user profile link
    click(settings_id)  # Click Settings link
    fill(url_field_id, new_url)  # Fill website URL field
    click(update_button_id)  # Click update button
```
