```python
def search_and_select_repository(search_box_id: str | int, repo_name: str):
    """Search for a repository and select it from search results.
    
    Args:
        search_box_id: The ID of the search box element
        repo_name: The name of the repository to search for
        
    Returns:
        None
        
    Examples:
        search_and_select_repository('242', 'keycloak')
        search_and_select_repository('143', 'keycloak/keycloak')
    """
    search_project(search_box_id, repo_name)  # Search for the repository
    noop(1000)  # Wait for search results to load
    click('284')  # Click on the repository link from search results
    noop(1500)  # Wait for repository page to load
```

```python
def filter_issues_by_label(issues_link_id: str | int, search_box_id: str | int, repo_name: str, label_pattern: str):
    """Navigate to issues and filter by label pattern.
    
    Args:
        issues_link_id: The ID of the Issues link
        search_box_id: The ID of the issues search box
        repo_name: The repository name
        label_pattern: The label pattern to filter by (e.g., '~flaky-test')
        
    Returns:
        None
        
    Examples:
        filter_issues_by_label('260', '212', 'keycloak/keycloak', '~flaky-test')
    """
    click(issues_link_id)  # Click on Issues link
    fill(search_box_id, f'{repo_name} label:{label_pattern}')  # Fill search with repo and label filter
    keyboard_press("Enter")  # Execute the search
```
