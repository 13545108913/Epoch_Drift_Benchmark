```python
def search_and_open_issue_by_keyword(issues_link_id: str | int, keyword: str):
    """Search for and open the most recent issue containing a keyword in its title.
    
    Args:
        issues_link_id: The ID of the Issues link/button
        keyword: The keyword to search for in issue titles
        
    Returns:
        None
        
    Examples:
        search_and_open_issue_by_keyword('162', 'feature')
        search_and_open_issue_by_keyword('162', 'bug')
    """
    click(issues_link_id)  # Click Issues link to navigate to issues page
    # Find and click the most recent issue containing the keyword in title
    issue_element = page.get_by_role("link").filter(has_text=keyword).first
    issue_element.click()
```

```python
def check_issue_status_and_report():
    """Check if the current issue is open or closed and report the status.
    
    Args:
        None
        
    Returns:
        None
        
    Examples:
        check_issue_status_and_report()
    """
    # Check for "Open" status label
    open_label = page.get_by_text("Open", exact=True)
    close_button = page.get_by_role("button", name="Close issue")
    
    if open_label.is_visible() and close_button.is_visible():
        issue_title = page.get_by_role("heading").first.text_content()
        send_msg_to_user(f"The issue '{issue_title}' is currently open, not closed. I can see the 'Open' status label and a 'Close issue' button available.")
    else:
        issue_title = page.get_by_role("heading").first.text_content()
        send_msg_to_user(f"The issue '{issue_title}' appears to be closed. I don't see the 'Open' status label or the 'Close issue' button is not available.")
```
