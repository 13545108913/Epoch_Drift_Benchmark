## Reusable Functions
```python
def navigate_to_repository_issues(repo_link_id: str, issues_link_id: str):
    """Navigate from homepage to repository issues page.
    
    Args:
        repo_link_id: The ID of the repository link
        issues_link_id: The ID of the issues link
        
    Returns:
        None
        
    Examples:
        navigate_to_repository_issues('290', '283')
    """
    click(repo_link_id)  # Click repository link
    click(issues_link_id)  # Click issues link

def filter_issues_by_label(search_field_id: str, label_name: str):
    """Filter issues by specific label in the issues page.
    
    Args:
        search_field_id: The ID of the search/filter field
        label_name: The label name to filter by
        
    Returns:
        None
        
    Examples:
        filter_issues_by_label('541', 'help wanted')
    """
    fill(search_field_id, f"label_name[]={label_name}")  # Fill label filter
    keyboard_press("Enter")  # Apply the filter
```
