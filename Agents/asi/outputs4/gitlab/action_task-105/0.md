```python
def navigate_to_repo_members(repo_id: str, settings_id: str, members_id: str):
    """Navigate from repository page to members management section.
    
    Args:
        repo_id: The ID of the repository link
        settings_id: The ID of the Settings link
        members_id: The ID of the Members link
        
    Returns:
        None
        
    Examples:
        navigate_to_repo_members('560', '395', '407')
    """
    click(repo_id)  # Click repository link
    click(settings_id)  # Click Settings link
    click(members_id)  # Click Members link

def add_user_to_repo(search_box_id: str, user_name: str, role_dropdown_id: str, role: str, add_button_id: str):
    """Add a user to repository with specified role.
    
    Args:
        search_box_id: The ID of the user search box
        user_name: The username to search for
        role_dropdown_id: The ID of the role selection dropdown
        role: The role to assign (e.g., 'Developer')
        add_button_id: The ID of the "Add to project" button
        
    Returns:
        None
        
    Examples:
        add_user_to_repo('491', 'abisubramanya27', '499', 'Developer', '513')
    """
    fill(search_box_id, user_name)  # Search for user
    click(f"{user_name}_result")  # Click user from search results (assumes result ID format)
    select_option(role_dropdown_id, role)  # Select role from dropdown
    click(add_button_id)  # Click add button
```
