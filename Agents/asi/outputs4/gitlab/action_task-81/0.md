```python
def update_gitlab_profile_url(profile_dropdown_id: str, website_field_id: str, url: str):
    """Update the website URL in GitLab profile settings.
    
    Args:
        profile_dropdown_id: The ID of the profile dropdown menu
        website_field_id: The ID of the website URL input field
        url: The URL to set as homepage
        
    Returns:
        None
        
    Examples:
        update_gitlab_profile_url('187', '482', 'example.com')
    """
    click(profile_dropdown_id)  # Click profile dropdown to access settings
    fill(website_field_id, url)  # Fill website URL field with provided URL
    click('513')  # Click "Update profile settings" button
```
