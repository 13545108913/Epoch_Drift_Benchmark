def search_project(search_bar_id: str, project_name: str):
    """Search for a project using the search bar.
    
    Args:
        search_bar_id: The ID of the search bar element
        project_name: The name of the project to search for
        
    Returns:
        None
        
    Examples:
        search_project('143', 'time')
        search_project('search_box', 'marketing')
    """
    click(search_bar_id)
    fill(search_bar_id, project_name)
    press(search_bar_id, 'Enter')

def navigate_to_project_settings(project_id: str):
    """Navigate to a project's settings page.
    
    Args:
        project_id: The ID of the project link/element
        
    Returns:
        None
        
    Examples:
        navigate_to_project_settings('403')
        navigate_to_project_settings('project_123')
    """
    click(project_id)
    click("395")
    click("407")

def add_user_to_project(user_name: str, add_button_id: str):
    """Add a specific user to the current project.
    
    Args:
        user_name: The name of the user to add
        add_button_id: The ID of the "Add to project" button
        
    Returns:
        None
        
    Examples:
        add_user_to_project('yjlou', '513')
        add_user_to_project('alice', 'add_btn')
    """
    click("607")
    click("615")
    click(add_button_id)
