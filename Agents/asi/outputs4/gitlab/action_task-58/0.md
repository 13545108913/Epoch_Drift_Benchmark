```python
def navigate_to_merge_request(project_id: str, merge_requests_id: str, target_mr_id: str):
    """Navigate from project page to specific merge request page.
    
    Args:
        project_id: The ID of the project link
        merge_requests_id: The ID of the "Merge Requests" link
        target_mr_id: The ID of the specific merge request
        
    Returns:
        None
        
    Examples:
        navigate_to_merge_request('284', '307', '790')
    """
    click(project_id)  # Click project link
    click(merge_requests_id)  # Click Merge Requests link
    click(target_mr_id)  # Click specific merge request
```

```python
def post_comment_to_mr(write_button_id: str, comment_box_id: str, comment_button_id: str, content: str):
    """Post a comment to a merge request.
    
    Args:
        write_button_id: The ID of the "Write" button
        comment_box_id: The ID of the comment input box
        comment_button_id: The ID of the comment submit button
        content: The comment content to post
        
    Returns:
        None
        
    Examples:
        post_comment_to_mr('709', '740', '769', 'Good idea')
    """
    click(write_button_id)  # Click Write button to activate comment area
    fill(comment_box_id, content)  # Fill comment box with content
    click(comment_button_id)  # Click Comment button to submit
    keyboard_press("Enter")  # Press Enter to ensure submission
```
